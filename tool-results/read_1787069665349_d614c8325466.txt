    38→// Baileys can deliver JIDs in several forms:
    39→//   5511999888777@s.whatsapp.net            — canonical DM
    40→//   5511999888777:7@s.whatsapp.net          — DM with device suffix (multi-device send)
    41→//   5511999888777:7:3@s.whatsapp.net        — DM with device + agent suffix
    42→//   5511999888777@lid                        — Linked Identity (newer protocol)
    43→// We normalize everything to canonical form: <digits>@s.whatsapp.net
    44→function normalizeJid(jid) {
    45→  if (!jid || typeof jid !== 'string') return null
    46→  const atIdx = jid.indexOf('@')
    47→  if (atIdx < 0) return null
    48→  const server = jid.slice(atIdx + 1)
    49→  if (server !== 's.whatsapp.net') return null
    50→  const userCombined = jid.slice(0, atIdx)
    51→  const user = userCombined.split(':')[0].split('_')[0]
    52→  if (!/^\d{7,}$/.test(user)) return null
    53→  return `${user}@s.whatsapp.net`
    54→}

    70→// Normalize a phone number (strip non-digits) to a WhatsApp JID

    78→function escapeHtml(text) {
    79→  return String(text || '')
    80→    .replace(/&/g, '&amp;')
    81→    .replace(/</g, '&lt;')
    82→    .replace(/>/g, '&gt;')
    83→    .replace(/\n/g, '<br/>')
    84→}

    86→function stripHtml(html) {
    87→  if (!html) return ''
    88→  let text = String(html)
    89→    .replace(/<br\s*\/?>/gi, '\n')
    90→    .replace(/<\/p>/gi, '\n')
    91→    .replace(/<p[^>]*>/gi, '')
    92→    .replace(/<[^>]+>/g, '')
    93→    .replace(/&nbsp;/g, ' ')
    94→    .replace(/&amp;/g, '&')
    95→    .replace(/&lt;/g, '<')
    96→    .replace(/&gt;/g, '>')
    97→    .replace(/&quot;/g, '"')
    98→    .replace(/&#39;/g, "'")
    99→    .trim()
   100→  text = text.replace(/\n{3,}/g, '\n\n')
   101→  return text
   102→}

   104→// v7.28: Parse a chatter body produced by buildChatterBody() back into clean
   105→// message components (direction, timestamp, text). The chatter body format is:
   106→//   <div><strong>📱 WhatsApp {Enviada|Recebida}:</strong> {ts}<br/>{mediaLine}<span>{text}</span></div>
   107→// We want to extract ONLY the user-visible text (the <span> content), detect
   108→// the direction from "Enviada" vs "Recebida", and parse the timestamp.
   109→// Falls back to stripHtml() for any non-conforming body (e.g. messages posted
   110→// by other Odoo users from the web UI — those are returned as-is, plain text).
   111→function parseChatterBody(body) {
   112→  if (!body) return { fromMe: false, timestamp: null, text: '' }
   113→  const s = String(body)
   114→
   115→  // Detect direction from the "WhatsApp Enviada/Recebida:" marker
   116→  const isSent = /WhatsApp\s+Enviada/i.test(s)
   117→  const fromMe = isSent
   118→
   119→  // Extract timestamp from "(...)" — buildChatterBody uses toLocaleString('pt-BR')
   120→  // inside parentheses as part of the label. Actually buildChatterBody does NOT
   121→  // wrap in parens — it inserts the timestamp directly after the colon.
   122→  // But older messages may have the "(date)" format. Support both.
   123→  let timestamp = null
   124→  const parenMatch = s.match(/\(([^)]+)\)/)
   125→  if (parenMatch) {
   126→    const parsed = new Date(parenMatch[1])
   127→    if (!isNaN(parsed.getTime())) timestamp = parsed
   128→  }
   129→
   130→  // Extract the text content from the <span>...</span> block (the real message)
   131→  let text = ''
   132→  const spanMatch = s.match(/<span[^>]*>([\s\S]*?)<\/span>/i)
   133→  if (spanMatch) {
   134→    text = stripHtml(spanMatch[1])
   135→  } else {
   136→    // No <span> — this isn't a WhatsApp chatter message produced by us.
   137→    // It's probably a note posted by a human via the Odoo UI, or a different
   138→    // format entirely. Strip HTML and return the whole body as plain text.
   139→    text = stripHtml(s)
   140→    // If the body still starts with the WhatsApp metadata header, strip it.
   141→    // This handles the case where buildChatterBody emitted no <span> because
   142→    // textContent was empty (media-only message).
   143→    const headerMatch = text.match(/^📱\s*WhatsApp\s+(Enviada|Recebida)[:\s]*(.*)$/is)
   144→    if (headerMatch) {
   145→      // For media-only messages, the "text" will be empty after the header.
   146→      // Strip the header line and keep the remaining content (media label).
   147→      const remaining = text.slice(headerMatch[0].length).trim()
   148→      // Try to detect a media label line like "🎙️ Áudio"
   149→      const mediaMatch = remaining.match(/^([🎙️🖼️📄🎬🏷️][^\n]*)/)
   150→      text = mediaMatch ? mediaMatch[1] : remaining
   151→    }
   152→  }
   153→
   154→  // Try to parse timestamp from the label after the colon (no parens case)
   155→  if (!timestamp) {
   156→    // Look for a date pattern like "14/08/2026, 10:30:45" or "14/08/2026 10:30"
   157→    const dateMatch = s.match(/(\d{1,2}\/\d{1,2}\/\d{4})[,\s]+(\d{1,2}:\d{2}(?::\d{2})?)/)
   158→    if (dateMatch) {
   159→      const [_, d, t] = dateMatch
   160→      // pt-BR format: DD/MM/YYYY HH:MM:SS
   161→      const parsed = new Date(`${d.split('/').reverse().join('-')}T${t}`)
   162→      if (!isNaN(parsed.getTime())) timestamp = parsed
   163→    }
   164→  }
   165→
   166→  return { fromMe, timestamp, text }
   167→}

   174→// v7.19: Build a plain-text transcript of the WhatsApp conversation that will
   175→// be written to the `description` field (Notes) of the CRM lead/opportunity.
   176→function buildConversationTranscript(messages) {
   177→  if (!Array.isArray(messages) || messages.length === 0) return ''
   178→  const sorted = [...messages].sort(
   179→    (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   180→  )
   181→  const lines = sorted.map((m) => {
   182→    const direction = m.fromMe ? 'Eu' : 'Contato'
   183→    const dateLabel = new Date(m.timestamp).toLocaleString('pt-BR')
   184→    const mediaLabel = m.mediaType ? ` [${m.mediaType}]` : ''
   185→    const text = m.textContent && m.textContent.trim() ? m.textContent : '[Mídia]'
   186→    return `[${dateLabel}] ${direction}${mediaLabel}: ${text}`
   187→  })
   188→  return `Conversa WhatsApp:\n${lines.join('\n')}`
   189→}

   331→  async start() {
   332→    if (this.isStarted || this.isStarting) return
   333→    this.isStarting = true
   334→    try {
   335→      fs.mkdirSync(this.authFolder, { recursive: true })
   336→      fs.mkdirSync(this.mediaDir, { recursive: true })
   337→
   338→      this.loadConversationsFromDisk()
   339→
   340→      // Persist conversations every 30 seconds (debounced — only if changed)
   341→      this.persistTimer = setInterval(() => {
   342→        if (this.conversationDirty && Date.now() - this.lastPersistTs > 25_000) {
   343→          this.conversationDirty = false
   344→          this.lastPersistTs = Date.now()
   345→          this.persistConversationsToDisk()
   346→        }
   347→      }, 30_000)
   348→
   349→      // v7.30: Also check Firestore for an existing WhatsApp session.
   350→      // If Firestore has creds for this user but the local filesystem
   351→      // doesn't (e.g., after a deploy that wiped the disk), we still
   352→      // have a saved session. Updates hasSavedSession before logging.
   353→      try {
   354→        const { getFirestoreOrNull } = require('./wa-firestore-auth-state.cjs')
   355→        const db = getFirestoreOrNull()
   356→        if (db && !this.hasSavedSession) {
   357→          const snap = await db.collection('wa_auth_states').doc(String(this.userId)).get()
   358→          if (snap.exists && snap.data() && snap.data().creds) {
   359→            this.hasSavedSession = true
   360→            console.log(`[UserSession:${this.userId}] Found saved session in Firestore (no local creds.json)`)
   361→          }
   362→        }
   363→      } catch (err) {
   364→        console.warn(`[UserSession:${this.userId}] Firestore session check failed:`, err && err.message)
   365→      }
   366→
   367→      console.log(`[UserSession:${this.userId}] Starting — user=${this.user.email}, hasSavedSession=${this.hasSavedSession}, usingGlobalOdooConfig=${this.usingGlobalOdooConfig}`)
   368→
   369→      // Load Odoo config from DB (if needed) before starting auto-auth
   370→      await this.loadOdooConfig()
   371→
   372→      // Start WA + Odoo (fire-and-forget — they self-retry internally)
   373→      this.connectWhatsApp().catch(err => console.error(`[UserSession:${this.userId}] WA startup error:`, err.message))
   374→      this.startOdooAutoAuth().catch(err => console.error(`[UserSession:${this.userId}] Odoo startup error:`, err.message))
   375→
   376→      this.isStarted = true
   377→    } finally {
   378→      this.isStarting = false
   379→    }
   380→  }

   382→  stop() {
   383→    try {
   384→      if (this.persistTimer) { clearInterval(this.persistTimer); this.persistTimer = null }

   387→      if (this.odooReauthTimer) { clearInterval(this.odooReauthTimer); this.odooReauthTimer = null }
   388→      try { this.waSocket?.end(undefined) } catch {}
   389→      this.persistConversationsToDisk()
   390→      console.log(`[UserSession:${this.userId}] Stopped`)
   391→    } catch (err) {
   392→      console.error(`[UserSession:${this.userId}] Stop error:`, err.message)
   393→    }
   394→  }

   396→  // v7.24 (R6): Pre-deploy backup — dump this user's WA creds + conversation
   397→  // state to Odoo chatter so the data survives even if the disk is wiped
   398→  // during a Render deploy. Posted as a mail.message on a designated
   399→  // res.partner (the user's own partner if findable, otherwise partner 1).
   400→  //
   401→  // Format: subject `[BACKUP-CREDS] User <email> — <timestamp>`, body is
   402→  // JSON-stringified { creds, conversations }. If the JSON is too large
   403→  // (>600 KB per chunk — Odoo's chatter body limit is ~1MB), it's split
   404→  // into N chunks posted as separate messages with subject suffix `[1/N]`.
   405→  //
   406→  // Returns: { success, chunksPosted, partnerId, error? }
   407→  async backupToOdoo() {
   408→    if (!this.odooConfig.uid) {
   409→      return { success: false, chunksPosted: 0, error: 'Odoo not authenticated' }
   410→    }
   411→    try {
   412→      // Persist current state to disk first (so the backup reflects the
   413→      // latest conversations, not the last periodic save).
   414→      this.persistConversationsToDisk()
   415→
   416→      // Read creds.json
   417→      const credsPath = path.join(this.authFolder, 'creds.json')
   418→      let credsJson = null
   419→      try {
   420→        if (fs.existsSync(credsPath)) {
   421→          credsJson = JSON.parse(fs.readFileSync(credsPath, 'utf8'))
   422→        }
   423→      } catch (err) {
   424→        console.warn(`[Backup:${this.userId}] Could not read creds.json:`, err.message)
   425→      }
   426→
   427→      // Read conv_state_<userId>.json
   428→      let convState = null
   429→      try {
   430→        if (fs.existsSync(this.stateFile)) {
   431→          convState = JSON.parse(fs.readFileSync(this.stateFile, 'utf8'))
   432→        }
   433→      } catch (err) {
   434→        console.warn(`[Backup:${this.userId}] Could not read conv state:`, err.message)
   435→      }
   436→
   437→      // Build the JSON payload
   438→      const payload = JSON.stringify({
   439→        userId: this.userId,
   440→        email: this.user.email,
   441→        backedUpAt: new Date().toISOString(),
   442→        version: '7.24',
   443→        creds: credsJson,
   444→        conversations: convState,
   445→      })
   446→
   447→      // Find a partner to post to. Strategy: look up res.partner by the
   448→      // user's email; if not found, fall back to partner id 1
   449→      // (Administrator in most Odoo installs).
   450→      let backupPartnerId = 1
   451→      try {
   452→        const partners = await this.odooExecuteKw('res.partner', 'search_read', [
   453→          [['email', '=', this.user.email]],
   454→        ], { fields: ['id', 'name'], limit: 1 })
   455→        if (partners && partners.length > 0) {
   456→          backupPartnerId = partners[0].id
   457→        } else {
   458→          // Fall back: try admin user's partner (res.users where login = 'admin')
   459→          const adminUsers = await this.odooExecuteKw('res.users', 'search_read', [
   460→            [['login', '=', 'admin']],
   461→          ], { fields: ['partner_id'], limit: 1 })
   462→          if (adminUsers && adminUsers.length > 0 && adminUsers[0].partner_id) {
   463→            backupPartnerId = adminUsers[0].partner_id[0]
   464→          }
   465→        }
   466→      } catch (err) {
   467→        console.warn(`[Backup:${this.userId}] Partner lookup failed, using id=1:`, err.message)
   468→      }
   469→
   470→      // Split into chunks of ~600 KB (base64 expansion + Odoo overhead
   471→      // means 600 KB raw → ~800 KB on the wire, safely under 1 MB).
   472→      const CHUNK_SIZE = 600_000
   473→      const chunks = []
   474→      for (let i = 0; i < payload.length; i += CHUNK_SIZE) {
   475→        chunks.push(payload.slice(i, i + CHUNK_SIZE))
   476→      }
   477→      if (chunks.length === 0) chunks.push('')
   478→
   479→      const tsLabel = new Date().toISOString()
   480→      const subject = `[BACKUP-CREDS] User ${this.user.email} — ${tsLabel}`
   481→      const total = chunks.length
   482→
   483→      for (let i = 0; i < chunks.length; i++) {
   484→        const partNum = i + 1
   485→        const body = `<div><strong>${subject}</strong> [${partNum}/${total}]<br/>` +
   486→          `<em>Backup dos dados de conexão e conversas — não editar.</em>` +
   487→          `<pre style="font-size:10px; white-space:pre-wrap; word-break:break-all;">${escapeHtml(chunks[i])}</pre></div>`
   488→        try {
   489→          await this.odooPostMessage('res.partner', backupPartnerId, body)
   490→        } catch (err) {
