   113→// v7.17: Persist in-memory conversations to disk so they survive deploys.
   114→// Baileys auth files already persist on disk, but the in-memory conversations
   115→// Map (chats + messages) is lost on every restart. We save a snapshot to

   117→// it back so the user doesn't see an empty conversation list while the
   118→// resyncAppState completes.

   125→        jid: c.jid,

   130→        lastMessage: c.lastMessage,
   131→        lastMessageAt: c.lastMessageAt ? c.lastMessageAt.toISOString() : null,
   132→        unreadCount: c.unreadCount,
   133→        messages: c.messages.map(m => ({
   134→          ...m,
   135→          timestamp: m.timestamp instanceof Date ? m.timestamp.toISOString() : m.timestamp,
   136→        })),
   137→      })),

   140→      lidToPhoneMap: Array.from(lidToPhoneMap.entries()),
   141→    }

   146→  }

   149→function loadConversationsFromDisk() {
   150→  try {
   151→    if (!fs.existsSync(CONV_STATE_FILE)) {
   152→      console.log(`[Server] No saved conversation state — starting fresh`)
   153→      return
   154→    }
   155→    const raw = fs.readFileSync(CONV_STATE_FILE, 'utf8')
   156→    const snapshot = JSON.parse(raw)
   157→    if (!snapshot || snapshot.version !== 1) {
   158→      console.log(`[Server] Saved conversation state is old format — ignoring`)
   159→      return
   160→    }
   161→
   162→    let loaded = 0
   163→    for (const c of snapshot.conversations || []) {
   164→      if (!c.jid || !isValidPhoneJid(c.jid)) continue
   165→      conversations.set(c.jid, {
   166→        jid: c.jid,
   167→        name: c.name || null,
   168→        phone: c.phone || extractPhone(c.jid),
   169→        pushName: c.pushName || null,
   170→        avatarUrl: c.avatarUrl || null,
   171→        lastMessage: c.lastMessage || null,
   172→        lastMessageAt: c.lastMessageAt ? new Date(c.lastMessageAt) : null,
   173→        unreadCount: c.unreadCount || 0,
   174→        messages: (c.messages || []).map(m => ({
   175→          ...m,
   176→          timestamp: m.timestamp ? new Date(m.timestamp) : new Date(),
   177→        })),
   178→      })
   179→      loaded++
   180→    }
   181→
   182→    for (const [jid, name] of snapshot.contactNames || []) {
   183→      contactNames.set(jid, name)
   184→    }
   185→
   186→    for (const c of snapshot.deviceContacts || []) {
   187→      if (c.jid) deviceContacts.set(c.jid, c)
   188→    }
   189→
   190→    for (const [lid, phone] of snapshot.lidToPhoneMap || []) {
   191→      lidToPhoneMap.set(lid, phone)
   192→    }
   193→
   194→    console.log(`[Server] ✓ Loaded ${loaded} conversations, ${contactNames.size} contact names, ${deviceContacts.size} device contacts from disk (saved ${snapshot.savedAt})`)
   195→  } catch (err) {
   196→    console.error(`[Server] Failed to load conversations from disk: ${err.message}`)
   197→  }
   198→}

   292→function odooAuthenticate() {
   293→  return new Promise((resolve, reject) => {
   294→    const client = makeXmlRpcClient('/xmlrpc/2/common')
   295→    client.methodCall('authenticate', [odooConfig.db, odooConfig.username, odooConfig.password, {}], (error, value) => {
   296→      if (error) reject(error)
   297→      else if (!value) reject(new Error('Authentication failed - invalid credentials'))
   298→      else resolve(value)
   299→    })
   300→  })
   301→}

   303→function odooExecuteKw(model, method, args, kwargs = {}) {
   304→  return new Promise((resolve, reject) => {
   305→    if (!odooConfig.uid) {
   306→      reject(new Error('Not authenticated with Odoo'))
   307→      return
   308→    }
   309→    const client = makeXmlRpcClient('/xmlrpc/2/object')
   310→    client.methodCall('execute_kw', [odooConfig.db, odooConfig.uid, odooConfig.password, model, method, args, kwargs], (error, value) => {
   311→      if (error) reject(error)
   312→      else resolve(value)
   313→    })
   314→  })
   315→}

   319→  try {
   320→    const fields = await odooExecuteKw(model, 'fields_get', [], { attributes: ['string', 'type'] })
   321→    const fieldNames = new Set(Object.keys(fields))

   324→    return fieldNames

   327→    return new Set()
   328→  }

   331→async function filterExistingFields(model, requestedFields) {
   332→  const available = await getAvailableFields(model)
   333→  const existing = requestedFields.filter(f => available.has(f))
   334→  if (!existing.includes('id') && available.has('id')) existing.unshift('id')
   335→  if (!existing.includes('name') && available.has('name')) existing.push('name')
   336→  return existing
   337→}

   339→async function buildSafeValues(model, values) {
   340→  const available = await getAvailableFields(model)
   341→  const safe = {}
   342→  for (const [key, value] of Object.entries(values)) {
   343→    if (available.has(key)) safe[key] = value
   344→  }
   345→  return safe
   346→}

   348→async function odooSearch(model, domain, fields = [], limit = 80, offset = 0) {
   349→  const safeFields = fields.length > 0 ? await filterExistingFields(model, fields) : []
   350→  return odooExecuteKw(model, 'search_read', [domain], {

   353→    offset,

   357→// v7.20: Build a phone-number search domain that ONLY uses fields the model
   358→// actually has. This prevents "Invalid field X.Y" errors on Odoo versions
   359→// where some phone-like fields don't exist or aren't searchable (e.g. on
   360→// Odoo 19.4, res.partner.mobile is no longer directly searchable).
   361→// Returns a domain like:
   362→//   [['phone', 'ilike', x]]
   363→//   ['|', ['phone', 'ilike', x], ['whatsapp', 'ilike', x]]
   364→//   ['|', ['phone', 'ilike', x], ['|', ['mobile', 'ilike', x], ['whatsapp', 'ilike', x]]]
   365→// If no candidate fields exist, falls back to searching by name.
   366→async function buildPhoneSearchDomain(model, phone, candidateFields = ['phone', 'mobile', 'whatsapp', 'whatsapp_number']) {
   367→  if (!phone) return []
   368→  const available = await getAvailableFields(model)
   369→  const usable = candidateFields.filter(f => available.has(f))
   370→  if (usable.length === 0) {
   371→    return [['name', 'ilike', phone]]
   372→  }
   373→  if (usable.length === 1) {
   374→    return [[usable[0], 'ilike', phone]]
   375→  }
   376→  const domain = []
   377→  for (let i = 0; i < usable.length - 1; i++) domain.push('|')
   378→  for (const f of usable) domain.push([f, 'ilike', phone])
   379→  return domain
   380→}

   447→    .replace(/&/g, '&amp;')

   449→    .replace(/>/g, '&gt;')

   453→// v7.19: Build a plain-text transcript of the WhatsApp conversation that will
   454→// be written to the `description` field (Notes) of the CRM lead/opportunity.
   455→// Format is human-readable, chronological, with direction + date + text per line.
   456→// Returns an empty string if there are no messages.
   457→function buildConversationTranscript(messages) {
   458→  if (!Array.isArray(messages) || messages.length === 0) return ''
   459→
   460→  const sorted = [...messages].sort(
   461→    (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   462→  )
   463→
   464→  const lines = sorted.map((m) => {
   465→    const direction = m.fromMe ? 'Eu' : 'Contato'
   466→    const dateLabel = new Date(m.timestamp).toLocaleString('pt-BR')
   467→    const mediaLabel = m.mediaType ? ` [${m.mediaType}]` : ''
   468→    const text = m.textContent && m.textContent.trim()
   469→      ? m.textContent
   470→      : '[Mídia]'
   471→    return `[${dateLabel}] ${direction}${mediaLabel}: ${text}`
   472→  })
   473→
   474→  return `Conversa WhatsApp:\n${lines.join('\n')}`
   475→}

   477→async function autoSyncWhatsAppMessage(data) {
   478→  const result = {
   479→    partnerId: null, leadId: null, mailMessageId: null, activityId: null,
   480→    created: { partner: false, lead: false }, errors: [],
   481→    chatterPosted: false, chatterLeadPosted: false,
   482→  }
   483→
   484→  // v7.17: Detailed skip logging — when lead creation silently fails, we need
   485→  // to know WHY. This was the root cause of "Parou de Criar Leads".
   486→  if (!autoSyncSettings.enabled) {
   487→    console.log('[AutoSync] SKIP — autoSyncSettings.enabled is false')
   488→    return result
   489→  }
   490→  if (!odooConfig.uid) {
   491→    console.log('[AutoSync] SKIP — Odoo not authenticated (uid is null). Lead creation and chatter sync will NOT happen. Check ODOO_URL/DB/USERNAME/PASSWORD env vars.')
   492→    return result
   493→  }
   494→
   495→  // v7.14: NEVER duplicate chatter posts — check dedup set FIRST.
   496→  // We dedup by whatsappId (or fallback id) so the same message is never posted twice.
   497→  const dedupKey = data.dedupId
   498→    ? `${data.jid}|${data.dedupId}`
   499→    : `${data.jid}|${data.fromMe ? 'out' : 'in'}|${data.timestamp}|${(data.textContent || '').slice(0, 50)}`
   500→  if (postedChatterIds.has(dedupKey)) {
   501→    console.log(`[AutoSync] Skipping — already posted to chatter: ${dedupKey}`)
   502→    return result
   503→  }
   504→
   505→  console.log(`[AutoSync] Processing message from ${data.phone} (fromMe=${data.fromMe}, autoCreateLead=${autoSyncSettings.autoCreateLead}, autoCreateContact=${autoSyncSettings.autoCreateContact})`)
   506→
   507→  try {
   508→    // ===== Step 1: Always ensure partner exists (for both sent AND received) =====
   509→    if (autoSyncSettings.autoCreateContact) {
   510→      const contactName = data.pushName || `WhatsApp ${data.phone}`
   511→      // v7.20: Use buildPhoneSearchDomain to avoid the "Invalid field res.partner.mobile"
   512→      // error on Odoo 19.4 (mobile is not directly searchable there).
   513→      const partnerFields = await getAvailableFields('res.partner')
   514→      const domain = await buildPhoneSearchDomain('res.partner', data.phone)
   515→
   516→      // Build contact values — only include fields that actually exist on the model
   517→      const contactValues = { name: contactName, phone: data.phone }
   518→      if (partnerFields.has('mobile')) contactValues.mobile = data.phone
   519→      if (partnerFields.has('whatsapp')) contactValues.whatsapp = data.phone
   520→      if (partnerFields.has('whatsapp_number')) contactValues.whatsapp_number = data.phone
   521→

   523→      result.partnerId = partnerResult.id
   524→      result.created.partner = partnerResult.created
   525→
   526→      const cached = phoneToPartnerCache.get(data.phone)
   527→      if (cached) cached.partnerId = partnerResult.id
   528→      else phoneToPartnerCache.set(data.phone, { partnerId: partnerResult.id, leadId: null, leadCreated: false })
   529→    }
   530→
   531→    // ===== Step 2: For INCOMING messages, ensure a lead exists =====
   532→    // v7.14: Only create leads for incoming messages (new prospect reaching out).
   533→    if (autoSyncSettings.autoCreateLead && !data.fromMe && result.partnerId) {
   534→      const cached = phoneToPartnerCache.get(data.phone)
   535→      if (cached && cached.leadId && cached.leadCreated) {
   536→        result.leadId = cached.leadId
   537→        console.log(`[AutoSync] Lead reused from cache: crm.lead#${result.leadId} (partner=${result.partnerId})`)
   538→      } else {
   539→        // Search for an EXISTING open lead for this partner (any name, not just our prefix)
   540→        // — this prevents creating duplicate leads if one already exists.
   541→        const existingLeads = await odooSearch('crm.lead', [
   542→          ['partner_id', '=', result.partnerId],
   543→          ['type', '=', 'lead'],
   544→          ['active', '=', true],
   545→        ], ['id', 'name', 'stage_id'], 1)
   546→
   547→        if (existingLeads && existingLeads.length > 0) {
   548→          result.leadId = existingLeads[0].id
   549→          if (cached) { cached.leadId = existingLeads[0].id; cached.leadCreated = true }
   550→          else phoneToPartnerCache.set(data.phone, { partnerId: result.partnerId, leadId: existingLeads[0].id, leadCreated: true })
   551→          console.log(`[AutoSync] Lead reused from Odoo search: crm.lead#${result.leadId} (name="${existingLeads[0].name}")`)
   552→        } else {
   553→          // No existing lead — create one with the WhatsApp prefix
   554→          const leadName = `${autoSyncSettings.leadPrefix}${data.pushName || data.phone}`
   555→          const leadValues = {
   556→            name: leadName, type: 'lead', partner_id: result.partnerId, phone: data.phone,
   557→            description: `Conversa iniciada via WhatsApp em ${new Date().toLocaleString('pt-BR')}`,
   558→          }
   559→          const leadFields = await getAvailableFields('crm.lead')
   560→          if (leadFields.has('whatsapp_number')) leadValues.whatsapp_number = data.phone
   561→          if (autoSyncSettings.leadTeamId) leadValues.team_id = autoSyncSettings.leadTeamId
   562→          if (autoSyncSettings.leadUserId) leadValues.user_id = autoSyncSettings.leadUserId
   563→
   564→          try {
   565→            result.leadId = await odooCreate('crm.lead', leadValues)
   566→            result.created.lead = true
   567→            if (cached) { cached.leadId = result.leadId; cached.leadCreated = true }
   568→            else phoneToPartnerCache.set(data.phone, { partnerId: result.partnerId, leadId: result.leadId, leadCreated: true })
   569→            console.log(`[AutoSync] ✓ NEW Lead created: crm.lead#${result.leadId} (name="${leadName}", partner=${result.partnerId})`)
   570→          } catch (error) {
   571→            console.error(`[AutoSync] ✗ Lead creation FAILED for partner ${result.partnerId}: ${error.message}`)
   572→            result.errors.push(`Lead creation failed: ${error.message}`)
   573→          }
   574→        }
   575→      }
   576→    } else if (autoSyncSettings.autoCreateLead && data.fromMe) {
   577→      // v7.17: Log why we're skipping lead creation for outgoing messages
   578→      console.log(`[AutoSync] Lead creation skipped — outgoing message (fromMe=true). Leads are only created for incoming messages.`)
   579→    }
   580→
   581→    // ===== Step 2.5: For OUTGOING messages, find an active lead by phone (if any) =====
   582→    // v7.14: When user sends a message, if there's an open lead for this contact,
   583→    // we also want to post to that lead's chatter.
   584→    if (data.fromMe && result.partnerId) {
   585→      // Check cache first
   586→      let cachedLeadId = phoneToActiveLeadCache.get(data.phone)
   587→      if (!cachedLeadId) {
   588→        const cached = phoneToPartnerCache.get(data.phone)
   589→        if (cached && cached.leadId && cached.leadCreated) {
   590→          cachedLeadId = cached.leadId
   591→        } else {
   592→          // Search for any active lead for this partner
   593→          const existingLeads = await odooSearch('crm.lead', [
   594→            ['partner_id', '=', result.partnerId],
   595→            ['type', '=', 'lead'],
   596→            ['active', '=', true],
   597→          ], ['id'], 1)
   598→          if (existingLeads && existingLeads.length > 0) {
   599→            cachedLeadId = existingLeads[0].id
   600→            if (cached) { cached.leadId = cachedLeadId; cached.leadCreated = true }
   601→          }
   602→        }
   603→        if (cachedLeadId) {
   604→          phoneToActiveLeadCache.set(data.phone, cachedLeadId)
   605→          result.leadId = cachedLeadId
   606→        }
   607→      } else {
   608→        result.leadId = cachedLeadId
   609→      }
   610→    }
   611→
   612→    // ===== Step 3: ALWAYS post to chatter (incoming AND outgoing) =====
   613→    // v7.14: User requested "Atualizar o chatter no contato à cada mensagem enviada/recebida"
   614→    // We post to: (a) partner chatter ALWAYS, (b) lead chatter if there's an open lead.
   615→    if (autoSyncSettings.autoPostMessages && result.partnerId) {
   616→      const direction = data.fromMe ? 'Enviada' : 'Recebida'
   617→      const mediaLabel = data.mediaType ? ` [${data.mediaType}]` : ''
   618→      const tsLabel = `(${data.timestamp})`
   619→      const msgBody = data.textContent
   620→        ? `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel} ${tsLabel}</p><p>${escapeHtml(data.textContent)}</p>`
   621→        : `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel} ${tsLabel} [Mídia]</p>`
   622→
   623→      // (a) Always post to partner chatter
   624→      try {
   625→        result.mailMessageId = await odooPostMessage('res.partner', result.partnerId, msgBody)
   626→        result.chatterPosted = true
   627→        console.log(`[AutoSync] ✓ Posted to partner ${result.partnerId} chatter: ${direction}`)
   628→      } catch (error) {
   629→        result.errors.push(`Failed to post to partner chatter: ${error.message}`)
   630→      }
   631→
   632→      // (b) If there's an open lead, ALSO post to lead chatter (but only once per message)
   633→      if (result.leadId) {
   634→        try {
   635→          await odooPostMessage('crm.lead', result.leadId, msgBody)
   636→          result.chatterLeadPosted = true
   637→          console.log(`[AutoSync] ✓ Posted to lead ${result.leadId} chatter: ${direction}`)
   638→        } catch (error) {
   639→          result.errors.push(`Failed to post to lead chatter: ${error.message}`)
   640→        }
   641→      }
   642→    }
   643→
   644→    // ===== Step 4: Create activity ONLY for new leads =====
   645→    if (autoSyncSettings.autoCreateActivity && result.created.lead && result.leadId) {
   646→      try {
   647→        result.activityId = await odooCreateActivity('crm.lead', result.leadId, 'Nova mensagem WhatsApp', `Contato ${data.pushName || data.phone} iniciou uma conversa via WhatsApp.\n\nMensagem: ${data.textContent || '[Mídia]'}`)
   648→      } catch (error) {
   649→        result.errors.push(`Failed to create activity: ${error.message}`)
   650→      }
   651→    }
   652→
   653→    // ===== Step 5: Mark this message as posted (for dedup) =====
   654→    // Only mark if at least one chatter post succeeded
   655→    if (result.chatterPosted || result.chatterLeadPosted) {
   656→      postedChatterIds.add(dedupKey)
   657→      // Cap the set size to prevent memory leak (keep last 5000 entries)
   658→      if (postedChatterIds.size > 5000) {
   659→        const firstKey = postedChatterIds.values().next().value
   660→        postedChatterIds.delete(firstKey)
   661→      }
   662→    }
   663→  } catch (error) {
   664→    result.errors.push(`Auto-sync error: ${error.message}`)
   665→    console.error(`[AutoSync] Error:`, error.message)
   666→  }
   667→
   668→  return result
   669→}

   671→async function autoAuthenticateFromEnv() {
   672→  const envUrl = process.env.ODOO_URL
   673→  const envDb = process.env.ODOO_DB
   674→  const envUsername = process.env.ODOO_USERNAME
   675→  const envPassword = process.env.ODOO_PASSWORD
   676→
   677→  if (envUrl && envDb && envUsername && envPassword) {
   678→    console.log(`[Odoo] Auto-authenticating with env vars: ${envUrl} / ${envDb} / ${envUsername}`)
   679→    // v7.17: Retry auth up to 3 times with 5s delay between attempts.
   680→    // This handles transient network errors and Odoo being briefly unavailable
   681→    // during a cold start. If all retries fail, lead creation / chatter sync
   682→    // will be skipped silently — which is the "Parou de Criar Leads" bug.
   683→    for (let attempt = 1; attempt <= 3; attempt++) {
   684→      try {
   685→        odooConfig = { url: envUrl, db: envDb, username: envUsername, password: envPassword, uid: null }
   686→        if (attempt === 1) {
   687→          modelFieldsCache.clear()
   688→          phoneToPartnerCache.clear()
   689→        }
   690→        const uid = await odooAuthenticate()
   691→        odooConfig.uid = uid
   692→        console.log(`[Odoo] Auto-authenticated as ${envUsername} (uid: ${uid}) on attempt ${attempt}`)
   693→        await getAvailableFields('res.partner')
   694→        await getAvailableFields('crm.lead')
   695→        io.of('/odoo').emit('odoo:status', {
   696→          connected: true, url: odooConfig.url, db: odooConfig.db, username: odooConfig.username,
   697→        })
   698→        // v7.21: If WhatsApp is already connected, pull conversation history
   699→        // from Odoo chatter now. This handles the case where WA connected
   700→        // before Odoo auth completed (e.g. Odoo was slow on startup).
   701→        if (waSocket && connectionState.connection === 'open') {
   702→          console.log('[Odoo] WA is connected — scheduling Odoo history sync in 3s')
   703→          setTimeout(() => {
   704→            syncAllConversationsFromOdoo(io, { silent: false }).catch(err =>
   705→              console.error('[Odoo] Post-auth history sync failed:', err.message)
   706→            )
   707→          }, 3000)
   708→        }
   709→        return // success
   710→      } catch (error) {
   711→        console.error(`[Odoo] Auto-auth attempt ${attempt}/3 failed: ${error.message}`)
   712→        if (attempt < 3) {
   713→          console.log(`[Odoo] Retrying in 5s...`)
   714→          await new Promise(r => setTimeout(r, 5000))
   715→        }
   716→      }
   717→    }
   718→    console.error(`[Odoo] All auto-auth attempts failed. Lead creation and chatter sync will NOT work until manual auth via UI.`)
   719→    // v7.17: Schedule a background retry every 60s until auth succeeds.
   720→    // This handles the case where Odoo was down during startup but comes back later.
   721→    if (!odooReauthTimer) {
   722→      odooReauthTimer = setInterval(async () => {
   723→        if (odooConfig.uid) {
   724→          clearInterval(odooReauthTimer)
   725→          odooReauthTimer = null
   726→          return
   727→        }
   728→        console.log(`[Odoo] Background re-auth attempt...`)
   729→        try {
   730→          const uid = await odooAuthenticate()
   731→          odooConfig.uid = uid
   732→          console.log(`[Odoo] Background re-auth SUCCESS (uid: ${uid})`)
   733→          await getAvailableFields('res.partner')
   734→          await getAvailableFields('crm.lead')
   735→          io.of('/odoo').emit('odoo:status', {
   736→            connected: true, url: odooConfig.url, db: odooConfig.db, username: odooConfig.username,
   737→          })
   738→          clearInterval(odooReauthTimer)
   739→          odooReauthTimer = null
   740→          // v7.21: Pull Odoo chatter history now that we finally have auth
   741→          if (waSocket && connectionState.connection === 'open') {
   742→            console.log('[Odoo] WA is connected — scheduling Odoo history sync in 3s (post re-auth)')
   743→            setTimeout(() => {
   744→              syncAllConversationsFromOdoo(io, { silent: false }).catch(err =>
   745→                console.error('[Odoo] Post re-auth history sync failed:', err.message)
   746→              )
   747→            }, 3000)
   748→          }
   749→        } catch (error) {
   750→          console.log(`[Odoo] Background re-auth still failing: ${error.message}`)
   751→        }
   752→      }, 60_000)
   753→    }
   754→  }
   755→}

   791→// ========== JID Normalization ==========
   792→// Baileys can deliver JIDs in several forms:
   793→//   5511999888777@s.whatsapp.net            — canonical DM
   794→//   5511999888777:7@s.whatsapp.net          — DM with device suffix (multi-device send)
   795→//   5511999888777:7:3@s.whatsapp.net        — DM with device + agent suffix
   796→//   5511999888777@lid                        — Linked Identity (newer protocol)
   797→// We normalize everything to canonical form: <digits>@s.whatsapp.net
   798→function normalizeJid(jid) {
   799→  if (!jid || typeof jid !== 'string') return null
   800→  const atIdx = jid.indexOf('@')
   801→  if (atIdx < 0) return null
   802→  const server = jid.slice(atIdx + 1)
   803→  if (server !== 's.whatsapp.net') return null
   804→  const userCombined = jid.slice(0, atIdx)
   805→  // Strip :device and _agent suffixes — keep only the phone digits
   806→  const user = userCombined.split(':')[0].split('_')[0]
   807→  if (!/^\d{7,}$/.test(user)) return null
   808→  return `${user}@s.whatsapp.net`
   809→}

   825→function getOrCreateConversation(jid, pushName) {
   826→  // Always normalize first — Baileys may send 5511999888777:7@s.whatsapp.net
   827→  const normalizedJid = normalizeJid(jid)
   828→  if (!normalizedJid) return null

   830→  const phone = extractPhone(normalizedJid)

   832→  if (!conversations.has(normalizedJid)) {

   834→      jid: normalizedJid,
   835→      name: cachedName || pushName || null,
   836→      phone,
   837→      pushName: pushName || null,
   838→      avatarUrl: null,
   839→      lastMessage: null,
   840→      lastMessageAt: new Date(),
   841→      unreadCount: 0,
   842→      messages: [],
   843→    })
   844→  }
   845→  const conv = conversations.get(normalizedJid)
   846→  if (cachedName) {
   847→    conv.name = cachedName
   848→  } else if (pushName && !conv.name) {
   849→    conv.pushName = pushName
   850→  }
   851→
   852→  // Also track in deviceContacts phonebook (so conversations started from
   853→  // incoming messages appear in the Contacts tab too)
   854→  const displayName = cachedName || pushName || phone
   855→  if (phone) {
   856→    if (!deviceContacts.has(normalizedJid)) {
   857→      deviceContacts.set(normalizedJid, { jid: normalizedJid, phone, name: displayName, avatarUrl: null })
   858→    } else if (cachedName) {
   859→      deviceContacts.get(normalizedJid).name = cachedName
   860→    }
   861→  }
   862→
   863→  return conv
   864→}

   881→function getSortedConversations() {
   882→  // v7.20: Restored v7.16 behavior — Conversas tab shows BOTH actual
   883→  // conversations AND device contacts that don't have a conversation yet.
   884→  // The user can click any device contact to start chatting immediately
   885→  // (whatsapp:send-message creates the conversation on-the-fly via
   886→  // getOrCreateConversation).
   887→  //
   888→  // v7.17 had separated these into two tabs, but that broke the UX because
   889→  // clicking a contact in the Contatos tab would fail to open a chat view
   890→  // (selectedConversation was null since the contact wasn't in conversations).
   891→  // The v7.18 fix tried to work around this with an explicit start-conversation
   892→  // flow, but that flow had its own bugs (errors when Odoo mobile field invalid,
   893→  // race conditions, etc). Reverting to the unified list is the simplest fix.
   894→
   895→  // Build a set of JIDs that are already in conversations
   896→  const conversationJids = new Set(conversations.keys())
   897→
   898→  // Convert device contacts (that don't have a conversation yet) into
   899→  // conversation-like objects so they show up in the Conversas list.
   900→  const deviceContactEntries = []
   901→  for (const contact of deviceContacts.values()) {
   902→    if (conversationJids.has(contact.jid)) continue
   903→    deviceContactEntries.push({
   904→      jid: contact.jid,
   905→      name: contact.name,
   906→      phone: contact.phone,
   907→      pushName: null,
   908→      avatarUrl: contact.avatarUrl || null,
   909→      lastMessage: null,
   910→      lastMessageAt: null,
   911→      unreadCount: 0,
   912→      messageCount: 0,
   913→      // Custom flag — frontend can use this to show "Sem mensagens" placeholder
   914→      _isDeviceContact: true,
   915→    })
   916→  }
   917→
   918→  return Array.from(conversations.values())
   919→    .filter(conv => isValidPhoneJid(conv.jid))
   920→    .map(serializeConversation)
   921→    .filter(Boolean)
   922→    .concat(deviceContactEntries)
   923→    .sort((a, b) => {
   924→      // Conversations with messages always come first
   925→      const tA = a.lastMessageAt ? new Date(a.lastMessageAt).getTime() : 0
   926→      const tB = b.lastMessageAt ? new Date(b.lastMessageAt).getTime() : 0
   927→      if (tA === 0 && tB === 0) {
   928→        // Both are device contacts without messages — sort by name
   929→        return (a.name || a.phone || '').localeCompare(b.name || b.phone || '')
   930→      }
   931→      return tB - tA
   932→    })
   933→}

   972→// Strip HTML tags from a string (used to convert Odoo chatter HTML to plain text)
   973→function stripHtml(html) {
   974→  if (!html) return ''
   975→  let text = String(html)
   976→    .replace(/<br\s*\/?>/gi, '\n')
   977→    .replace(/<\/p>/gi, '\n')
   978→    .replace(/<p[^>]*>/gi, '')
   979→    .replace(/<[^>]+>/g, '')
   980→    .replace(/&nbsp;/g, ' ')
   981→    .replace(/&amp;/g, '&')
   982→    .replace(/&lt;/g, '<')
   983→    .replace(/&gt;/g, '>')
   984→    .replace(/&quot;/g, '"')
   985→    .replace(/&#39;/g, "'")
   986→    .trim()
   987→  text = text.replace(/\n{3,}/g, '\n\n')
   988→  return text
   989→}

   991→// Fetch chatter messages from Odoo (mail.message) and merge them into a conversation.
   992→// Returns the number of messages added. Used by "start conversation from Odoo"
   993→// and by the "Pull from Odoo" button.
   994→async function pullOdooChatterIntoConversation(jid, model, recordId, limit = 200) {
   995→  if (!odooConfig.uid) return 0
   996→  if (!isValidPhoneJid(jid)) return 0
   997→  const conv = conversations.get(jid)
   998→  if (!conv) return 0
   999→
  1000→  try {
  1001→    const domain = [
  1002→      ['model', '=', model],
  1003→      ['res_id', '=', recordId],
  1004→    ]
  1005→    const messages = await odooExecuteKw('mail.message', 'search_read', [domain], {
  1006→      fields: ['id', 'body', 'author_id', 'email_from', 'date', 'message_type', 'subtype_id', 'create_date'],
  1007→      order: 'date asc',
  1008→      limit,
  1009→    })
  1010→
  1011→    let added = 0
  1012→    for (const msg of (messages || [])) {
  1013→      const body = msg.body || ''
  1014→      const plainText = stripHtml(body)
  1015→      // Detect direction from "Enviada" / "Recebida" markers we wrote earlier
  1016→      const isSent = /WhatsApp\s+Enviada/i.test(body)
  1017→      const fromMe = isSent
  1018→
  1019→      // Extract timestamp — prefer date in body parentheses, fall back to msg.date
  1020→      let timestamp
  1021→      const dateMatch = body.match(/\(([^)]+)\)/)
  1022→      if (dateMatch) {
  1023→        const parsed = new Date(dateMatch[1])
  1024→        if (!isNaN(parsed.getTime())) timestamp = parsed
  1025→      }
  1026→      if (!timestamp) timestamp = new Date(msg.date || msg.create_date || Date.now())
  1027→
  1028→      // Dedup by externalId
  1029→      const externalId = `odoo-${msg.id}`
  1030→      if (conv.messages.some(m => m.whatsappId === externalId)) continue
  1031→      // Dedup by content + timestamp window
  1032→      const isDup = conv.messages.some(m =>
  1033→        m.fromMe === fromMe &&
  1034→        m.textContent === plainText &&
  1035→        Math.abs(m.timestamp.getTime() - timestamp.getTime()) < 5000
  1036→      )
  1037→      if (isDup) continue
  1038→
  1039→      conv.messages.push({
  1040→        id: externalId,
  1041→        whatsappId: externalId,
  1042→        fromMe,
  1043→        textContent: plainText || null,
  1044→        mediaType: null,
  1045→        timestamp,
  1046→        status: 'delivered',
  1047→      })
  1048→      added++
  1049→    }
  1050→
  1051→    if (added > 0) {
  1052→      // Sort messages by timestamp
  1053→      conv.messages.sort((a, b) => a.timestamp.getTime() - b.timestamp.getTime())
  1054→      // Update lastMessage if needed
  1055→      const last = conv.messages[conv.messages.length - 1]
  1056→      if (!conv.lastMessageAt || last.timestamp > conv.lastMessageAt) {
  1057→        conv.lastMessageAt = last.timestamp
  1058→        if (last.textContent) conv.lastMessage = last.textContent
  1059→      }
  1060→    }
  1061→    return added
  1062→  } catch (err) {
  1063→    console.error(`[Odoo] Chatter pull failed for ${model}#${recordId}:`, err.message)
  1064→    return 0
  1065→  }
  1066→}

  1069→// v7.21: syncAllConversationsFromOdoo()
  1070→// ---------------------------------------------------------------
  1071→// Called automatically when:
  1072→//   (a) WhatsApp reconnects (8s delay after 'open')
  1073→//   (b) Odoo auto-auth succeeds (if WA is already connected)
  1074→//   (c) Manually via the 'odoo:sync-all-history' socket endpoint
  1075→//
  1076→// The Odoo chatter is the "source of truth" for conversation history.
  1077→// On every reconnect/redeploy, we scan all mail.message records that
  1078→// were posted by us (body contains "WhatsApp" + our direction marker),
  1079→// group them by res.partner, and pull each partner's chatter into the
  1080→// corresponding local conversation (creating the conversation if it
  1081→// doesn't exist yet).
  1082→//
  1083→// This makes the conversation history survive:
  1084→//   - Render redeployments (process restart)
  1085→//   - WA disconnect/reconnect (network issue, phone offline)
  1086→//   - Disk wipe (conversation-state.json missing or stale)
  1087→// The Odoo chatter remains the durable store.
  1088→//
  1089→// Returns: { partnersProcessed, messagesAdded, partnersFailed }
  1090→// ====================================================================
  1091→let odooHistorySyncInProgress = false
  1092→let odooHistorySyncLastRun = null

  1094→async function syncAllConversationsFromOdoo(io, opts = {}) {
  1095→  const { silent = false, limit = 1000 } = opts
  1096→
  1097→  if (!odooConfig.uid) {
  1098→    if (!silent) console.log('[OdooSync] SKIP — Odoo not authenticated yet')
  1099→    return { skipped: true, reason: 'not_authenticated' }
  1100→  }
  1101→  if (odooHistorySyncInProgress) {
  1102→    if (!silent) console.log('[OdooSync] SKIP — another sync is already running')
  1103→    return { skipped: true, reason: 'already_running' }
  1104→  }
  1105→
  1106→  odooHistorySyncInProgress = true
  1107→  const startedAt = Date.now()
  1108→  if (io) {
  1109→    io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1110→      phase: 'starting', total: 0, processed: 0, added: 0, failed: 0,
  1111→    })
  1112→  }
  1113→
  1114→  try {
  1115→    // ===== Step 1: Find all mail.message records we posted to res.partner chatter =====
  1116→    // We search by body containing "WhatsApp" (our marker — every message we post
  1117→    // includes "<strong> 📱 WhatsApp Enviada:" or "Recebida:"). This avoids
  1118→    // pulling notes/activities posted by users manually in Odoo.
  1119→    console.log(`[OdooSync] Scanning mail.message for WhatsApp chatter (limit=${limit})...`)
  1120→    const domain = [
  1121→      ['model', '=', 'res.partner'],
  1122→      ['body', 'ilike', 'WhatsApp'],
  1123→    ]
  1124→    const messages = await odooExecuteKw('mail.message', 'search_read', [domain], {
  1125→      fields: ['id', 'res_id', 'body', 'date', 'create_date'],
  1126→      order: 'date asc',
  1127→      limit,
  1128→    })
  1129→
  1130→    if (!messages || messages.length === 0) {
  1131→      console.log('[OdooSync] No WhatsApp chatter messages found in Odoo')
  1132→      odooHistorySyncLastRun = { at: new Date().toISOString(), partnersProcessed: 0, messagesAdded: 0, partnersFailed: 0, durationMs: Date.now() - startedAt }
  1133→      if (io) {
  1134→        io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1135→          phase: 'complete', total: 0, processed: 0, added: 0, failed: 0,
  1136→        })
  1137→      }
  1138→      return { partnersProcessed: 0, messagesAdded: 0, partnersFailed: 0 }
  1139→    }
  1140→
  1141→    // ===== Step 2: Group by partner ID (res_id) =====
  1142→    const partnerIds = [...new Set(messages.map(m => m.res_id).filter(id => typeof id === 'number' && id > 0))]
  1143→    console.log(`[OdooSync] Found ${messages.length} chatter messages across ${partnerIds.length} partners`)
  1144→
  1145→    if (io) {
  1146→      io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1147→        phase: 'fetching_partners', total: partnerIds.length, processed: 0, added: 0, failed: 0,
  1148→      })
  1149→    }
  1150→
  1151→    // ===== Step 3: Read partner phone fields =====
  1152→    // We read whichever phone-like fields exist on res.partner (detected dynamically
  1153→    // to avoid the Odoo 19.4 "Invalid field res.partner.mobile" error).
  1154→    const partnerFields = await getAvailableFields('res.partner')
  1155→    const readFields = ['id', 'name']
  1156→    if (partnerFields.has('phone')) readFields.push('phone')
  1157→    if (partnerFields.has('mobile')) readFields.push('mobile')
  1158→    if (partnerFields.has('whatsapp')) readFields.push('whatsapp')
  1159→    if (partnerFields.has('whatsapp_number')) readFields.push('whatsapp_number')
  1160→
  1161→    const partners = await odooExecuteKw('res.partner', 'read', [partnerIds], { fields: readFields })
  1162→    const partnerMap = new Map()
  1163→    for (const p of (partners || [])) partnerMap.set(p.id, p)
  1164→
  1165→    // ===== Step 4: For each partner, find/create local conversation + pull chatter =====
  1166→    let processed = 0
  1167→    let totalAdded = 0
  1168→    let failed = 0
  1169→    for (const partnerId of partnerIds) {
  1170→      const p = partnerMap.get(partnerId)
  1171→      if (!p) { failed++; continue }
  1172→
  1173→      // Pick the first non-empty phone-like field
  1174→      const phoneRaw = p.phone || p.mobile || p.whatsapp || p.whatsapp_number || ''
  1175→      const phoneDigits = String(phoneRaw).replace(/\D/g, '')
  1176→      if (phoneDigits.length < 7) {
  1177→        console.log(`[OdooSync] Partner ${partnerId} (${p.name || '?'}) has no valid phone — skipping`)
  1178→        failed++
  1179→        continue
  1180→      }
  1181→
  1182→      // Build the WhatsApp JID from the phone digits
  1183→      const jid = `${phoneDigits}@s.whatsapp.net`
  1184→      if (!isValidPhoneJid(jid)) {
  1185→        console.log(`[OdooSync] Partner ${partnerId} phone "${phoneRaw}" → invalid JID — skipping`)
  1186→        failed++
  1187→        continue
  1188→      }
  1189→
  1190→      // Find or create the local conversation
  1191→      let conv = conversations.get(jid)
  1192→      if (!conv) {
  1193→        conv = getOrCreateConversation(jid, p.name || null)
  1194→        if (conv && p.name && !contactNames.has(jid)) {
  1195→          contactNames.set(jid, p.name)
  1196→          conv.name = p.name
  1197→        }
  1198→      } else if (p.name && !conv.name) {
  1199→        conv.name = p.name
  1200→        if (!contactNames.has(jid)) contactNames.set(jid, p.name)
  1201→      }
  1202→
  1203→      if (!conv) { failed++; continue }
  1204→
  1205→      // Pull the chatter (this function already dedupes by externalId + content+timestamp)
  1206→      try {
  1207→        const added = await pullOdooChatterIntoConversation(jid, 'res.partner', partnerId, 500)
  1208→        totalAdded += added
  1209→        processed++
  1210→        if (added > 0) {
  1211→          console.log(`[OdooSync] ✓ Partner ${partnerId} (${p.name || phoneDigits}): +${added} messages`)
  1212→        }
  1213→      } catch (err) {
  1214→        console.error(`[OdooSync] Partner ${partnerId} pull failed: ${err.message}`)
  1215→        failed++
  1216→      }
  1217→
  1218→      // Notify frontend of the link
  1219→      if (io) {
  1220→        io.of('/odoo').emit('odoo:conversation:linked', { jid, model: 'res.partner', recordId: partnerId })
  1221→      }
  1222→
  1223→      // Throttle to avoid hammering Odoo (rate-limit safe)
  1224→      await new Promise(r => setTimeout(r, 50))
  1225→
  1226→      // Progress update every 5 partners
  1227→      if (io && processed % 5 === 0) {
  1228→        io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1229→          phase: 'processing', total: partnerIds.length, processed, added: totalAdded, failed,
  1230→        })
  1231→      }
  1232→    }
  1233→
  1234→    // ===== Step 5: Sort all affected conversations + emit final update =====
  1235→    for (const [, conv] of conversations) {
  1236→      conv.messages.sort((a, b) => a.timestamp.getTime() - b.timestamp.getTime())
  1237→    }
  1238→
  1239→    markConversationsDirty()
  1240→    if (io) {
  1241→      io.of('/whatsapp').emit('whatsapp:conversations', getSortedConversations())
  1242→      io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1243→        phase: 'complete', total: partnerIds.length, processed, added: totalAdded, failed,
  1244→      })
  1245→    }
  1246→
  1247→    const durationMs = Date.now() - startedAt
  1248→    odooHistorySyncLastRun = { at: new Date().toISOString(), partnersProcessed: processed, messagesAdded: totalAdded, partnersFailed: failed, durationMs }
  1249→    console.log(`[OdooSync] ✓ DONE in ${durationMs}ms — partners processed: ${processed}, messages added: ${totalAdded}, failed: ${failed}`)
  1250→    return { partnersProcessed: processed, messagesAdded: totalAdded, partnersFailed: failed }
  1251→  } catch (err) {
  1252→    console.error(`[OdooSync] FAILED:`, err.message)
  1253→    if (io) {
  1254→      io.of('/whatsapp').emit('whatsapp:odoo-sync-progress', {
  1255→        phase: 'error', error: err.message,
  1256→      })
  1257→    }
  1258→    return { error: err.message }
  1259→  } finally {
  1260→    odooHistorySyncInProgress = false
  1261→  }
  1262→}

  1448→    })
  1449→
  1450→    // v7.17: FIRST PASS — Build a LID → phone JID map from messages.
  1451→    // WhatsApp's new privacy format sends chat IDs as @lid (Linked Identity Device).
  1452→    // The actual phone JID is in msg.key.senderPn for each message.
  1453→    // We scan all messages first to build this map, then use it to resolve
  1454→    // LID chats to phone JIDs so active conversations appear in the Conversas list.

  1459→      if (rawJid && rawJid.endsWith('@lid')) {
  1460→        const phoneCandidate = msg.key.senderPn || msg.key.participant
  1461→        if (phoneCandidate) {
  1462→          const normalized = normalizeJid(phoneCandidate)
  1463→          if (normalized) {
