     4→// Each logged-in middleware user has their own UserSession instance:

    33→// Baileys can deliver JIDs in several forms:
    34→//   5511999888777@s.whatsapp.net            — canonical DM
    35→//   5511999888777:7@s.whatsapp.net          — DM with device suffix (multi-device send)
    36→//   5511999888777:7:3@s.whatsapp.net        — DM with device + agent suffix
    37→//   5511999888777@lid                        — Linked Identity (newer protocol)
    38→// We normalize everything to canonical form: <digits>@s.whatsapp.net
    39→function normalizeJid(jid) {
    40→  if (!jid || typeof jid !== 'string') return null
    41→  const atIdx = jid.indexOf('@')
    42→  if (atIdx < 0) return null
    43→  const server = jid.slice(atIdx + 1)
    44→  if (server !== 's.whatsapp.net') return null
    45→  const userCombined = jid.slice(0, atIdx)
    46→  const user = userCombined.split(':')[0].split('_')[0]
    47→  if (!/^\d{7,}$/.test(user)) return null
    48→  return `${user}@s.whatsapp.net`
    49→}

    65→// Normalize a phone number (strip non-digits) to a WhatsApp JID
    66→function normalizePhoneToJid(phone) {
    67→  if (!phone) return null
    68→  const digits = String(phone).replace(/\D/g, '')
    69→  if (digits.length < 7) return null
    70→  return `${digits}@s.whatsapp.net`

    73→function escapeHtml(text) {
    74→  return String(text || '')
    75→    .replace(/&/g, '&amp;')
    76→    .replace(/</g, '&lt;')
    77→    .replace(/>/g, '&gt;')
    78→    .replace(/\n/g, '<br/>')
    79→}
    80→
    81→function stripHtml(html) {
    82→  if (!html) return ''
    83→  let text = String(html)
    84→    .replace(/<br\s*\/?>/gi, '\n')
    85→    .replace(/<\/p>/gi, '\n')
    86→    .replace(/<p[^>]*>/gi, '')
    87→    .replace(/<[^>]+>/g, '')
    88→    .replace(/&nbsp;/g, ' ')
    89→    .replace(/&amp;/g, '&')
    90→    .replace(/&lt;/g, '<')
    91→    .replace(/&gt;/g, '>')
    92→    .replace(/&quot;/g, '"')
    93→    .replace(/&#39;/g, "'")
    94→    .trim()
    95→  text = text.replace(/\n{3,}/g, '\n\n')
    96→  return text
    97→}

    99→// v7.28: Parse a chatter body produced by buildChatterBody() back into clean
   100→// message components (direction, timestamp, text). The chatter body format is:
   101→//   <div><strong>📱 WhatsApp {Enviada|Recebida}:</strong> {ts}<br/>{mediaLine}<span>{text}</span></div>
   102→// We want to extract ONLY the user-visible text (the <span> content), detect
   103→// the direction from "Enviada" vs "Recebida", and parse the timestamp.
   104→// Falls back to stripHtml() for any non-conforming body (e.g. messages posted
   105→// by other Odoo users from the web UI — those are returned as-is, plain text).
   106→function parseChatterBody(body) {
   107→  if (!body) return { fromMe: false, timestamp: null, text: '' }
   108→  const s = String(body)
   109→
   110→  // Detect direction from the "WhatsApp Enviada/Recebida:" marker
   111→  const isSent = /WhatsApp\s+Enviada/i.test(s)
   112→  const fromMe = isSent
   113→
   114→  // Extract timestamp from "(...)" — buildChatterBody uses toLocaleString('pt-BR')
   115→  // inside parentheses as part of the label. Actually buildChatterBody does NOT
   116→  // wrap in parens — it inserts the timestamp directly after the colon.
   117→  // But older messages may have the "(date)" format. Support both.
   118→  let timestamp = null
   119→  const parenMatch = s.match(/\(([^)]+)\)/)
   120→  if (parenMatch) {
   121→    const parsed = new Date(parenMatch[1])
   122→    if (!isNaN(parsed.getTime())) timestamp = parsed
   123→  }
   124→
   125→  // Extract the text content from the <span>...</span> block (the real message)
   126→  let text = ''
   127→  const spanMatch = s.match(/<span[^>]*>([\s\S]*?)<\/span>/i)
   128→  if (spanMatch) {
   129→    text = stripHtml(spanMatch[1])
   130→  } else {
   131→    // No <span> — this isn't a WhatsApp chatter message produced by us.
   132→    // It's probably a note posted by a human via the Odoo UI, or a different
   133→    // format entirely. Strip HTML and return the whole body as plain text.
   134→    text = stripHtml(s)
   135→    // If the body still starts with the WhatsApp metadata header, strip it.
   136→    // This handles the case where buildChatterBody emitted no <span> because
   137→    // textContent was empty (media-only message).
   138→    const headerMatch = text.match(/^📱\s*WhatsApp\s+(Enviada|Recebida)[:\s]*(.*)$/is)
   139→    if (headerMatch) {
   140→      // For media-only messages, the "text" will be empty after the header.
   141→      // Strip the header line and keep the remaining content (media label).
   142→      const remaining = text.slice(headerMatch[0].length).trim()
   143→      // Try to detect a media label line like "🎙️ Áudio"
   144→      const mediaMatch = remaining.match(/^([🎙️🖼️📄🎬🏷️][^\n]*)/)
   145→      text = mediaMatch ? mediaMatch[1] : remaining
   146→    }
   147→  }
   148→
   149→  // Try to parse timestamp from the label after the colon (no parens case)
   150→  if (!timestamp) {
   151→    // Look for a date pattern like "14/08/2026, 10:30:45" or "14/08/2026 10:30"
   152→    const dateMatch = s.match(/(\d{1,2}\/\d{1,2}\/\d{4})[,\s]+(\d{1,2}:\d{2}(?::\d{2})?)/)
   153→    if (dateMatch) {
   154→      const [_, d, t] = dateMatch
   155→      // pt-BR format: DD/MM/YYYY HH:MM:SS
   156→      const parsed = new Date(`${d.split('/').reverse().join('-')}T${t}`)
   157→      if (!isNaN(parsed.getTime())) timestamp = parsed
   158→    }
   159→  }
   160→
   161→  return { fromMe, timestamp, text }
   162→}

   169→// v7.19: Build a plain-text transcript of the WhatsApp conversation that will
   170→// be written to the `description` field (Notes) of the CRM lead/opportunity.
   171→function buildConversationTranscript(messages) {
   172→  if (!Array.isArray(messages) || messages.length === 0) return ''
   173→  const sorted = [...messages].sort(
   174→    (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   175→  )
   176→  const lines = sorted.map((m) => {
   177→    const direction = m.fromMe ? 'Eu' : 'Contato'
   178→    const dateLabel = new Date(m.timestamp).toLocaleString('pt-BR')
   179→    const mediaLabel = m.mediaType ? ` [${m.mediaType}]` : ''
   180→    const text = m.textContent && m.textContent.trim() ? m.textContent : '[Mídia]'
   181→    return `[${dateLabel}] ${direction}${mediaLabel}: ${text}`
   182→  })
   183→  return `Conversa WhatsApp:\n${lines.join('\n')}`
   184→}

   295→  async loadOdooConfig() {
   296→    // If per-user creds are already set, nothing to do
   297→    if (this.odooConfig.url && this.odooConfig.db && this.odooConfig.username && this.odooConfig.password) {
   298→      this.usingGlobalOdooConfig = false
   299→      return
   300→    }

   302→    try {
   303→      const globalConfig = await this.prisma.odooConfig.findFirst({
   304→        where: { active: true },
   305→        orderBy: { updatedAt: 'desc' },
   306→      })
   307→      if (globalConfig) {
   308→        this.odooConfig = {
   309→          url: globalConfig.url,
   310→          db: globalConfig.db,
   311→          username: globalConfig.username,
   312→          password: globalConfig.password,
   313→          uid: null,
   314→        }
   315→        this.usingGlobalOdooConfig = true
   316→        console.log(`[UserSession:${this.userId}] Using GLOBAL OdooConfig (no per-user creds set)`)
   317→      } else {
   318→        this.usingGlobalOdooConfig = true
   319→        console.log(`[UserSession:${this.userId}] No Odoo config (per-user or global) — Odoo features disabled`)
   320→      }
   321→    } catch (err) {
   322→      console.error(`[UserSession:${this.userId}] loadOdooConfig failed: ${err.message}`)
   323→    }

   326→  async start() {
   327→    if (this.isStarted || this.isStarting) return
   328→    this.isStarting = true
   329→    try {
   330→      fs.mkdirSync(this.authFolder, { recursive: true })
   331→      fs.mkdirSync(this.mediaDir, { recursive: true })
   332→
   333→      this.loadConversationsFromDisk()
   334→
   335→      // Persist conversations every 30 seconds (debounced — only if changed)
   336→      this.persistTimer = setInterval(() => {
   337→        if (this.conversationDirty && Date.now() - this.lastPersistTs > 25_000) {
   338→          this.conversationDirty = false
   339→          this.lastPersistTs = Date.now()
   340→          this.persistConversationsToDisk()
   341→        }
   342→      }, 30_000)
   343→
   344→      console.log(`[UserSession:${this.userId}] Starting — user=${this.user.email}, hasSavedSession=${this.hasSavedSession}, usingGlobalOdooConfig=${this.usingGlobalOdooConfig}`)
   345→
   346→      // Load Odoo config from DB (if needed) before starting auto-auth
   347→      await this.loadOdooConfig()
   348→
   349→      // Start WA + Odoo (fire-and-forget — they self-retry internally)
   350→      this.connectWhatsApp().catch(err => console.error(`[UserSession:${this.userId}] WA startup error:`, err.message))
   351→      this.startOdooAutoAuth().catch(err => console.error(`[UserSession:${this.userId}] Odoo startup error:`, err.message))
   352→
   353→      this.isStarted = true
   354→    } finally {
   355→      this.isStarting = false
   356→    }
   357→  }
   358→
   359→  stop() {
   360→    try {
   361→      if (this.persistTimer) { clearInterval(this.persistTimer); this.persistTimer = null }
   362→      if (this.watchdogTimer) { clearInterval(this.watchdogTimer); this.watchdogTimer = null }
   363→      if (this.waReconnectTimer) { clearTimeout(this.waReconnectTimer); this.waReconnectTimer = null }
   364→      if (this.odooReauthTimer) { clearInterval(this.odooReauthTimer); this.odooReauthTimer = null }
   365→      try { this.waSocket?.end(undefined) } catch {}
   366→      this.persistConversationsToDisk()
   367→      console.log(`[UserSession:${this.userId}] Stopped`)
   368→    } catch (err) {
   369→      console.error(`[UserSession:${this.userId}] Stop error:`, err.message)
   370→    }
   371→  }

   373→  // v7.24 (R6): Pre-deploy backup — dump this user's WA creds + conversation
   374→  // state to Odoo chatter so the data survives even if the disk is wiped
   375→  // during a Render deploy. Posted as a mail.message on a designated
   376→  // res.partner (the user's own partner if findable, otherwise partner 1).
   377→  //
   378→  // Format: subject `[BACKUP-CREDS] User <email> — <timestamp>`, body is
   379→  // JSON-stringified { creds, conversations }. If the JSON is too large
   380→  // (>600 KB per chunk — Odoo's chatter body limit is ~1MB), it's split
   381→  // into N chunks posted as separate messages with subject suffix `[1/N]`.
   382→  //
   383→  // Returns: { success, chunksPosted, partnerId, error? }
   384→  async backupToOdoo() {
   385→    if (!this.odooConfig.uid) {
   386→      return { success: false, chunksPosted: 0, error: 'Odoo not authenticated' }
   387→    }
   388→    try {
   389→      // Persist current state to disk first (so the backup reflects the
   390→      // latest conversations, not the last periodic save).
   391→      this.persistConversationsToDisk()
   392→
   393→      // Read creds.json
   394→      const credsPath = path.join(this.authFolder, 'creds.json')
   395→      let credsJson = null
   396→      try {
   397→        if (fs.existsSync(credsPath)) {
   398→          credsJson = JSON.parse(fs.readFileSync(credsPath, 'utf8'))
   399→        }
   400→      } catch (err) {
   401→        console.warn(`[Backup:${this.userId}] Could not read creds.json:`, err.message)
   402→      }
   403→
   404→      // Read conv_state_<userId>.json
   405→      let convState = null
   406→      try {
   407→        if (fs.existsSync(this.stateFile)) {
   408→          convState = JSON.parse(fs.readFileSync(this.stateFile, 'utf8'))
   409→        }
   410→      } catch (err) {
   411→        console.warn(`[Backup:${this.userId}] Could not read conv state:`, err.message)
   412→      }
   413→
   414→      // Build the JSON payload
   415→      const payload = JSON.stringify({
   416→        userId: this.userId,
   417→        email: this.user.email,
   418→        backedUpAt: new Date().toISOString(),
   419→        version: '7.24',
   420→        creds: credsJson,
   421→        conversations: convState,
   422→      })
   423→
   424→      // Find a partner to post to. Strategy: look up res.partner by the
   425→      // user's email; if not found, fall back to partner id 1
   426→      // (Administrator in most Odoo installs).
   427→      let backupPartnerId = 1
   428→      try {
   429→        const partners = await this.odooExecuteKw('res.partner', 'search_read', [
   430→          [['email', '=', this.user.email]],
   431→        ], { fields: ['id', 'name'], limit: 1 })
   432→        if (partners && partners.length > 0) {
   433→          backupPartnerId = partners[0].id
   434→        } else {
   435→          // Fall back: try admin user's partner (res.users where login = 'admin')
   436→          const adminUsers = await this.odooExecuteKw('res.users', 'search_read', [
   437→            [['login', '=', 'admin']],
   438→          ], { fields: ['partner_id'], limit: 1 })
   439→          if (adminUsers && adminUsers.length > 0 && adminUsers[0].partner_id) {
   440→            backupPartnerId = adminUsers[0].partner_id[0]
   441→          }
   442→        }
   443→      } catch (err) {
   444→        console.warn(`[Backup:${this.userId}] Partner lookup failed, using id=1:`, err.message)
   445→      }
   446→
   447→      // Split into chunks of ~600 KB (base64 expansion + Odoo overhead
   448→      // means 600 KB raw → ~800 KB on the wire, safely under 1 MB).
   449→      const CHUNK_SIZE = 600_000
   450→      const chunks = []
   451→      for (let i = 0; i < payload.length; i += CHUNK_SIZE) {
   452→        chunks.push(payload.slice(i, i + CHUNK_SIZE))
   453→      }
   454→      if (chunks.length === 0) chunks.push('')
   455→
   456→      const tsLabel = new Date().toISOString()
   457→      const subject = `[BACKUP-CREDS] User ${this.user.email} — ${tsLabel}`
   458→      const total = chunks.length
   459→
   460→      for (let i = 0; i < chunks.length; i++) {
   461→        const partNum = i + 1
   462→        const body = `<div><strong>${subject}</strong> [${partNum}/${total}]<br/>` +
   463→          `<em>Backup dos dados de conexão e conversas — não editar.</em>` +
   464→          `<pre style="font-size:10px; white-space:pre-wrap; word-break:break-all;">${escapeHtml(chunks[i])}</pre></div>`
   465→        try {
   466→          await this.odooPostMessage('res.partner', backupPartnerId, body)
   467→        } catch (err) {
   468→          console.error(`[Backup:${this.userId}] Chunk ${partNum}/${total} failed:`, err.message)
   469→          return { success: false, chunksPosted: i, partnerId: backupPartnerId, error: err.message }
   470→        }
   471→      }
   472→
   473→      console.log(`[Backup:${this.userId}] ✓ Posted ${total} chunk(s) to partner ${backupPartnerId}`)
   474→      return { success: true, chunksPosted: total, partnerId: backupPartnerId }
   475→    } catch (err) {
   476→      console.error(`[Backup:${this.userId}] Error:`, err.message)
   477→      return { success: false, chunksPosted: 0, error: err.message }
   478→    }
   479→  }

   485→  persistConversationsToDisk() {
   486→    try {
   487→      const snapshot = {
   488→        version: 1,
   489→        userId: this.userId,
   490→        savedAt: new Date().toISOString(),
   491→        conversations: Array.from(this.conversations.values()).map(c => ({
   492→          jid: c.jid,
   493→          name: c.name,

   694→  odooAuthenticate() {
   695→    return new Promise((resolve, reject) => {
   696→      // v7.29.2: 30s timeout on auth — same reasoning as odooExecuteKw
   697→      let settled = false
   698→      const timer = setTimeout(() => {
   699→        if (settled) return
   700→        settled = true
   701→        reject(new Error('Odoo authenticate timeout after 30s'))
   702→      }, 30_000)
   703→      try {
   704→        const client = this.makeXmlRpcClient('/xmlrpc/2/common')
   705→        client.methodCall('authenticate', [this.odooConfig.db, this.odooConfig.username, this.odooConfig.password, {}], (error, value) => {
   706→          if (settled) return
   707→          settled = true
   708→          clearTimeout(timer)
   709→          if (error) reject(error)
   710→          else if (!value) reject(new Error('Authentication failed - invalid credentials'))
   711→          else resolve(value)
   712→        })
   713→      } catch (err) {
   714→        if (settled) return
   715→        settled = true
   716→        clearTimeout(timer)
   717→        reject(err)
   718→      }
   719→    })
   720→  }

   722→  odooExecuteKw(model, method, args, kwargs = {}) {
   723→    return new Promise((resolve, reject) => {
   724→      if (!this.odooConfig.uid) {
   725→        reject(new Error('Not authenticated with Odoo'))
   726→        return
   727→      }
   728→      // v7.29.2: Add 30s timeout. Without this, if Odoo hangs (network
   729→      // glitch, server slow), the Promise never resolves and the calling
   730→      // code stays blocked forever — which blocks the WA message handler
   731→      // and the user perceives it as "the page fell".
   732→      let settled = false
   733→      const timer = setTimeout(() => {
   734→        if (settled) return
   735→        settled = true
   736→        reject(new Error(`Odoo ${model}.${method} timeout after 30s`))
   737→      }, 30_000)
   738→      try {
   739→        const client = this.makeXmlRpcClient('/xmlrpc/2/object')
   740→        client.methodCall('execute_kw', [this.odooConfig.db, this.odooConfig.uid, this.odooConfig.password, model, method, args, kwargs], (error, value) => {
   741→          if (settled) return
   742→          settled = true
   743→          clearTimeout(timer)
   744→          if (error) reject(error)
   745→          else resolve(value)
   746→        })
   747→      } catch (err) {
   748→        if (settled) return
   749→        settled = true
   750→        clearTimeout(timer)
   751→        reject(err)
   752→      }
   753→    })
   754→  }

   796→  // v7.20: Build a phone-number search domain that ONLY uses fields the model has

   798→    if (!phone) return []

   800→    const usable = candidateFields.filter(f => available.has(f))
   801→    if (usable.length === 0) {
   802→      return [['name', 'ilike', phone]]
   803→    }
   804→    if (usable.length === 1) {
   805→      return [[usable[0], 'ilike', phone]]

   808→    for (let i = 0; i < usable.length - 1; i++) domain.push('|')
   809→    for (const f of usable) domain.push([f, 'ilike', phone])
   810→    return domain
   811→  }

   939→  async autoSyncWhatsAppMessage(data) {

   968→      if (this.autoSyncSettings.autoCreateContact) {

  1207→  async pullOdooChatterIntoConversation(jid, model, recordId, limit = 200) {
  1208→    if (!this.odooConfig.uid) return 0
  1209→    if (!isValidPhoneJid(jid)) return 0
  1210→    const conv = this.conversations.get(jid)
  1211→    if (!conv) return 0
  1212→
  1213→    try {
  1214→      const domain = [
  1215→        ['model', '=', model],
  1216→        ['res_id', '=', recordId],
  1217→      ]
  1218→      const messages = await this.odooExecuteKw('mail.message', 'search_read', [domain], {
  1219→        fields: ['id', 'body', 'author_id', 'email_from', 'date', 'message_type', 'subtype_id', 'create_date'],
  1220→        order: 'date asc',
  1221→        limit,
  1222→      })
  1223→
  1224→      let added = 0
  1225→      for (const msg of (messages || [])) {
  1226→        const body = msg.body || ''
  1227→        // v7.28: Use parseChatterBody() to extract ONLY the user-visible
  1228→        // text from the chatter body (the content inside <span>), instead
  1229→        // of the whole stripped-HTML which included "📱 WhatsApp Enviada: ..."
  1230→        // metadata prefixes that made messages look ugly in the chat view.
  1231→        const parsed = parseChatterBody(body)

  1233→        const plainText = parsed.text
  1234→
  1235→        let timestamp = parsed.timestamp
  1236→        if (!timestamp) timestamp = new Date(msg.date || msg.create_date || Date.now())
  1237→
  1238→        const externalId = `odoo-${msg.id}`
  1239→        if (conv.messages.some(m => m.whatsappId === externalId)) continue
  1240→        const isDup = conv.messages.some(m =>
  1241→          m.fromMe === fromMe &&
  1242→          m.textContent === plainText &&
  1243→          Math.abs(m.timestamp.getTime() - timestamp.getTime()) < 5000
  1244→        )
  1245→        if (isDup) continue
  1246→

  1248→          id: externalId,
  1249→          whatsappId: externalId,
  1250→          fromMe,
  1251→          textContent: plainText || null,
  1252→          mediaType: null,
  1253→          timestamp,
  1254→          status: 'delivered',
  1255→        })
  1256→        added++
  1257→      }
  1258→
  1259→      if (added > 0) {
  1260→        conv.messages.sort((a, b) => a.timestamp.getTime() - b.timestamp.getTime())
  1261→        const last = conv.messages[conv.messages.length - 1]
  1262→        if (!conv.lastMessageAt || last.timestamp > conv.lastMessageAt) {
  1263→          conv.lastMessageAt = last.timestamp
  1264→          if (last.textContent) conv.lastMessage = last.textContent
  1265→        }
  1266→      }
  1267→      return added
  1268→    } catch (err) {
  1269→      console.error(`[Odoo:${this.userId}] Chatter pull failed for ${model}#${recordId}:`, err.message)
  1270→      return 0
  1271→    }
  1272→  }
