   191→   191→function makeXmlRpcClient(path) {
   192→   192→  const url = new URL(odooConfig.url)
   193→   193→  const isHttps = url.protocol === 'https:'
   194→   194→  const options = {
   195→   195→    host: url.hostname,
   196→   196→    port: parseInt(url.port) || (isHttps ? 443 : 80),
   197→   197→    path,
   198→   198→  }
   199→   199→  return isHttps ? xmlrpc.createSecureClient(options) : xmlrpc.createClient(options)
   200→   200→}

   202→   202→function odooAuthenticate() {
   203→   203→  return new Promise((resolve, reject) => {
   204→   204→    const client = makeXmlRpcClient('/xmlrpc/2/common')
   205→   205→    client.methodCall('authenticate', [
   206→   206→      odooConfig.db, odooConfig.username, odooConfig.password, {},
   207→   207→    ], (error, value) => {
   208→   208→      if (error) return reject(error)
   209→   209→      if (!value) return reject(new Error('Authentication failed - invalid credentials'))
   210→   210→      resolve(value)
   211→   211→    })
   212→   212→  })
   213→   213→}

   215→   215→function odooExecuteKw(model, method, args, kwargs = {}) {
   216→   216→  return new Promise((resolve, reject) => {
   217→   217→    if (!odooConfig.uid) {
   218→   218→      return reject(new Error('Not authenticated with Odoo'))
   219→   219→    }
   220→   220→    const client = makeXmlRpcClient('/xmlrpc/2/object')
   221→   221→    client.methodCall('execute_kw', [
   222→   222→      odooConfig.db, odooConfig.uid, odooConfig.password, model, method, args, kwargs,
   223→   223→    ], (error, value) => {
   224→   224→      if (error) reject(error)
   225→   225→      else resolve(value)
   226→   226→    })
   227→   227→  })
   228→   228→}

   230→   230→// Smart Field Detection
   231→   231→async function getAvailableFields(model) {
   232→   232→  if (modelFieldsCache.has(model)) return modelFieldsCache.get(model)
   233→   233→  try {
   234→   234→    const fields = await odooExecuteKw(model, 'fields_get', [], { attributes: ['string', 'type'] })
   235→   235→    const fieldNames = new Set(Object.keys(fields))
   236→   236→    modelFieldsCache.set(model, fieldNames)
   237→   237→    return fieldNames
   238→   238→  } catch (error) {
   239→   239→    console.error(`[Odoo] Failed to get fields for ${model}:`, error.message)
   240→   240→    return new Set()
   241→   241→  }
   242→   242→}

   244→   244→async function filterExistingFields(model, requestedFields) {
   245→   245→  const available = await getAvailableFields(model)
   246→   246→  const existing = requestedFields.filter(f => available.has(f))
   247→   247→  if (!existing.includes('id') && available.has('id')) existing.unshift('id')
   248→   248→  if (!existing.includes('name') && available.has('name')) existing.push('name')
   249→   249→  return existing
   250→   250→}

   252→   252→async function buildSafeValues(model, values) {
   253→   253→  const available = await getAvailableFields(model)
   254→   254→  const safe = {}
   255→   255→  for (const [key, value] of Object.entries(values)) {
   256→   256→    if (available.has(key)) {
   257→   257→      safe[key] = value
   258→   258→    } else {
   259→   259→      console.log(`[Odoo] Field "${key}" does not exist on ${model}, skipping`)
   260→   260→    }
   261→   261→  }
   262→   262→  return safe
   263→   263→}

   366→   366→async function autoSyncWhatsAppMessage(data) {
   367→   367→  const result = {
   368→   368→    partnerId: null,
   369→   369→    leadId: null,
   370→   370→    mailMessageId: null,
   371→   371→    activityId: null,
   372→   372→    created: { partner: false, lead: false },
   373→   373→    errors: [],
   374→   374→  }
   375→   375→
   376→   376→  if (!autoSyncSettings.enabled || !odooConfig.uid) return result
   377→   377→
   378→   378→  console.log(`[AutoSync] Processing message from ${data.phone} (${data.pushName || 'unknown'})`)
   379→   379→
   380→   380→  try {
   381→   381→    // Step 1: Create or update contact
   382→   382→    if (autoSyncSettings.autoCreateContact) {
   383→   383→      const contactName = data.pushName || `WhatsApp ${data.phone}`
   384→   384→      const domain = ['|', ['phone', 'ilike', data.phone], ['mobile', 'ilike', data.phone]]
   385→   385→      const contactValues = { name: contactName, phone: data.phone, mobile: data.phone }
   386→   386→
   387→   387→      const partnerFields = await getAvailableFields('res.partner')
   388→   388→      if (partnerFields.has('whatsapp')) contactValues.whatsapp = data.phone
   389→   389→      if (partnerFields.has('whatsapp_number')) contactValues.whatsapp_number = data.phone
   390→   390→
   391→   391→      const partnerResult = await odooSearchOrCreate('res.partner', domain, contactValues)
   392→   392→      result.partnerId = partnerResult.id
   393→   393→      result.created.partner = partnerResult.created
   394→   394→
   395→   395→      const cached = phoneToPartnerCache.get(data.phone)
   396→   396→      if (cached) {
   397→   397→        cached.partnerId = partnerResult.id
   398→   398→      } else {
   399→   399→        phoneToPartnerCache.set(data.phone, { partnerId: partnerResult.id, leadId: null, leadCreated: false })
   400→   400→      }
   401→   401→      console.log(`[AutoSync] Contact ${partnerResult.created ? 'created' : 'updated'}: res.partner#${partnerResult.id}`)
   402→   402→    }
   403→   403→
   404→   404→    // Step 2: Create lead for new conversations
   405→   405→    if (autoSyncSettings.autoCreateLead && !data.fromMe && result.partnerId) {
   406→   406→      const cached = phoneToPartnerCache.get(data.phone)
   407→   407→      if (cached && cached.leadId && cached.leadCreated) {
   408→   408→        result.leadId = cached.leadId
   409→   409→      } else {
   410→   410→        const existingLeads = await odooSearch('crm.lead', [
   411→   411→          ['partner_id', '=', result.partnerId],
   412→   412→          ['name', 'like', autoSyncSettings.leadPrefix],
   413→   413→          ['type', '=', 'lead'],
   414→   414→        ], ['id', 'name'], 1)
   415→   415→
   416→   416→        if (existingLeads && existingLeads.length > 0) {
   417→   417→          result.leadId = existingLeads[0].id
   418→   418→          if (cached) { cached.leadId = existingLeads[0].id; cached.leadCreated = true }
   419→   419→          console.log(`[AutoSync] Found existing lead: crm.lead#${existingLeads[0].id}`)
   420→   420→        } else {
   421→   421→          const leadName = `${autoSyncSettings.leadPrefix}${data.pushName || data.phone}`
   422→   422→          const leadValues = {
   423→   423→            name: leadName,
   424→   424→            type: 'lead',
   425→   425→            partner_id: result.partnerId,
   426→   426→            phone: data.phone,
   427→   427→            description: `Conversa iniciada via WhatsApp em ${new Date().toLocaleString('pt-BR')}`,
   428→   428→          }
   429→   429→
   430→   430→          const leadFields = await getAvailableFields('crm.lead')
   431→   431→          if (leadFields.has('whatsapp_number')) leadValues.whatsapp_number = data.phone
   432→   432→
   433→   433→          if (autoSyncSettings.leadTeamId) leadValues.team_id = autoSyncSettings.leadTeamId
   434→   434→          if (autoSyncSettings.leadUserId) leadValues.user_id = autoSyncSettings.leadUserId
   435→   435→
   436→   436→          result.leadId = await odooCreate('crm.lead', leadValues)
   437→   437→          result.created.lead = true
   438→   438→
   439→   439→          if (cached) {
   440→   440→            cached.leadId = result.leadId
   441→   441→            cached.leadCreated = true
   442→   442→          } else {
   443→   443→            phoneToPartnerCache.set(data.phone, { partnerId: result.partnerId, leadId: result.leadId, leadCreated: true })
   444→   444→          }
   445→   445→          console.log(`[AutoSync] Lead created: crm.lead#${result.leadId}`)
   446→   446→        }
   447→   447→      }
   448→   448→    }
   449→   449→
   450→   450→    // Step 3: Post message in Odoo chatter
   451→   451→    if (autoSyncSettings.autoPostMessages && !data.fromMe) {
   452→   452→      const targetModel = result.leadId ? 'crm.lead' : (result.partnerId ? 'res.partner' : null)
   453→   453→      const targetId = result.leadId || result.partnerId
   454→   454→      if (targetModel && targetId) {
   455→   455→        const direction = data.fromMe ? 'Enviada' : 'Recebida'
   456→   456→        const mediaLabel = data.mediaType ? ` [${data.mediaType}]` : ''
   457→   457→        const msgBody = data.textContent
   458→   458→          ? `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel}</p><p>${escapeHtml(data.textContent)}</p>`
   459→   459→          : `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel} [Mídia]</p>`
   460→   460→        try {
   461→   461→          result.mailMessageId = await odooPostMessage(targetModel, targetId, msgBody)
   462→   462→          console.log(`[AutoSync] Message posted on ${targetModel}#${targetId}`)
   463→   463→        } catch (error) {
   464→   464→          result.errors.push(`Failed to post message: ${error.message}`)
   465→   465→        }
   466→   466→      }
   467→   467→    }
   468→   468→
   469→   469→    // Step 4: Create activity notification for first message of a new lead
   470→   470→    if (autoSyncSettings.autoCreateActivity && result.created.lead && result.leadId) {
   471→   471→      const summary = 'Nova mensagem WhatsApp'
   472→   472→      const note = `Contato ${data.pushName || data.phone} iniciou uma conversa via WhatsApp.\n\nMensagem: ${data.textContent || '[Mídia]'}`
   473→   473→      try {
   474→   474→        result.activityId = await odooCreateActivity('crm.lead', result.leadId, summary, note)
   475→   475→      } catch (error) {
   476→   476→        result.errors.push(`Failed to create activity: ${error.message}`)
   477→   477→      }
   478→   478→    }
   479→   479→  } catch (error) {
   480→   480→    result.errors.push(`Auto-sync error: ${error.message}`)
   481→   481→    console.error('[AutoSync] Error:', error.message)
   482→   482→  }
   483→   483→
   484→   484→  // Emit sync result to all connected clients
   485→   485→  if (odooNamespace) {
   486→   486→    odooNamespace.emit('odoo:autosync:result', { phone: data.phone, ...result })
   487→   487→  }
   488→   488→
   489→   489→  return result
   490→   490→}

   496→   496→// waLogger is created AFTER loadModules() in main(). This is a placeholder.
   497→   497→let waLogger = { info: () => {}, error: () => {}, warn: () => {}, child: () => ({ info: () => {}, error: () => {}, warn: () => {} }) }
   498→   498→
   499→   499→async function connectWhatsApp() {
   500→   500→  // Guard: ensure Baileys modules are loaded
   501→   501→  if (typeof makeWASocket !== 'function' || typeof useMultiFileAuthState !== 'function') {
   502→   502→    console.error('[WA] Cannot connect: Baileys modules not loaded. WhatsApp will not work.')
   503→   503→    if (waNamespace) waNamespace.emit('whatsapp:status', { connected: false, reason: 'modules_not_loaded' })
   504→   504→    return
   505→   505→  }
   506→   506→
   507→   507→  try {
   508→   508→  const { state, saveCreds } = await useMultiFileAuthState(AUTH_FOLDER)
   509→   509→  const { version } = await fetchLatestBaileysVersion()
   510→   510→
   511→   511→  waSocket = makeWASocket({
   512→   512→    version,
   513→   513→    logger: waLogger,
   514→   514→    auth: {
   515→   515→      creds: state.creds,
   516→   516→      keys: makeCacheableSignalKeyStore(state.keys, waLogger),
   517→   517→    },
   518→   518→    browser: Browsers.ubuntu('Chrome'),
   519→   519→    printQRInTerminal: false,
   520→   520→    connectTimeoutMs: 60_000,
   521→   521→    defaultQueryTimeoutMs: 30_000,
   522→   522→    keepAliveIntervalMs: 30_000, // increased from 25s for better stability
   523→   523→    markOnlineOnConnect: true,
   524→   524→    syncFullHistory: true, // Pull conversations from phone so user can create leads
   525→   525→    // Memory optimization: limit message history sync
   526→   526→    messageCaching: false,
   527→   527→  })
   528→   528→
   529→   529→  // Save credentials on update
   530→   530→  waSocket.ev.on('creds.update', saveCreds)
   531→   531→
   532→   532→  // Connection events
   533→   533→  waSocket.ev.on('connection.update', async (update) => {
   534→   534→    const { connection, lastDisconnect, qr } = update
   535→   535→    connectionState = { connection, lastDisconnect }
   536→   536→
   537→   537→    console.log(`[WA] Connection update: ${connection}`)
   538→   538→
   539→   539→    if (qr) {
   540→   540→      console.log('[WA] QR Code generated, sending to clients')
   541→   541→      if (waNamespace) waNamespace.emit('whatsapp:qr', { qr })
   542→   542→    }
   543→   543→
   544→   544→    if (connection === 'close') {
   545→   545→      const statusCode = lastDisconnect?.error?.output?.statusCode
   546→   546→      const shouldReconnect = statusCode !== DisconnectReason.loggedOut
   547→   547→
   548→   548→      console.log(`[WA] Connection closed. Status: ${statusCode}, Reconnect: ${shouldReconnect}`)
   549→   549→
   550→   550→      if (waNamespace) waNamespace.emit('whatsapp:status', {
   551→   551→        connected: false,
   552→   552→        reason: statusCode === DisconnectReason.loggedOut ? 'logged_out' : 'disconnected',
   553→   553→      })
   554→   554→
   555→   555→      if (shouldReconnect) {
   556→   556→        reconnectAttempts++
   557→   557→        // Exponential backoff: 2s, 4s, 8s, 16s, 30s, 30s, 30s...
   558→   558→        const delay = Math.min(2000 * Math.pow(2, Math.min(reconnectAttempts - 1, 4)), 30000)
   559→   559→        // Infinite reconnects (no max) — Render will restart the dyno if truly stuck
   560→   560→        console.log(`[WA] Reconnecting in ${delay}ms (attempt ${reconnectAttempts})`)
   561→   561→        setTimeout(() => connectWhatsApp(), delay)
   562→   562→      }
   563→   563→    }
   564→   564→
   565→   565→    if (connection === 'open') {
   566→   566→      reconnectAttempts = 0
   567→   567→      console.log('[WA] Connected successfully!')
   568→   568→      if (waNamespace) waNamespace.emit('whatsapp:status', { connected: true })
   569→   569→
   570→   570→      // Get profile picture for connected user
   571→   571→      try {
   572→   572→        const meId = waSocket?.user?.id
   573→   573→        if (meId) {
   574→   574→          const profilePicUrl = await waSocket.profilePictureUrl(meId, 'image')
   575→   575→          if (waNamespace) waNamespace.emit('whatsapp:me', {
   576→   576→            id: meId,
   577→   577→            name: waSocket?.user?.name,
   578→   578→            profilePicUrl,
   579→   579→          })
   580→   580→        }
   581→   581→      } catch { /* Profile pic might not be available */ }
   582→   582→    }
   583→   583→  })
   584→   584→
   585→   585→  } catch (err) {
   586→   586→    console.error('[WA] connectWhatsApp error:', err.message)
   587→   587→    if (waNamespace) waNamespace.emit('whatsapp:status', { connected: false, reason: 'connection_error' })
   588→   588→    // Retry after delay
   589→   589→    reconnectAttempts++
   590→   590→    const delay = Math.min(5000 * reconnectAttempts, 60000)
   591→   591→    console.log(`[WA] Will retry connection in ${delay}ms`)
   592→   592→    setTimeout(() => connectWhatsApp(), delay)
   593→   593→    return
   594→   594→  }
   595→   595→
   596→   596→  // ========== Contacts Sync ==========
   597→   597→  waSocket.ev.on('contacts.upsert', (contacts) => {
   598→   598→    console.log(`[WA] Contacts upsert: ${contacts.length} contacts`)
   599→   599→    for (const contact of contacts) {
   600→   600→      if (!contact.id) continue
   601→   601→      // Skip group contacts and broadcast
   602→   602→      if (contact.id.includes('@g.us') || contact.id === 'status@broadcast') continue
   603→   603→      const phone = extractPhone(contact.id)
   604→   604→      if (!phone) continue
   605→   605→      waContacts.set(contact.id, {
   606→   606→        jid: contact.id,
   607→   607→        name: contact.name || contact.notify || null,
   608→   608→        phone,
   609→   609→        notify: contact.notify || null,
   610→   610→      })
   611→   611→    }
   612→   612→    // Trim if too many
   613→   613→    if (waContacts.size > MAX_CONTACTS) {
   614→   614→      const entries = Array.from(waContacts.keys())
   615→   615→      const toRemove = entries.slice(0, entries.length - MAX_CONTACTS)
   616→   616→      toRemove.forEach(k => waContacts.delete(k))
   617→   617→    }
   618→   618→    // Emit to clients
   619→   619→    if (waNamespace) {
   620→   620→      const contactList = Array.from(waContacts.values())
   621→   621→      waNamespace.emit('whatsapp:contacts', contactList)
   622→   622→    }
   623→   623→  })
   624→   624→
   625→   625→  waSocket.ev.on('contacts.update', (updates) => {
   626→   626→    for (const update of updates) {
   627→   627→      if (!update.id) continue
   628→   628→      const existing = waContacts.get(update.id)
   629→   629→      if (existing) {
   630→   630→        if (update.name) existing.name = update.name
   631→   631→        if (update.notify) existing.notify = update.notify
   632→   632→      }
   633→   633→    }
   634→   634→    if (waNamespace) {
   635→   635→      const contactList = Array.from(waContacts.values())
   636→   636→      waNamespace.emit('whatsapp:contacts', contactList)
   637→   637→    }
   638→   638→  })
   639→   639→
   640→   640→  // ========== Chats/Conversations Sync ==========
   641→   641→  waSocket.ev.on('chats.upsert', (chats) => {
   642→   642→    console.log(`[WA] Chats upsert: ${chats.length} chats`)
   643→   643→    for (const chat of chats) {
   644→   644→      if (!chat.id) continue
   645→   645→      if (chat.id.includes('@g.us') || chat.id === 'status@broadcast') continue
   646→   646→      const conv = getOrCreateConversation(chat.id)
   647→   647→      if (chat.name) conv.name = chat.name
   648→   648→      conv.unreadCount = chat.unreadCount || 0
   649→   649→    }
   650→   650→    if (waNamespace) {
   651→   651→      const convList = Array.from(conversations.values()).map(serializeConversation)
   652→   652→      waNamespace.emit('whatsapp:conversations', convList)
   653→   653→    }
   654→   654→  })
   655→   655→
   656→   656→  // Message events
   657→   657→  waSocket.ev.on('messages.upsert', async ({ messages, type }) => {
   658→   658→    console.log(`[WA] Messages upsert: ${type}, count: ${messages.length}`)
   659→   659→
   660→   660→    for (const msg of messages) {
   661→   661→      if (!msg.key) continue
   662→   662→
   663→   663→      const jid = msg.key.remoteJid
   664→   664→      const fromMe = msg.key.fromMe || false
   665→   665→      const pushName = msg.pushName || null
   666→   666→      const msgKeyId = msg.key.id || ''
   667→   667→
   668→   668→      // Skip status/broadcast messages
   669→   669→      if (jid === 'status@broadcast') continue
   670→   670→
   671→   671→      // ========== DEDUPLICATION ==========
   672→   672→      // Skip our own echoed messages from Baileys (fromMe + append = echo)
   673→   673→      if (fromMe && type === 'append') {
   674→   674→        if (msgKeyId) processedMessageIds.add(msgKeyId)
   675→   675→        continue
   676→   676→      }
   677→   677→
   678→   678→      // Check dedup set
   679→   679→      if (msgKeyId && processedMessageIds.has(msgKeyId)) {
   680→   680→        console.log(`[WA] Skipping duplicate message: ${msgKeyId}`)
   681→   681→        continue
   682→   682→      }
   683→   683→      if (msgKeyId) {
   684→   684→        processedMessageIds.add(msgKeyId)
   685→   685→        // Prevent memory leak
   686→   686→        if (processedMessageIds.size > MAX_DEDUP_IDS) {
   687→   687→          const entries = Array.from(processedMessageIds)
   688→   688→          const toRemove = entries.slice(0, entries.length - MAX_DEDUP_IDS / 2)
   689→   689→          toRemove.forEach(id => processedMessageIds.delete(id))
   690→   690→        }
   691→   691→      }
   692→   692→
   693→   693→      // Get or create conversation
   694→   694→      const conv = getOrCreateConversation(jid, fromMe ? undefined : pushName)
   695→   695→
   696→   696→      // Extract text content
   697→   697→      const textContent =
   698→   698→        msg.message?.conversation ||
   699→   699→        msg.message?.extendedTextMessage?.text ||
   700→   700→        msg.message?.imageMessage?.caption ||
   701→   701→        msg.message?.videoMessage?.caption ||
   702→   702→        msg.message?.documentMessage?.caption ||
   703→   703→        null
   704→   704→
   705→   705→      // Determine media type
   706→   706→      let mediaType = null
   707→   707→      if (msg.message?.imageMessage) mediaType = 'image'
   708→   708→      else if (msg.message?.videoMessage) mediaType = 'video'
   709→   709→      else if (msg.message?.audioMessage) mediaType = 'audio'
   710→   710→      else if (msg.message?.documentMessage) mediaType = 'document'
   711→   711→      else if (msg.message?.stickerMessage) mediaType = 'sticker'
   712→   712→      else if (msg.message?.contactMessage) mediaType = 'contact'
   713→   713→      else if (msg.message?.locationMessage) mediaType = 'location'
   714→   714→
   715→   715→      // Only process messages with content
   716→   716→      if (!textContent && !mediaType) continue
   717→   717→
   718→   718→      const messageData = {
   719→   719→        id: msgKeyId || Math.random().toString(36).substr(2, 9),
   720→   720→        whatsappId: msgKeyId || null,
   721→   721→        fromMe,
   722→   722→        textContent,
   723→   723→        mediaType,
   724→   724→        timestamp: new Date((msg.messageTimestamp || Date.now() / 1000) * 1000),
   725→   725→        status: fromMe ? 'delivered' : 'received',
   726→   726→      }
   727→   727→
   728→   728→      // Add to conversation
   729→   729→      conv.messages.push(messageData)
   730→   730→      if (conv.messages.length > MAX_MESSAGES_PER_CONVERSATION) {
   731→   731→        conv.messages = conv.messages.slice(-MAX_MESSAGES_PER_CONVERSATION)
   732→   732→      }
   733→   733→      conv.lastMessage = textContent || `[${mediaType}]`
   734→   734→      conv.lastMessageAt = messageData.timestamp
   735→   735→      if (!fromMe) conv.unreadCount++
   736→   736→
   737→   737→      // Try to get profile picture (lazy, only once)
   738→   738→      try {
   739→   739→        if (!conv.avatarUrl && waSocket) {
   740→   740→          const picUrl = await waSocket.profilePictureUrl(jid, 'image')
   741→   741→          conv.avatarUrl = picUrl
   742→   742→        }
   743→   743→      } catch { /* No profile pic */ }
   744→   744→
   745→   745→      // Emit to connected clients
   746→   746→      if (waNamespace) {
   747→   747→        waNamespace.emit('whatsapp:message', {
   748→   748→          conversationJid: jid,
   749→   749→          message: messageData,
   750→   750→          conversation: serializeConversation(conv),
   751→   751→        })
   752→   752→        waNamespace.emit('whatsapp:conversation:update', serializeConversation(conv))
   753→   753→      }
   754→   754→
   755→   755→      // ========== AUTO-SYNC TO ODOO (direct call, no bridge) ==========
   756→   756→      if (!fromMe) {
   757→   757→        const phone = extractPhone(jid)
   758→   758→        if (phone) {
   759→   759→          // Non-blocking: fire and forget, errors logged in autoSyncWhatsAppMessage
   760→   760→          autoSyncWhatsAppMessage({
   761→   761→            jid, phone,
   762→   762→            pushName: conv.pushName,
   763→   763→            textContent, mediaType, fromMe,
   764→   764→            timestamp: messageData.timestamp.toISOString(),
   765→   765→          }).catch(err => console.error('[AutoSync] Unhandled error:', err.message))
   766→   766→
   767→   767→          // Also notify frontend about the sync
   768→   768→          if (waNamespace) {
   769→   769→            waNamespace.emit('whatsapp:odoo-sync', { jid, phone, syncing: true })
   770→   770→          }
   771→   771→        }
   772→   772→      }
   773→   773→    }
   774→   774→  })
   775→   775→}

   817→   817→function setupWhatsAppNamespace(namespace) {
   818→   818→  namespace.on('connection', (socket) => {
   819→   819→    console.log(`[WA IO] Client connected: ${socket.id}`)
   820→   820→
   821→   821→    // Send current status
   822→   822→    socket.emit('whatsapp:status', {
   823→   823→      connected: connectionState.connection === 'open',
   824→   824→    })
   825→   825→
   826→   826→    // Send current conversations list
   827→   827→    const convList = Array.from(conversations.values()).map(serializeConversation)
   828→   828→    socket.emit('whatsapp:conversations', convList)
   829→   829→
   830→   830→    // Send current contacts list
   831→   831→    const contactList = Array.from(waContacts.values())
   832→   832→    socket.emit('whatsapp:contacts', contactList)
   833→   833→
   834→   834→    // Request QR code regeneration
   835→   835→    socket.on('whatsapp:request-qr', () => {
   836→   836→      reconnectAttempts = 0
   837→   837→      if (connectionState.connection !== 'open' && waSocket) {
   838→   838→        socket.emit('whatsapp:status', { connected: false, reason: 'connecting' })
   839→   839→      } else if (!waSocket) {
   840→   840→        connectWhatsApp()
   841→   841→      }
   842→   842→    })
   843→   843→
   844→   844→    // Get conversation messages
   845→   845→    socket.on('whatsapp:get-messages', (data, callback) => {
   846→   846→      const conv = conversations.get(data.jid)
   847→   847→      callback?.(conv ? { messages: conv.messages.slice(-100) } : { messages: [] })
   848→   848→    })
   849→   849→
   850→   850→    // Get conversation info
   851→   851→    socket.on('whatsapp:get-conversation-info', (data, callback) => {
   852→   852→      const conv = conversations.get(data.jid)
   853→   853→      callback?.(conv
   854→   854→        ? { success: true, conversation: serializeConversation(conv), messages: conv.messages.slice(-100) }
   855→   855→        : { success: false, conversation: null, messages: [] })
   856→   856→    })
   857→   857→
   858→   858→    // Send a text message
   859→   859→    socket.on('whatsapp:send-message', async (data, callback) => {
   860→   860→      try {
   861→   861→        if (!waSocket || connectionState.connection !== 'open') {
   862→   862→          return callback?.({ success: false, error: 'WhatsApp not connected' })
   863→   863→        }
   864→   864→
   865→   865→        const sent = await waSocket.sendMessage(data.jid, { text: data.text })
   866→   866→        const conv = getOrCreateConversation(data.jid)
   867→   867→        const msgKeyId = sent.key.id || ''
   868→   868→        const messageData = {
   869→   869→          id: msgKeyId || Math.random().toString(36).substr(2, 9),
   870→   870→          whatsappId: msgKeyId || null,
   871→   871→          fromMe: true,
   872→   872→          textContent: data.text,
   873→   873→          mediaType: null,
   874→   874→          timestamp: new Date(),
   875→   875→          status: 'sent',
   876→   876→        }
   877→   877→
   878→   878→        // Mark as processed to prevent duplicate from Baileys echo
   879→   879→        if (msgKeyId) processedMessageIds.add(msgKeyId)
   880→   880→
   881→   881→        conv.messages.push(messageData)
   882→   882→        if (conv.messages.length > MAX_MESSAGES_PER_CONVERSATION) {
   883→   883→          conv.messages = conv.messages.slice(-MAX_MESSAGES_PER_CONVERSATION)
   884→   884→        }
   885→   885→        conv.lastMessage = data.text
   886→   886→        conv.lastMessageAt = messageData.timestamp
   887→   887→
   888→   888→        if (waNamespace) {
   889→   889→          waNamespace.emit('whatsapp:message', {
   890→   890→            conversationJid: data.jid,
   891→   891→            message: messageData,
   892→   892→            conversation: serializeConversation(conv),
   893→   893→          })
   894→   894→          waNamespace.emit('whatsapp:conversation:update', serializeConversation(conv))
   895→   895→        }
   896→   896→
   897→   897→        callback?.({ success: true, messageId: sent.key.id })
   898→   898→      } catch (error) {
   899→   899→        console.error('[WA] Send message error:', error.message)
   900→   900→        callback?.({ success: false, error: error.message })
   901→   901→      }
   902→   902→    })
   903→   903→
   904→   904→    // Send media message
   905→   905→    socket.on('whatsapp:send-media', async (data, callback) => {
   906→   906→      try {
