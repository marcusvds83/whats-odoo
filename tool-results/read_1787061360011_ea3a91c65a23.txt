    58→interface UserRow {
    59→  id: string
    60→  email: string
    61→  name: string | null
    62→  role: string
    63→  isActive: boolean
    64→  whatsappPhone: string | null
    65→  odooUrl: string | null
    66→  odooDb: string | null
    67→  odooUsername: string | null
    68→  createdAt: string
    69→  updatedAt: string

    71→
    72→interface EditState {
    73→  id?: string
    74→  email: string
    75→  name: string
    76→  password: string
    77→  role: 'user' | 'admin'
    78→  isActive: boolean
    79→  whatsappPhone: string
    80→  odooUrl: string
    81→  odooDb: string
    82→  odooUsername: string
    83→  odooPassword: string
    84→}
    85→
    86→const EMPTY_EDIT: EditState = {
    87→  email: '',
    88→  name: '',
    89→  password: '',
    90→  role: 'user',
    91→  isActive: true,
    92→  whatsappPhone: '',
    93→  odooUrl: '',
    94→  odooDb: '',
    95→  odooUsername: '',
    96→  odooPassword: '',
    97→}
    98→
    99→export function UsersPanel() {
   100→  const [users, setUsers] = useState<UserRow[]>([])
   101→  const [isLoading, setIsLoading] = useState(true)
   102→  const [error, setError] = useState<string | null>(null)
   103→  const [dialogOpen, setDialogOpen] = useState(false)
   104→  const [editState, setEditState] = useState<EditState>(EMPTY_EDIT)
   105→  const [isSaving, setIsSaving] = useState(false)
   106→  // v7.24 (R6): pre-deploy backup state
   107→  const odoo = useOdoo()
   108→  const [isBackingUp, setIsBackingUp] = useState(false)
   109→  const [backupResult, setBackupResult] = useState<{
   110→    success: boolean
   111→    backed: number
   112→    total: number
   113→    failed: Array<{ userId: string; email: string; error: string }>
   114→    at: string
   115→  } | null>(null)
   116→  // v7.26: Test-login state
   117→  const [testLoginUser, setTestLoginUser] = useState<UserRow | null>(null)
   118→  const [testLoginPassword, setTestLoginPassword] = useState('')
   119→  const [testLoginResult, setTestLoginResult] = useState<null | {
   120→    found: boolean
   121→    isActive: boolean
   122→    role: string
   123→    passwordOk: boolean
   124→    hashPrefix: string
   125→    hashLength: number
   126→    providedHashPreview: string
   127→  }>(null)
   128→  const [isTestingLogin, setIsTestingLogin] = useState(false)
   129→  const [showPassword, setShowPassword] = useState(false)
   130→  // v7.29: "Alterar senha" dialog state
   131→  const [pwdUser, setPwdUser] = useState<UserRow | null>(null)
   132→  const [newPassword, setNewPassword] = useState('')
   133→  const [showNewPassword, setShowNewPassword] = useState(false)
   134→  const [isChangingPwd, setIsChangingPwd] = useState(false)
   135→  const [pwdCopied, setPwdCopied] = useState(false)
   136→
   137→  const loadUsers = useCallback(async () => {
   138→    setIsLoading(true)
   139→    setError(null)
   140→    try {
   141→      const res = await fetch('/api/users', { cache: 'no-store' })
   142→      const data = await res.json()
   143→      if (data.success) {
   144→        setUsers(data.users)
   145→      } else {
   146→        setError(data.error || 'Falha ao carregar usuários')
   147→      }
   148→    } catch (err: any) {
   149→      setError(err.message)
   150→    } finally {
   151→      setIsLoading(false)
   152→    }
   153→  }, [])
   154→
   155→  useEffect(() => {
   156→    loadUsers()
   157→  }, [loadUsers])
   158→
   159→  // v7.24 (R6): Backup all active user sessions to Odoo chatter.
   160→  // Should be triggered manually BEFORE any deploy.
   161→  const handleBackupToOdoo = useCallback(async () => {
   162→    if (isBackingUp) return
   163→    setIsBackingUp(true)
   164→    setBackupResult(null)
   165→    try {
   166→      const r = await odoo.backupAllToOdoo()
   167→      if (r.success) {
   168→        setBackupResult({
   169→          success: true,
   170→          backed: r.backed || 0,
   171→          total: r.total || 0,
   172→          failed: r.failed || [],
   173→          at: new Date().toISOString(),
   174→        })
   175→      } else {
   176→        setBackupResult({
   177→          success: false,
   178→          backed: r.backed || 0,
   179→          total: r.total || 0,
   180→          failed: r.failed || [{ userId: '-', email: '-', error: r.error || 'Falha desconhecida' }],
   181→          at: new Date().toISOString(),
   182→        })
   183→      }
   184→    } catch (err: any) {
   185→      setBackupResult({
   186→        success: false,
   187→        backed: 0,
   188→        total: 0,
   189→        failed: [{ userId: '-', email: '-', error: err.message }],
   190→        at: new Date().toISOString(),
   191→      })
   192→    } finally {
   193→      setIsBackingUp(false)
   194→    }
   195→  }, [isBackingUp, odoo])
   196→
   197→  const handleCreate = () => {
   198→    setEditState(EMPTY_EDIT)
   199→    setDialogOpen(true)
   200→  }
   201→
   202→  const handleEdit = (user: UserRow) => {
   203→    setEditState({
   204→      id: user.id,
   205→      email: user.email,
   206→      name: user.name || '',
   207→      password: '',
   208→      role: user.role as 'user' | 'admin',
   209→      isActive: user.isActive,
   210→      whatsappPhone: user.whatsappPhone || '',
   211→      odooUrl: user.odooUrl || '',
   212→      odooDb: user.odooDb || '',
   213→      odooUsername: user.odooUsername || '',
   214→      odooPassword: '',
   215→    })
   216→    setDialogOpen(true)
   217→  }
   218→
   219→  const handleSave = async () => {
   220→    setIsSaving(true)
   221→    setError(null)
   222→    try {
   223→      const isEditing = !!editState.id
   224→      const url = isEditing ? `/api/users/${editState.id}` : '/api/users'
   225→      const method = isEditing ? 'PATCH' : 'POST'
   226→
   227→      const body: any = {
   228→        email: editState.email,
   229→        name: editState.name,
   230→        role: editState.role,
   231→        isActive: editState.isActive,
   232→        whatsappPhone: editState.whatsappPhone,
   233→        odooUrl: editState.odooUrl,
   234→        odooDb: editState.odooDb,
   235→        odooUsername: editState.odooUsername,
   236→      }
   237→      if (editState.password) body.password = editState.password
   238→      if (editState.odooPassword) body.odooPassword = editState.odooPassword
   239→
   240→      const res = await fetch(url, {
   241→        method,
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify(body),
   244→      })
   245→      const data = await res.json()
   246→      if (data.success) {
   247→        setDialogOpen(false)
   248→        await loadUsers()
   249→      } else {
   250→        setError(data.error || 'Falha ao salvar')
   251→      }
   252→    } catch (err: any) {
   253→      setError(err.message)
   254→    } finally {
   255→      setIsSaving(false)
   256→    }
   257→  }
   258→
   259→  // v7.26: Generate a random strong password and put it into the edit form
   260→  const generatePassword = () => {
   261→    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789!@#$%&*'
   262→    let pwd = ''
   263→    const arr = new Uint32Array(12)
   264→    crypto.getRandomValues(arr)
   265→    for (let i = 0; i < 12; i++) pwd += chars[arr[i] % chars.length]
   266→    setEditState(s => ({ ...s, password: pwd }))
   267→    setShowPassword(true)
   268→  }
   269→
   270→  // v7.26: Test a user's password without impersonating them
   271→  const handleTestLogin = async () => {
   272→    if (!testLoginUser || !testLoginPassword) return
   273→    setIsTestingLogin(true)
   274→    setTestLoginResult(null)
   275→    try {
   276→      const res = await fetch(`/api/users/${testLoginUser.id}/test-login`, {
   277→        method: 'POST',
   278→        headers: { 'Content-Type': 'application/json' },
   279→        body: JSON.stringify({ password: testLoginPassword }),
   280→      })
   281→      const data = await res.json()
   282→      if (data.success) {
   283→        setTestLoginResult({
   284→          found: data.found,
   285→          isActive: data.isActive,
   286→          role: data.role,
   287→          passwordOk: data.passwordOk,
   288→          hashPrefix: data.hashPrefix,
   289→          hashLength: data.hashLength,
   290→          providedHashPreview: data.providedHashPreview,
   291→        })
   292→      } else {
   293→        setTestLoginResult({
   294→          found: false,
   295→          isActive: false,
   296→          role: '?',
   297→          passwordOk: false,
   298→          hashPrefix: 'error: ' + (data.error || 'unknown'),
   299→          hashLength: 0,
   300→          providedHashPreview: '',
   301→        })
   302→      }
   303→    } catch (err: any) {
   304→      setTestLoginResult({
   305→        found: false,
   306→        isActive: false,
   307→        role: '?',
   308→        passwordOk: false,
   309→        hashPrefix: 'fetch error: ' + err.message,
   310→        hashLength: 0,
   311→        providedHashPreview: '',
   312→      })
   313→    } finally {
   314→      setIsTestingLogin(false)
   315→    }
   316→  }
   317→
   318→  const handleDelete = async (user: UserRow) => {
   319→    if (!confirm(`Excluir o usuário "${user.email}"? Esta ação não pode ser desfeita.`)) return
   320→    try {
   321→      const res = await fetch(`/api/users/${user.id}`, { method: 'DELETE' })
   322→      const data = await res.json()
   323→      if (data.success) {
   324→        await loadUsers()
   325→      } else {
   326→        setError(data.error || 'Falha ao excluir')
   327→      }
   328→    } catch (err: any) {
   329→      setError(err.message)
   330→    }
   331→  }
   332→
   333→  const activeCount = users.filter(u => u.isActive).length
   334→  const adminCount = users.filter(u => u.role === 'admin').length
   335→
   336→  // v7.29: Open the "Alterar senha" dialog for a user
   337→  const openChangePassword = (user: UserRow) => {
   338→    setPwdUser(user)
   339→    setNewPassword('')
   340→    setShowNewPassword(false)
   341→    setPwdCopied(false)
   342→  }
   343→
   344→  // v7.29: Save a new password for the user (hashed on the server)
   345→  const handleChangePassword = async () => {
   346→    if (!pwdUser || !newPassword.trim()) return
   347→    setIsChangingPwd(true)
   348→    setError(null)
   349→    try {
   350→      const res = await fetch(`/api/users/${pwdUser.id}`, {
   351→        method: 'PATCH',
   352→        headers: { 'Content-Type': 'application/json' },
   353→        body: JSON.stringify({ password: newPassword.trim() }),
   354→      })
   355→      const data = await res.json()
   356→      if (data.success) {
   357→        setPwdUser(null)
   358→        await loadUsers()
   359→      } else {
   360→        setError(data.error || 'Falha ao alterar a senha')
   361→      }
   362→    } catch (err: any) {
   363→      setError(err.message)
   364→    } finally {
   365→      setIsChangingPwd(false)
   366→    }
   367→  }
   368→
   369→  const copyNewPassword = async () => {
   370→    try {
   371→      await navigator.clipboard?.writeText(newPassword)
   372→      setPwdCopied(true)
   373→      setTimeout(() => setPwdCopied(false), 2000)
   374→    } catch {}
   375→  }
   376→
   377→  return (
   378→    <div className="space-y-4">
   379→      <div className="flex items-center justify-between gap-3">
   380→        <div>
   381→          <h2 className="text-xl font-bold flex items-center gap-2">
   382→            <UserIcon className="size-5" />
   383→            Usuários do Middleware
   384→          </h2>
   385→          <p className="text-sm text-muted-foreground mt-1">
   386→            {users.length} total • {activeCount} ativos • {adminCount} admin
   387→          </p>
   388→        </div>
   389→        <div className="flex gap-2">
   390→          <Button variant="outline" size="sm" onClick={loadUsers} disabled={isLoading}>
   391→            <RefreshCw className={`size-4 mr-1.5 ${isLoading ? 'animate-spin' : ''}`} />
   392→            Atualizar
   393→          </Button>
   394→          {/* v7.24 (R6): Pre-deploy backup button */}
   395→          <Button
   396→            variant="outline"
   397→            size="sm"
   398→            onClick={handleBackupToOdoo}
   399→            disabled={isBackingUp || !odoo.isConnected}
   400→            title={odoo.isConnected ? 'Salva credenciais e conversas de todos os usuários no chatter do Odoo (use antes de deploy)' : 'Conecte ao Odoo primeiro'}
   401→          >
   402→            {isBackingUp ? <Loader2 className="size-4 mr-1.5 animate-spin" /> : <Database className="size-4 mr-1.5" />}
   403→            {isBackingUp ? 'Backup em andamento…' : 'Backup no Odoo'}
   404→          </Button>
   405→          <Button size="sm" onClick={handleCreate} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   406→            <UserPlus className="size-4 mr-1.5" />
   407→            Novo Usuário
   408→          </Button>
   409→        </div>
   410→      </div>
   411→
   412→      {/* v7.24 (R6): Backup result banner */}
   413→      {backupResult && (
   414→        <Alert variant={backupResult.success ? 'default' : 'destructive'}>
   415→          {backupResult.success
   416→            ? <CheckCircle2 className="size-4" />
   417→            : <AlertCircle className="size-4" />}
   418→          <AlertDescription>
   419→            {backupResult.success
   420→              ? `Backup concluído: ${backupResult.backed}/${backupResult.total} usuário(s) salvos no chatter do Odoo.`
   421→              : `Backup parcial: ${backupResult.backed}/${backupResult.total} ok, ${backupResult.failed.length} falha(s).`}
   422→            {backupResult.failed.length > 0 && (
   423→              <ul className="mt-1 ml-4 text-xs list-disc">
   424→                {backupResult.failed.map((f, i) => (
   425→                  <li key={i}>{f.email}: {f.error}</li>
   426→                ))}
   427→              </ul>
   428→            )}
   429→            <span className="block mt-1 text-xs text-muted-foreground">
   430→              Realizado em {new Date(backupResult.at).toLocaleString('pt-BR')}
   431→            </span>
   432→          </AlertDescription>
   433→        </Alert>
   434→      )}
   435→
   436→      {error && (
   437→        <Alert variant="destructive">
   438→          <AlertDescription>{error}</AlertDescription>
   439→        </Alert>
   440→      )}
   441→
   442→      <Card>
   443→        <CardContent className="p-0">
   444→          {isLoading ? (
   445→            <div className="flex items-center justify-center py-12">
   446→              <Loader2 className="size-6 animate-spin text-muted-foreground" />
   447→            </div>
   448→          ) : users.length === 0 ? (
   449→            <div className="text-center py-12">
   450→              <UserIcon className="size-10 mx-auto text-muted-foreground/50 mb-2" />
   451→              <p className="text-sm text-muted-foreground">Nenhum usuário cadastrado</p>
   452→            </div>
   453→          ) : (
   454→            <Table>
   455→              <TableHeader>
   456→                <TableRow>
   457→                  <TableHead>Usuário</TableHead>
   458→                  <TableHead>Telefone WA</TableHead>
   459→                  <TableHead>Odoo</TableHead>
   460→                  <TableHead>Perfil</TableHead>
   461→                  <TableHead>Status</TableHead>
   462→                  <TableHead className="text-right">Ações</TableHead>
   463→                </TableRow>
   464→              </TableHeader>
   465→              <TableBody>
   466→                {users.map((user) => (
   467→                  <TableRow key={user.id}>
   468→                    <TableCell>
   469→                      <div className="flex flex-col">
   470→                        <span className="font-medium text-sm flex items-center gap-1.5">
   471→                          {user.name || user.email}
   472→                          {user.role === 'admin' && (
   473→                            <Shield className="size-3 text-amber-600" />
   474→                          )}
   475→                        </span>
   476→                        <span className="text-xs text-muted-foreground flex items-center gap-1">
   477→                          <Mail className="size-3" />
   478→                          {user.email}
   479→                        </span>
   480→                      </div>
   481→                    </TableCell>
   482→                    <TableCell>
   483→                      {user.whatsappPhone ? (
   484→                        <span className="text-sm flex items-center gap-1">
   485→                          <Phone className="size-3 text-muted-foreground" />
   486→                          {user.whatsappPhone}
   487→                        </span>
   488→                      ) : (
   489→                        <span className="text-xs text-muted-foreground">—</span>
   490→                      )}
   491→                    </TableCell>
   492→                    <TableCell>
   493→                      {user.odooUrl ? (
   494→                        <div className="flex flex-col text-xs">
   495→                          <span className="font-medium truncate max-w-[160px]">{user.odooUsername || '?'}</span>
   496→                          <span className="text-muted-foreground truncate max-w-[160px]">
   497→                            {user.odooDb}@{(() => {
   498→                              try { return new URL(user.odooUrl!).hostname } catch { return user.odooUrl }
   499→                            })()}
   500→                          </span>
   501→                        </div>
   502→                      ) : (
   503→                        <span className="text-xs text-muted-foreground">Usa config global</span>
   504→                      )}
   505→                    </TableCell>
   506→                    <TableCell>
   507→                      {user.role === 'admin' ? (
   508→                        <Badge className="bg-amber-100 text-amber-800 border-amber-200">Admin</Badge>
   509→                      ) : (
   510→                        <Badge variant="outline">Usuário</Badge>
   511→                      )}
   512→                    </TableCell>
   513→                    <TableCell>
   514→                      {user.isActive ? (
   515→                        <Badge className="bg-emerald-100 text-emerald-800 border-emerald-200">Ativo</Badge>
   516→                      ) : (
   517→                        <Badge variant="secondary">Inativo</Badge>
   518→                      )}
   519→                    </TableCell>
   520→                    <TableCell className="text-right">
   521→                      <div className="flex items-center justify-end gap-1">
   522→                        <Button
   523→                          variant="ghost"
   524→                          size="icon"
   525→                          className="size-8"
   526→                          onClick={() => openChangePassword(user)}
   527→                          title="Alterar senha"
   528→                        >
   529→                          <Key className="size-4" />
   530→                        </Button>
   531→                        <Button
   532→                          variant="ghost"
   533→                          size="icon"
   534→                          className="size-8"
   535→                          onClick={() => {
   536→                            setTestLoginUser(user)
   537→                            setTestLoginPassword('')
   538→                            setTestLoginResult(null)
   539→                          }}
   540→                          title="Testar login"
   541→                        >
   542→                          <FlaskConical className="size-4" />
   543→                        </Button>
   544→                        <Button
   545→                          variant="ghost"
   546→                          size="icon"
   547→                          className="size-8"
   548→                          onClick={() => handleEdit(user)}
   549→                          title="Editar"
   550→                        >
   551→                          <Pencil className="size-4" />
   552→                        </Button>
   553→                        <Button
   554→                          variant="ghost"
   555→                          size="icon"
   556→                          className="size-8 hover:bg-destructive/10 hover:text-destructive"
   557→                          onClick={() => handleDelete(user)}
   558→                          title="Excluir"
   559→                        >
   560→                          <Trash2 className="size-4" />
   561→                        </Button>
   562→                      </div>
   563→                    </TableCell>
   564→                  </TableRow>
   565→                ))}
   566→              </TableBody>
   567→            </Table>
   568→          )}
   569→        </CardContent>
   570→      </Card>
   571→
   572→      {/* v7.29: Alterar senha dialog */}
   573→      <Dialog open={!!pwdUser} onOpenChange={(o) => { if (!o) setPwdUser(null) }}>
   574→        <DialogContent className="max-w-md">
   575→          <DialogHeader>
   576→            <DialogTitle className="flex items-center gap-2">
   577→              <Key className="size-5 text-emerald-600" />
   578→              Alterar senha
   579→            </DialogTitle>
   580→            <DialogDescription>
   581→              {pwdUser && (
   582→                <>Redefinir a senha de <strong>{pwdUser.email}</strong>. A nova senha fica salva no Firebase (hasheada) e passa a valer no próximo login.</>
   583→              )}
   584→            </DialogDescription>
   585→          </DialogHeader>
   586→
   587→          <div className="space-y-3 py-2">
   588→            <div className="space-y-1.5">
   589→              <Label htmlFor="new-pwd" className="flex items-center gap-1">
   590→                Nova senha *
   591→                <Button
   592→                  type="button"
   593→                  variant="ghost"
