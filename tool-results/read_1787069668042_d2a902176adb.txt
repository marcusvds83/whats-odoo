    38→    38→// Baileys can deliver JIDs in several forms:
    39→    39→//   5511999888777@s.whatsapp.net            — canonical DM
    40→    40→//   5511999888777:7@s.whatsapp.net          — DM with device suffix (multi-device send)
    41→    41→//   5511999888777:7:3@s.whatsapp.net        — DM with device + agent suffix
    42→    42→//   5511999888777@lid                        — Linked Identity (newer protocol)
    43→    43→// We normalize everything to canonical form: <digits>@s.whatsapp.net
    44→    44→function normalizeJid(jid) {
    45→    45→  if (!jid || typeof jid !== 'string') return null
    46→    46→  const atIdx = jid.indexOf('@')
    47→    47→  if (atIdx < 0) return null
    48→    48→  const server = jid.slice(atIdx + 1)
    49→    49→  if (server !== 's.whatsapp.net') return null
    50→    50→  const userCombined = jid.slice(0, atIdx)
    51→    51→  const user = userCombined.split(':')[0].split('_')[0]
    52→    52→  if (!/^\d{7,}$/.test(user)) return null
    53→    53→  return `${user}@s.whatsapp.net`
    54→    54→}

    78→    78→function escapeHtml(text) {
    79→    79→  return String(text || '')
    80→    80→    .replace(/&/g, '&amp;')
    81→    81→    .replace(/</g, '&lt;')
    82→    82→    .replace(/>/g, '&gt;')
    83→    83→    .replace(/\n/g, '<br/>')
    84→    84→}

    86→    86→function stripHtml(html) {
    87→    87→  if (!html) return ''
    88→    88→  let text = String(html)
    89→    89→    .replace(/<br\s*\/?>/gi, '\n')
    90→    90→    .replace(/<\/p>/gi, '\n')
    91→    91→    .replace(/<p[^>]*>/gi, '')
    92→    92→    .replace(/<[^>]+>/g, '')
    93→    93→    .replace(/&nbsp;/g, ' ')
    94→    94→    .replace(/&amp;/g, '&')
    95→    95→    .replace(/&lt;/g, '<')
    96→    96→    .replace(/&gt;/g, '>')
    97→    97→    .replace(/&quot;/g, '"')
    98→    98→    .replace(/&#39;/g, "'")
    99→    99→    .trim()
   100→   100→  text = text.replace(/\n{3,}/g, '\n\n')
   101→   101→  return text
   102→   102→}

   104→   104→// v7.28: Parse a chatter body produced by buildChatterBody() back into clean
   105→   105→// message components (direction, timestamp, text). The chatter body format is:
   106→   106→//   <div><strong>📱 WhatsApp {Enviada|Recebida}:</strong> {ts}<br/>{mediaLine}<span>{text}</span></div>
   107→   107→// We want to extract ONLY the user-visible text (the <span> content), detect
   108→   108→// the direction from "Enviada" vs "Recebida", and parse the timestamp.
   109→   109→// Falls back to stripHtml() for any non-conforming body (e.g. messages posted
   110→   110→// by other Odoo users from the web UI — those are returned as-is, plain text).
   111→   111→function parseChatterBody(body) {
   112→   112→  if (!body) return { fromMe: false, timestamp: null, text: '' }
   113→   113→  const s = String(body)
   114→   114→
   115→   115→  // Detect direction from the "WhatsApp Enviada/Recebida:" marker
   116→   116→  const isSent = /WhatsApp\s+Enviada/i.test(s)
   117→   117→  const fromMe = isSent
   118→   118→
   119→   119→  // Extract timestamp from "(...)" — buildChatterBody uses toLocaleString('pt-BR')
   120→   120→  // inside parentheses as part of the label. Actually buildChatterBody does NOT
   121→   121→  // wrap in parens — it inserts the timestamp directly after the colon.
   122→   122→  // But older messages may have the "(date)" format. Support both.
   123→   123→  let timestamp = null
   124→   124→  const parenMatch = s.match(/\(([^)]+)\)/)
   125→   125→  if (parenMatch) {
   126→   126→    const parsed = new Date(parenMatch[1])
   127→   127→    if (!isNaN(parsed.getTime())) timestamp = parsed
   128→   128→  }
   129→   129→
   130→   130→  // Extract the text content from the <span>...</span> block (the real message)
   131→   131→  let text = ''
   132→   132→  const spanMatch = s.match(/<span[^>]*>([\s\S]*?)<\/span>/i)
   133→   133→  if (spanMatch) {
   134→   134→    text = stripHtml(spanMatch[1])
   135→   135→  } else {
   136→   136→    // No <span> — this isn't a WhatsApp chatter message produced by us.
   137→   137→    // It's probably a note posted by a human via the Odoo UI, or a different
   138→   138→    // format entirely. Strip HTML and return the whole body as plain text.
   139→   139→    text = stripHtml(s)
   140→   140→    // If the body still starts with the WhatsApp metadata header, strip it.
   141→   141→    // This handles the case where buildChatterBody emitted no <span> because
   142→   142→    // textContent was empty (media-only message).
   143→   143→    const headerMatch = text.match(/^📱\s*WhatsApp\s+(Enviada|Recebida)[:\s]*(.*)$/is)
   144→   144→    if (headerMatch) {
   145→   145→      // For media-only messages, the "text" will be empty after the header.
   146→   146→      // Strip the header line and keep the remaining content (media label).
   147→   147→      const remaining = text.slice(headerMatch[0].length).trim()
   148→   148→      // Try to detect a media label line like "🎙️ Áudio"
   149→   149→      const mediaMatch = remaining.match(/^([🎙️🖼️📄🎬🏷️][^\n]*)/)
   150→   150→      text = mediaMatch ? mediaMatch[1] : remaining
   151→   151→    }
   152→   152→  }
   153→   153→
   154→   154→  // Try to parse timestamp from the label after the colon (no parens case)
   155→   155→  if (!timestamp) {
   156→   156→    // Look for a date pattern like "14/08/2026, 10:30:45" or "14/08/2026 10:30"
   157→   157→    const dateMatch = s.match(/(\d{1,2}\/\d{1,2}\/\d{4})[,\s]+(\d{1,2}:\d{2}(?::\d{2})?)/)
   158→   158→    if (dateMatch) {
   159→   159→      const [_, d, t] = dateMatch
   160→   160→      // pt-BR format: DD/MM/YYYY HH:MM:SS
   161→   161→      const parsed = new Date(`${d.split('/').reverse().join('-')}T${t}`)
   162→   162→      if (!isNaN(parsed.getTime())) timestamp = parsed
   163→   163→    }
   164→   164→  }
   165→   165→
   166→   166→  return { fromMe, timestamp, text }
   167→   167→}

   174→   174→// v7.19: Build a plain-text transcript of the WhatsApp conversation that will
   175→   175→// be written to the `description` field (Notes) of the CRM lead/opportunity.
   176→   176→function buildConversationTranscript(messages) {
   177→   177→  if (!Array.isArray(messages) || messages.length === 0) return ''
   178→   178→  const sorted = [...messages].sort(
   179→   179→    (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   180→   180→  )
   181→   181→  const lines = sorted.map((m) => {
   182→   182→    const direction = m.fromMe ? 'Eu' : 'Contato'
   183→   183→    const dateLabel = new Date(m.timestamp).toLocaleString('pt-BR')
   184→   184→    const mediaLabel = m.mediaType ? ` [${m.mediaType}]` : ''
   185→   185→    const text = m.textContent && m.textContent.trim() ? m.textContent : '[Mídia]'
   186→   186→    return `[${dateLabel}] ${direction}${mediaLabel}: ${text}`
   187→   187→  })
   188→   188→  return `Conversa WhatsApp:\n${lines.join('\n')}`
   189→   189→}

   396→   396→  // v7.24 (R6): Pre-deploy backup — dump this user's WA creds + conversation
   397→   397→  // state to Odoo chatter so the data survives even if the disk is wiped
   398→   398→  // during a Render deploy. Posted as a mail.message on a designated
   399→   399→  // res.partner (the user's own partner if findable, otherwise partner 1).
   400→   400→  //
   401→   401→  // Format: subject `[BACKUP-CREDS] User <email> — <timestamp>`, body is
   402→   402→  // JSON-stringified { creds, conversations }. If the JSON is too large
   403→   403→  // (>600 KB per chunk — Odoo's chatter body limit is ~1MB), it's split
   404→   404→  // into N chunks posted as separate messages with subject suffix `[1/N]`.
   405→   405→  //
   406→   406→  // Returns: { success, chunksPosted, partnerId, error? }
   407→   407→  async backupToOdoo() {
   408→   408→    if (!this.odooConfig.uid) {
   409→   409→      return { success: false, chunksPosted: 0, error: 'Odoo not authenticated' }
   410→   410→    }
   411→   411→    try {
   412→   412→      // Persist current state to disk first (so the backup reflects the
   413→   413→      // latest conversations, not the last periodic save).
   414→   414→      this.persistConversationsToDisk()
