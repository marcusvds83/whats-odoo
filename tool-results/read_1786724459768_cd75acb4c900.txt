   393→function NavItem({ icon, label, active, onClick, badge, collapsed }: {
   394→  icon: React.ReactNode
   395→  label: string
   396→  active: boolean
   397→  onClick: () => void
   398→  badge?: string
   399→  collapsed?: boolean
   400→}) {
   401→  return (
   402→    <button
   403→      onClick={onClick}
   404→      className={cn(
   405→        'w-full flex items-center gap-3 px-2 py-2 rounded-lg text-sm transition-colors',
   406→        'hover:bg-muted/80 focus-visible:outline-none focus-visible:bg-muted/80',
   407→        active ? 'bg-primary/10 text-primary font-medium' : 'text-muted-foreground',
   408→      )}
   409→      title={collapsed ? label : undefined}
   410→    >
   411→      <span className="shrink-0 mx-auto lg:mx-0">{icon}</span>
   412→      {!collapsed && <span className="hidden lg:block flex-1 text-left truncate">{label}</span>}
   413→      {badge && !collapsed && (
   414→        <Badge className="size-5 p-0 flex items-center justify-center text-[10px] bg-emerald-500 text-white border-0">
   415→          {badge === 'online' ? '\u25CF' : badge}
   416→        </Badge>
   417→      )}
   418→      {badge && collapsed && (
   419→        <Badge className="hidden lg:flex absolute top-0 right-0 size-2 p-0 bg-emerald-500 border-0" />
   420→      )}
   421→    </button>
   422→  )
   423→}
