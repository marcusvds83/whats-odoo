   152→function getOrCreateConversation(jid, pushName) {
   153→  if (!conversations.has(jid)) {
   154→    conversations.set(jid, {
   155→      jid,
   156→      name: null,
   157→      phone: extractPhone(jid),
   158→      pushName: pushName || null,
   159→      avatarUrl: null,
   160→      lastMessage: null,
   161→      lastMessageAt: null,
   162→      unreadCount: 0,

   166→  const conv = conversations.get(jid)
   167→  if (pushName && !conv.pushName) {
   168→    conv.pushName = pushName
   169→  }

   191→function makeXmlRpcClient(path) {
   192→  const url = new URL(odooConfig.url)
   193→  const isHttps = url.protocol === 'https:'
   194→  const options = {
   195→    host: url.hostname,
   196→    port: parseInt(url.port) || (isHttps ? 443 : 80),
   197→    path,
   198→  }
   199→  return isHttps ? xmlrpc.createSecureClient(options) : xmlrpc.createClient(options)
   200→}
   201→
   202→function odooAuthenticate() {
   203→  return new Promise((resolve, reject) => {
   204→    const client = makeXmlRpcClient('/xmlrpc/2/common')
   205→    client.methodCall('authenticate', [
   206→      odooConfig.db, odooConfig.username, odooConfig.password, {},
   207→    ], (error, value) => {
   208→      if (error) return reject(error)
   209→      if (!value) return reject(new Error('Authentication failed - invalid credentials'))
   210→      resolve(value)
   211→    })
   212→  })
   213→}
   214→
   215→function odooExecuteKw(model, method, args, kwargs = {}) {
   216→  return new Promise((resolve, reject) => {
   217→    if (!odooConfig.uid) {
   218→      return reject(new Error('Not authenticated with Odoo'))
   219→    }
   220→    const client = makeXmlRpcClient('/xmlrpc/2/object')
   221→    client.methodCall('execute_kw', [
   222→      odooConfig.db, odooConfig.uid, odooConfig.password, model, method, args, kwargs,
   223→    ], (error, value) => {
   224→      if (error) reject(error)
   225→      else resolve(value)
   226→    })
   227→  })
   228→}

   230→// Smart Field Detection
   231→async function getAvailableFields(model) {
   232→  if (modelFieldsCache.has(model)) return modelFieldsCache.get(model)
   233→  try {
   234→    const fields = await odooExecuteKw(model, 'fields_get', [], { attributes: ['string', 'type'] })
   235→    const fieldNames = new Set(Object.keys(fields))
   236→    modelFieldsCache.set(model, fieldNames)
   237→    return fieldNames
   238→  } catch (error) {
   239→    console.error(`[Odoo] Failed to get fields for ${model}:`, error.message)
   240→    return new Set()
   241→  }
   242→}
   243→
   244→async function filterExistingFields(model, requestedFields) {
   245→  const available = await getAvailableFields(model)
   246→  const existing = requestedFields.filter(f => available.has(f))
   247→  if (!existing.includes('id') && available.has('id')) existing.unshift('id')
   248→  if (!existing.includes('name') && available.has('name')) existing.push('name')
   249→  return existing
   250→}

   252→async function buildSafeValues(model, values) {
   253→  const available = await getAvailableFields(model)
   254→  const safe = {}
   255→  for (const [key, value] of Object.entries(values)) {
   256→    if (available.has(key)) {
   257→      safe[key] = value
   258→    } else {
   259→      console.log(`[Odoo] Field "${key}" does not exist on ${model}, skipping`)
   260→    }
   261→  }
   262→  return safe
   263→}

   265→async function smartWriteWhatsAppNumber(model, ids, phone) {
   266→  const available = await getAvailableFields(model)
   267→  const values = {}
   268→  if (available.has('whatsapp')) values.whatsapp = phone
   269→  if (available.has('whatsapp_number')) values.whatsapp_number = phone
   270→  if (available.has('phone')) values.phone = phone
   271→  if (available.has('mobile') && !values.whatsapp) values.mobile = phone
   272→  if (Object.keys(values).length === 0) return false
   273→  return odooWrite(model, ids, values)
   274→}

   297→async function odooWrite(model, ids, values) {

   302→async function odooSearchOrCreate(model, domain, values) {
   303→  const existing = await odooExecuteKw(model, 'search', [domain], { limit: 1 })
   304→  if (existing && existing.length > 0) {
   305→    const safeValues = await buildSafeValues(model, values)
   306→    await odooWrite(model, existing, safeValues)
   307→    return { id: existing[0], created: false }
   308→  }
   309→  const safeValues = await buildSafeValues(model, values)
   310→  const newId = await odooCreate(model, safeValues)
   311→  return { id: newId, created: true }
   312→}

   366→async function autoSyncWhatsAppMessage(data) {
   367→  const result = {
   368→    partnerId: null,
   369→    leadId: null,
   370→    mailMessageId: null,
   371→    activityId: null,
   372→    created: { partner: false, lead: false },
   373→    errors: [],
   374→  }
   375→
   376→  if (!autoSyncSettings.enabled || !odooConfig.uid) return result
   377→
   378→  console.log(`[AutoSync] Processing message from ${data.phone} (${data.pushName || 'unknown'})`)
   379→
   380→  try {
   381→    // Step 1: Create or update contact
   382→    if (autoSyncSettings.autoCreateContact) {
   383→      const contactName = data.pushName || `WhatsApp ${data.phone}`
   384→      const domain = ['|', ['phone', 'ilike', data.phone], ['mobile', 'ilike', data.phone]]
   385→      const contactValues = { name: contactName, phone: data.phone, mobile: data.phone }
   386→
   387→      const partnerFields = await getAvailableFields('res.partner')
   388→      if (partnerFields.has('whatsapp')) contactValues.whatsapp = data.phone
   389→      if (partnerFields.has('whatsapp_number')) contactValues.whatsapp_number = data.phone
   390→
   391→      const partnerResult = await odooSearchOrCreate('res.partner', domain, contactValues)
   392→      result.partnerId = partnerResult.id
   393→      result.created.partner = partnerResult.created
   394→
   395→      const cached = phoneToPartnerCache.get(data.phone)
   396→      if (cached) {
   397→        cached.partnerId = partnerResult.id
   398→      } else {
   399→        phoneToPartnerCache.set(data.phone, { partnerId: partnerResult.id, leadId: null, leadCreated: false })
   400→      }
   401→      console.log(`[AutoSync] Contact ${partnerResult.created ? 'created' : 'updated'}: res.partner#${partnerResult.id}`)
   402→    }
   403→
   404→    // Step 2: Create lead for new conversations
   405→    if (autoSyncSettings.autoCreateLead && !data.fromMe && result.partnerId) {
   406→      const cached = phoneToPartnerCache.get(data.phone)
   407→      if (cached && cached.leadId && cached.leadCreated) {
   408→        result.leadId = cached.leadId
   409→      } else {
   410→        const existingLeads = await odooSearch('crm.lead', [
   411→          ['partner_id', '=', result.partnerId],
   412→          ['name', 'like', autoSyncSettings.leadPrefix],
   413→          ['type', '=', 'lead'],
   414→        ], ['id', 'name'], 1)
   415→
   416→        if (existingLeads && existingLeads.length > 0) {
   417→          result.leadId = existingLeads[0].id
   418→          if (cached) { cached.leadId = existingLeads[0].id; cached.leadCreated = true }
   419→          console.log(`[AutoSync] Found existing lead: crm.lead#${existingLeads[0].id}`)
   420→        } else {
   421→          const leadName = `${autoSyncSettings.leadPrefix}${data.pushName || data.phone}`
   422→          const leadValues = {
   423→            name: leadName,
   424→            type: 'lead',
   425→            partner_id: result.partnerId,
   426→            phone: data.phone,
   427→            description: `Conversa iniciada via WhatsApp em ${new Date().toLocaleString('pt-BR')}`,
   428→          }
   429→
   430→          const leadFields = await getAvailableFields('crm.lead')
   431→          if (leadFields.has('whatsapp_number')) leadValues.whatsapp_number = data.phone
   432→
   433→          if (autoSyncSettings.leadTeamId) leadValues.team_id = autoSyncSettings.leadTeamId
   434→          if (autoSyncSettings.leadUserId) leadValues.user_id = autoSyncSettings.leadUserId
   435→
   436→          result.leadId = await odooCreate('crm.lead', leadValues)
   437→          result.created.lead = true
   438→
   439→          if (cached) {
   440→            cached.leadId = result.leadId
   441→            cached.leadCreated = true
   442→          } else {
   443→            phoneToPartnerCache.set(data.phone, { partnerId: result.partnerId, leadId: result.leadId, leadCreated: true })
   444→          }
   445→          console.log(`[AutoSync] Lead created: crm.lead#${result.leadId}`)
   446→        }
   447→      }
   448→    }
   449→
   450→    // Step 3: Post message in Odoo chatter
   451→    if (autoSyncSettings.autoPostMessages && !data.fromMe) {
   452→      const targetModel = result.leadId ? 'crm.lead' : (result.partnerId ? 'res.partner' : null)
   453→      const targetId = result.leadId || result.partnerId
   454→      if (targetModel && targetId) {
   455→        const direction = data.fromMe ? 'Enviada' : 'Recebida'
   456→        const mediaLabel = data.mediaType ? ` [${data.mediaType}]` : ''
   457→        const msgBody = data.textContent
   458→          ? `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel}</p><p>${escapeHtml(data.textContent)}</p>`
   459→          : `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel} [Mídia]</p>`
   460→        try {
   461→          result.mailMessageId = await odooPostMessage(targetModel, targetId, msgBody)
   462→          console.log(`[AutoSync] Message posted on ${targetModel}#${targetId}`)
   463→        } catch (error) {
   464→          result.errors.push(`Failed to post message: ${error.message}`)
   465→        }
   466→      }
   467→    }
   468→
   469→    // Step 4: Create activity notification for first message of a new lead
   470→    if (autoSyncSettings.autoCreateActivity && result.created.lead && result.leadId) {
   471→      const summary = 'Nova mensagem WhatsApp'
   472→      const note = `Contato ${data.pushName || data.phone} iniciou uma conversa via WhatsApp.\n\nMensagem: ${data.textContent || '[Mídia]'}`
   473→      try {
   474→        result.activityId = await odooCreateActivity('crm.lead', result.leadId, summary, note)
   475→      } catch (error) {
   476→        result.errors.push(`Failed to create activity: ${error.message}`)
   477→      }
   478→    }
   479→  } catch (error) {
   480→    result.errors.push(`Auto-sync error: ${error.message}`)
   481→    console.error('[AutoSync] Error:', error.message)
   482→  }
   483→
   484→  // Emit sync result to all connected clients
   485→  if (odooNamespace) {
   486→    odooNamespace.emit('odoo:autosync:result', { phone: data.phone, ...result })
   487→  }
   488→
   489→  return result
   490→}

   499→async function connectWhatsApp() {
   500→  // Guard: ensure Baileys modules are loaded
   501→  if (typeof makeWASocket !== 'function' || typeof useMultiFileAuthState !== 'function') {
   502→    console.error('[WA] Cannot connect: Baileys modules not loaded. WhatsApp will not work.')
   503→    if (waNamespace) waNamespace.emit('whatsapp:status', { connected: false, reason: 'modules_not_loaded' })
   504→    return
   505→  }
   506→
   507→  try {
   508→  const { state, saveCreds } = await useMultiFileAuthState(AUTH_FOLDER)
   509→  const { version } = await fetchLatestBaileysVersion()
   510→
   511→  waSocket = makeWASocket({
   512→    version,
   513→    logger: waLogger,
   514→    auth: {
   515→      creds: state.creds,
   516→      keys: makeCacheableSignalKeyStore(state.keys, waLogger),
   517→    },
   518→    browser: Browsers.ubuntu('Chrome'),
   519→    printQRInTerminal: false,
   520→    connectTimeoutMs: 60_000,
   521→    defaultQueryTimeoutMs: 30_000,
   522→    keepAliveIntervalMs: 30_000, // increased from 25s for better stability
   523→    markOnlineOnConnect: true,
   524→    syncFullHistory: true, // Pull conversations from phone so user can create leads
   525→    // Memory optimization: limit message history sync
   526→    messageCaching: false,
   527→  })
   528→
   529→  // Save credentials on update
   530→  waSocket.ev.on('creds.update', saveCreds)
   531→
   532→  // Connection events
   533→  waSocket.ev.on('connection.update', async (update) => {
   534→    const { connection, lastDisconnect, qr } = update
   535→    connectionState = { connection, lastDisconnect }
   536→
   537→    console.log(`[WA] Connection update: ${connection}`)
   538→
   539→    if (qr) {
   540→      console.log('[WA] QR Code generated, sending to clients')
   541→      if (waNamespace) waNamespace.emit('whatsapp:qr', { qr })
   542→    }
   543→
   544→    if (connection === 'close') {
   545→      const statusCode = lastDisconnect?.error?.output?.statusCode
   546→      const shouldReconnect = statusCode !== DisconnectReason.loggedOut
   547→
   548→      console.log(`[WA] Connection closed. Status: ${statusCode}, Reconnect: ${shouldReconnect}`)
   549→
   550→      if (waNamespace) waNamespace.emit('whatsapp:status', {
   551→        connected: false,
   552→        reason: statusCode === DisconnectReason.loggedOut ? 'logged_out' : 'disconnected',
   553→      })
   554→
   555→      if (shouldReconnect) {
   556→        reconnectAttempts++
   557→        // Exponential backoff: 2s, 4s, 8s, 16s, 30s, 30s, 30s...
   558→        const delay = Math.min(2000 * Math.pow(2, Math.min(reconnectAttempts - 1, 4)), 30000)
   559→        // Infinite reconnects (no max) — Render will restart the dyno if truly stuck
   560→        console.log(`[WA] Reconnecting in ${delay}ms (attempt ${reconnectAttempts})`)
   561→        setTimeout(() => connectWhatsApp(), delay)
   562→      }
   563→    }
   564→
   565→    if (connection === 'open') {
   566→      reconnectAttempts = 0
   567→      console.log('[WA] Connected successfully!')
   568→      if (waNamespace) waNamespace.emit('whatsapp:status', { connected: true })
   569→
   570→      // Get profile picture for connected user
   571→      try {
   572→        const meId = waSocket?.user?.id
   573→        if (meId) {
   574→          const profilePicUrl = await waSocket.profilePictureUrl(meId, 'image')
   575→          if (waNamespace) waNamespace.emit('whatsapp:me', {
   576→            id: meId,
   577→            name: waSocket?.user?.name,
   578→            profilePicUrl,
   579→          })
   580→        }
   581→      } catch { /* Profile pic might not be available */ }
   582→    }
   583→  })
   584→
   585→  } catch (err) {
   586→    console.error('[WA] connectWhatsApp error:', err.message)
   587→    if (waNamespace) waNamespace.emit('whatsapp:status', { connected: false, reason: 'connection_error' })
   588→    // Retry after delay
   589→    reconnectAttempts++
   590→    const delay = Math.min(5000 * reconnectAttempts, 60000)
   591→    console.log(`[WA] Will retry connection in ${delay}ms`)
   592→    setTimeout(() => connectWhatsApp(), delay)
   593→    return
   594→  }
   595→
   596→  // ========== Contacts Sync ==========
   597→  waSocket.ev.on('contacts.upsert', (contacts) => {
   598→    console.log(`[WA] Contacts upsert: ${contacts.length} contacts`)
   599→    for (const contact of contacts) {
   600→      if (!contact.id) continue
   601→      // Skip group contacts and broadcast
   602→      if (contact.id.includes('@g.us') || contact.id === 'status@broadcast') continue
   603→      const phone = extractPhone(contact.id)
   604→      if (!phone) continue
   605→      waContacts.set(contact.id, {
   606→        jid: contact.id,
   607→        name: contact.name || contact.notify || null,
   608→        phone,
   609→        notify: contact.notify || null,
   610→      })
   611→    }
   612→    // Trim if too many
   613→    if (waContacts.size > MAX_CONTACTS) {
   614→      const entries = Array.from(waContacts.keys())
   615→      const toRemove = entries.slice(0, entries.length - MAX_CONTACTS)
   616→      toRemove.forEach(k => waContacts.delete(k))
   617→    }
   618→    // Emit to clients
   619→    if (waNamespace) {
   620→      const contactList = Array.from(waContacts.values())
   621→      waNamespace.emit('whatsapp:contacts', contactList)
   622→    }
   623→  })
   624→
   625→  waSocket.ev.on('contacts.update', (updates) => {
   626→    for (const update of updates) {
   627→      if (!update.id) continue
   628→      const existing = waContacts.get(update.id)
   629→      if (existing) {
   630→        if (update.name) existing.name = update.name
   631→        if (update.notify) existing.notify = update.notify
   632→      }
   633→    }
   634→    if (waNamespace) {
   635→      const contactList = Array.from(waContacts.values())
   636→      waNamespace.emit('whatsapp:contacts', contactList)
   637→    }
   638→  })
   639→
   640→  // ========== Chats/Conversations Sync ==========
   641→  waSocket.ev.on('chats.upsert', (chats) => {
   642→    console.log(`[WA] Chats upsert: ${chats.length} chats`)
   643→    for (const chat of chats) {
   644→      if (!chat.id) continue
   645→      if (chat.id.includes('@g.us') || chat.id === 'status@broadcast') continue
   646→      const conv = getOrCreateConversation(chat.id)
   647→      if (chat.name) conv.name = chat.name
   648→      conv.unreadCount = chat.unreadCount || 0
   649→    }
   650→    if (waNamespace) {
   651→      const convList = Array.from(conversations.values()).map(serializeConversation)
   652→      waNamespace.emit('whatsapp:conversations', convList)
   653→    }
   654→  })
   655→
   656→  // Message events
   657→  waSocket.ev.on('messages.upsert', async ({ messages, type }) => {
   658→    console.log(`[WA] Messages upsert: ${type}, count: ${messages.length}`)
   659→
   660→    for (const msg of messages) {
   661→      if (!msg.key) continue
   662→
   663→      const jid = msg.key.remoteJid
   664→      const fromMe = msg.key.fromMe || false
   665→      const pushName = msg.pushName || null
   666→      const msgKeyId = msg.key.id || ''
   667→
   668→      // Skip status/broadcast messages
   669→      if (jid === 'status@broadcast') continue
   670→
   671→      // ========== DEDUPLICATION ==========
   672→      // Skip our own echoed messages from Baileys (fromMe + append = echo)
   673→      if (fromMe && type === 'append') {
   674→        if (msgKeyId) processedMessageIds.add(msgKeyId)
   675→        continue
   676→      }
   677→
   678→      // Check dedup set
   679→      if (msgKeyId && processedMessageIds.has(msgKeyId)) {
   680→        console.log(`[WA] Skipping duplicate message: ${msgKeyId}`)
   681→        continue
   682→      }
   683→      if (msgKeyId) {
   684→        processedMessageIds.add(msgKeyId)
   685→        // Prevent memory leak
   686→        if (processedMessageIds.size > MAX_DEDUP_IDS) {
   687→          const entries = Array.from(processedMessageIds)
   688→          const toRemove = entries.slice(0, entries.length - MAX_DEDUP_IDS / 2)
   689→          toRemove.forEach(id => processedMessageIds.delete(id))
   690→        }
   691→      }
   692→
   693→      // Get or create conversation
   694→      const conv = getOrCreateConversation(jid, fromMe ? undefined : pushName)
   695→
   696→      // Extract text content
   697→      const textContent =
   698→        msg.message?.conversation ||
   699→        msg.message?.extendedTextMessage?.text ||
   700→        msg.message?.imageMessage?.caption ||
   701→        msg.message?.videoMessage?.caption ||
   702→        msg.message?.documentMessage?.caption ||
   703→        null
   704→
   705→      // Determine media type
   706→      let mediaType = null
   707→      if (msg.message?.imageMessage) mediaType = 'image'
   708→      else if (msg.message?.videoMessage) mediaType = 'video'
   709→      else if (msg.message?.audioMessage) mediaType = 'audio'
   710→      else if (msg.message?.documentMessage) mediaType = 'document'
   711→      else if (msg.message?.stickerMessage) mediaType = 'sticker'
   712→      else if (msg.message?.contactMessage) mediaType = 'contact'
   713→      else if (msg.message?.locationMessage) mediaType = 'location'
   714→
   715→      // Only process messages with content
   716→      if (!textContent && !mediaType) continue
   717→
   718→      const messageData = {
   719→        id: msgKeyId || Math.random().toString(36).substr(2, 9),
   720→        whatsappId: msgKeyId || null,
   721→        fromMe,
   722→        textContent,
   723→        mediaType,
   724→        timestamp: new Date((msg.messageTimestamp || Date.now() / 1000) * 1000),
   725→        status: fromMe ? 'delivered' : 'received',
   726→      }
   727→
   728→      // Add to conversation
   729→      conv.messages.push(messageData)
   730→      if (conv.messages.length > MAX_MESSAGES_PER_CONVERSATION) {
   731→        conv.messages = conv.messages.slice(-MAX_MESSAGES_PER_CONVERSATION)
   732→      }
   733→      conv.lastMessage = textContent || `[${mediaType}]`
   734→      conv.lastMessageAt = messageData.timestamp
   735→      if (!fromMe) conv.unreadCount++
   736→
   737→      // Try to get profile picture (lazy, only once)
   738→      try {
   739→        if (!conv.avatarUrl && waSocket) {
   740→          const picUrl = await waSocket.profilePictureUrl(jid, 'image')
   741→          conv.avatarUrl = picUrl
   742→        }
   743→      } catch { /* No profile pic */ }
   744→
   745→      // Emit to connected clients
   746→      if (waNamespace) {
   747→        waNamespace.emit('whatsapp:message', {
   748→          conversationJid: jid,
   749→          message: messageData,
   750→          conversation: serializeConversation(conv),
   751→        })
   752→        waNamespace.emit('whatsapp:conversation:update', serializeConversation(conv))
   753→      }
   754→
   755→      // ========== AUTO-SYNC TO ODOO (direct call, no bridge) ==========
   756→      if (!fromMe) {
   757→        const phone = extractPhone(jid)
   758→        if (phone) {
   759→          // Non-blocking: fire and forget, errors logged in autoSyncWhatsAppMessage
   760→          autoSyncWhatsAppMessage({
   761→            jid, phone,
   762→            pushName: conv.pushName,
   763→            textContent, mediaType, fromMe,
   764→            timestamp: messageData.timestamp.toISOString(),
   765→          }).catch(err => console.error('[AutoSync] Unhandled error:', err.message))
   766→
   767→          // Also notify frontend about the sync
   768→          if (waNamespace) {
   769→            waNamespace.emit('whatsapp:odoo-sync', { jid, phone, syncing: true })
   770→          }
   771→        }
   772→      }
   773→    }
   774→  })
   775→}

   781→async function autoAuthenticateFromEnv() {
   782→  const envUrl = process.env.ODOO_URL
   783→  const envDb = process.env.ODOO_DB
   784→  const envUsername = process.env.ODOO_USERNAME
   785→  const envPassword = process.env.ODOO_PASSWORD
   786→
   787→  if (envUrl && envDb && envUsername && envPassword) {
   788→    console.log(`[Odoo] Auto-authenticating with env vars: ${envUrl} / ${envDb} / ${envUsername}`)
   789→    try {
   790→      odooConfig = { url: envUrl, db: envDb, username: envUsername, password: envPassword, uid: null }
   791→      modelFieldsCache.clear()
   792→      phoneToPartnerCache.clear()
   793→      const uid = await odooAuthenticate()
   794→      odooConfig.uid = uid
   795→      console.log(`[Odoo] Auto-authenticated as ${envUsername} (uid: ${uid})`)
   796→
   797→      if (odooNamespace) {
   798→        odooNamespace.emit('odoo:status', {
   799→          connected: true,
   800→          url: odooConfig.url,
   801→          db: odooConfig.db,
   802→          username: odooConfig.username,
   803→        })
   804→      }
   805→    } catch (error) {
   806→      console.error(`[Odoo] Auto-authentication failed: ${error.message}`)
   807→    }
   808→  } else {
   809→    console.log('[Odoo] No ODOO_URL/DB/USERNAME/PASSWORD env vars. Waiting for manual connection.')
   810→  }
   811→}

   817→function setupWhatsAppNamespace(namespace) {
   818→  namespace.on('connection', (socket) => {
   819→    console.log(`[WA IO] Client connected: ${socket.id}`)
   820→
   821→    // Send current status
   822→    socket.emit('whatsapp:status', {
   823→      connected: connectionState.connection === 'open',
   824→    })
   825→
   826→    // Send current conversations list
   827→    const convList = Array.from(conversations.values()).map(serializeConversation)
   828→    socket.emit('whatsapp:conversations', convList)
   829→
   830→    // Send current contacts list
   831→    const contactList = Array.from(waContacts.values())
   832→    socket.emit('whatsapp:contacts', contactList)
   833→
   834→    // Request QR code regeneration
   835→    socket.on('whatsapp:request-qr', () => {
   836→      reconnectAttempts = 0
   837→      if (connectionState.connection !== 'open' && waSocket) {
   838→        socket.emit('whatsapp:status', { connected: false, reason: 'connecting' })
   839→      } else if (!waSocket) {
   840→        connectWhatsApp()
   841→      }
   842→    })
   843→
   844→    // Get conversation messages
   845→    socket.on('whatsapp:get-messages', (data, callback) => {
   846→      const conv = conversations.get(data.jid)
   847→      callback?.(conv ? { messages: conv.messages.slice(-100) } : { messages: [] })
   848→    })
   849→
   850→    // Get conversation info
   851→    socket.on('whatsapp:get-conversation-info', (data, callback) => {
   852→      const conv = conversations.get(data.jid)
   853→      callback?.(conv
   854→        ? { success: true, conversation: serializeConversation(conv), messages: conv.messages.slice(-100) }
   855→        : { success: false, conversation: null, messages: [] })
   856→    })
   857→
   858→    // Send a text message
   859→    socket.on('whatsapp:send-message', async (data, callback) => {
   860→      try {
   861→        if (!waSocket || connectionState.connection !== 'open') {
   862→          return callback?.({ success: false, error: 'WhatsApp not connected' })
   863→        }
   864→
   865→        const sent = await waSocket.sendMessage(data.jid, { text: data.text })
   866→        const conv = getOrCreateConversation(data.jid)
   867→        const msgKeyId = sent.key.id || ''
   868→        const messageData = {
   869→          id: msgKeyId || Math.random().toString(36).substr(2, 9),
   870→          whatsappId: msgKeyId || null,
   871→          fromMe: true,
   872→          textContent: data.text,
   873→          mediaType: null,
   874→          timestamp: new Date(),
   875→          status: 'sent',
   876→        }
   877→
   878→        // Mark as processed to prevent duplicate from Baileys echo
   879→        if (msgKeyId) processedMessageIds.add(msgKeyId)
   880→
   881→        conv.messages.push(messageData)
   882→        if (conv.messages.length > MAX_MESSAGES_PER_CONVERSATION) {
   883→          conv.messages = conv.messages.slice(-MAX_MESSAGES_PER_CONVERSATION)
   884→        }
   885→        conv.lastMessage = data.text
   886→        conv.lastMessageAt = messageData.timestamp
   887→
   888→        if (waNamespace) {
   889→          waNamespace.emit('whatsapp:message', {
   890→            conversationJid: data.jid,
   891→            message: messageData,
   892→            conversation: serializeConversation(conv),
   893→          })
   894→          waNamespace.emit('whatsapp:conversation:update', serializeConversation(conv))
   895→        }
   896→
   897→        callback?.({ success: true, messageId: sent.key.id })
   898→      } catch (error) {
   899→        console.error('[WA] Send message error:', error.message)
   900→        callback?.({ success: false, error: error.message })
   901→      }
   902→    })
   903→
   904→    // Send media message
   905→    socket.on('whatsapp:send-media', async (data, callback) => {
   906→      try {
   907→        if (!waSocket || connectionState.connection !== 'open') {
   908→          return callback?.({ success: false, error: 'WhatsApp not connected' })
   909→        }
   910→
   911→        let sent
   912→        if (data.type === 'image') {
   913→          sent = await waSocket.sendMessage(data.jid, { image: { url: data.url }, caption: data.caption })
   914→        } else if (data.type === 'document') {
   915→          sent = await waSocket.sendMessage(data.jid, { document: { url: data.url }, fileName: data.fileName || 'document', mimetype: data.mimeType, caption: data.caption })
   916→        } else if (data.type === 'video') {
   917→          sent = await waSocket.sendMessage(data.jid, { video: { url: data.url }, caption: data.caption })
   918→        } else if (data.type === 'audio') {
   919→          sent = await waSocket.sendMessage(data.jid, { audio: { url: data.url }, mimetype: data.mimeType || 'audio/mp4' })
   920→        } else {
   921→          return callback?.({ success: false, error: 'Unsupported media type' })
   922→        }
   923→
   924→        const conv = getOrCreateConversation(data.jid)
   925→        const msgKeyId = sent.key.id || ''
   926→        const messageData = {
   927→          id: msgKeyId || Math.random().toString(36).substr(2, 9),
   928→          whatsappId: msgKeyId || null,
   929→          fromMe: true,
   930→          textContent: data.caption || null,
   931→          mediaType: data.type,
   932→          timestamp: new Date(),
   933→          status: 'sent',
   934→        }
   935→
   936→        if (msgKeyId) processedMessageIds.add(msgKeyId)
   937→
   938→        conv.messages.push(messageData)
   939→        conv.lastMessage = data.caption || `[${data.type}]`
   940→        conv.lastMessageAt = messageData.timestamp
   941→
   942→        if (waNamespace) {
   943→          waNamespace.emit('whatsapp:message', {
   944→            conversationJid: data.jid,
   945→            message: messageData,
   946→            conversation: serializeConversation(conv),
   947→          })
   948→        }
   949→
   950→        callback?.({ success: true, messageId: sent.key.id })
   951→      } catch (error) {
   952→        console.error('[WA] Send media error:', error.message)
   953→        callback?.({ success: false, error: error.message })
   954→      }
   955→    })
   956→
   957→    // Mark conversation as read
   958→    socket.on('whatsapp:mark-read', async (data, callback) => {
   959→      try {
   960→        if (!waSocket || connectionState.connection !== 'open') {
   961→          return callback?.({ success: false, error: 'WhatsApp not connected' })
   962→        }
   963→        const conv = conversations.get(data.jid)
   964→        if (conv) {
   965→          conv.unreadCount = 0
   966→          if (waNamespace) waNamespace.emit('whatsapp:conversation:update', serializeConversation(conv))
   967→        }
   968→        await waSocket.readMessages([{ remoteJid: data.jid, id: '' }])
   969→        callback?.({ success: true })
   970→      } catch (error) {
   971→        callback?.({ success: false, error: error.message })
   972→      }
   973→    })
   974→
   975→    // Disconnect WhatsApp
   976→    socket.on('whatsapp:disconnect', async (callback) => {
   977→      try {
   978→        if (waSocket) {
   979→          await waSocket.logout('User requested disconnect')
   980→          waSocket = null
   981→          connectionState = { connection: 'close' }
   982→          if (waNamespace) waNamespace.emit('whatsapp:status', { connected: false, reason: 'logged_out' })
   983→          callback?.({ success: true })
   984→        } else {
   985→          callback?.({ success: false, error: 'Not connected' })
   986→        }
   987→      } catch (error) {
   988→        callback?.({ success: false, error: error.message })
   989→      }
   990→    })
   991→
   992→    // ========== Check if phone is on WhatsApp ==========
   993→    socket.on('whatsapp:check-number', async (data, callback) => {
   994→      try {
   995→        if (!waSocket || connectionState.connection !== 'open') {
   996→          return callback?.({ success: false, error: 'WhatsApp not connected' })
   997→        }
   998→        const phone = data.phone.replace(/[^0-9]/g, '')
   999→        if (!phone) return callback?.({ success: false, error: 'Invalid phone number' })
  1000→
  1001→        const [result] = await waSocket.onWhatsApp(`${phone}@s.whatsapp.net`)
  1002→        if (result && result.exists) {
  1003→          callback?.({ success: true, exists: true, jid: result.jid })
  1004→        } else {
  1005→          callback?.({ success: true, exists: false, jid: null })
  1006→        }
  1007→      } catch (error) {
  1008→        callback?.({ success: false, error: error.message })
  1009→      }
  1010→    })
  1011→
  1012→    // ========== Start new conversation by phone ==========
  1013→    socket.on('whatsapp:start-conversation', async (data, callback) => {
  1014→      try {
  1015→        if (!waSocket || connectionState.connection !== 'open') {
  1016→          return callback?.({ success: false, error: 'WhatsApp not connected' })
  1017→        }
  1018→        const phone = data.phone.replace(/[^0-9]/g, '')
  1019→        if (!phone) return callback?.({ success: false, error: 'Invalid phone number' })
  1020→
  1021→        const jid = data.jid || `${phone}@s.whatsapp.net`
  1022→
  1023→        // Verify number is on WhatsApp (unless jid already provided)
  1024→        if (!data.jid) {
  1025→          try {
  1026→            const [result] = await waSocket.onWhatsApp(jid)
  1027→            if (!result || !result.exists) {
  1028→              return callback?.({ success: false, error: 'Este numero nao esta no WhatsApp' })
  1029→            }
  1030→          } catch (err) {
  1031→            console.error('[WA] onWhatsApp check error:', err.message)
  1032→            // Continue anyway - might still work
  1033→          }
  1034→        }
  1035→
  1036→        // Create or get conversation
  1037→        const conv = getOrCreateConversation(jid, data.name || null)
  1038→
  1039→        // Try to get profile picture
  1040→        try {
  1041→          if (!conv.avatarUrl && waSocket) {
  1042→            const picUrl = await waSocket.profilePictureUrl(jid, 'image')
  1043→            conv.avatarUrl = picUrl
  1044→          }
  1045→        } catch { /* No profile pic */ }
  1046→
  1047→        const serialized = serializeConversation(conv)
  1048→
  1049→        // Emit to all clients
  1050→        if (waNamespace) {
  1051→          waNamespace.emit('whatsapp:conversation:update', serialized)
  1052→        }
  1053→
  1054→        callback?.({ success: true, conversation: serialized, jid })
  1055→      } catch (error) {
  1056→        console.error('[WA] Start conversation error:', error.message)
  1057→        callback?.({ success: false, error: error.message })
  1058→      }
  1059→    })
  1060→
  1061→    // ========== Get WhatsApp contacts ==========
  1062→    socket.on('whatsapp:get-contacts', (data, callback) => {
  1063→      const contactList = Array.from(waContacts.values())
  1064→      if (callback) {
  1065→        callback?.({ success: true, data: contactList })
  1066→      }
  1067→    })
  1068→
  1069→    // Get profile picture
  1070→    socket.on('whatsapp:get-profile-pic', async (data, callback) => {
  1071→      try {
  1072→        if (!waSocket || connectionState.connection !== 'open') {
  1073→          return callback?.({ success: false, url: null })
  1074→        }
  1075→        const url = await waSocket.profilePictureUrl(data.jid, 'image')
  1076→        const conv = conversations.get(data.jid)
  1077→        if (conv) conv.avatarUrl = url
  1078→        callback?.({ success: true, url })
  1079→      } catch {
  1080→        callback?.({ success: false, url: null })
  1081→      }
  1082→    })
  1083→
  1084→    socket.on('disconnect', () => {
  1085→      // console.log(`[WA IO] Client disconnected: ${socket.id}`)
  1086→    })
  1087→  })
  1088→}

  1091→// SECTION: Socket.io /odoo Namespace Handlers
