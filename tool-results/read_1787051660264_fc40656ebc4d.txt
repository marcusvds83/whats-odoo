   390→function NavItem({ icon, label, active, onClick, badge, collapsed }: {
   391→  icon: React.ReactNode
   392→  label: string
   393→  active: boolean
   394→  onClick: () => void
   395→  badge?: string
   396→  collapsed?: boolean
   397→}) {
   398→  return (
   399→    <button
   400→      onClick={onClick}
   401→      className={cn(
   402→        'w-full flex items-center gap-3 px-2 py-2 rounded-lg text-sm transition-colors',
   403→        'hover:bg-muted/80 focus-visible:outline-none focus-visible:bg-muted/80',
   404→        active ? 'bg-primary/10 text-primary font-medium' : 'text-muted-foreground',
   405→      )}
   406→      title={collapsed ? label : undefined}
   407→    >
   408→      <span className="shrink-0 mx-auto lg:mx-0">{icon}</span>
   409→      {!collapsed && <span className="hidden lg:block flex-1 text-left truncate">{label}</span>}
   410→      {badge && !collapsed && (
   411→        <Badge className="size-5 p-0 flex items-center justify-center text-[10px] bg-emerald-500 text-white border-0">
   412→          {badge === 'online' ? '\u25CF' : badge}
   413→        </Badge>
   414→      )}
   415→      {badge && collapsed && (
   416→        <Badge className="hidden lg:flex absolute top-0 right-0 size-2 p-0 bg-emerald-500 border-0" />
   417→      )}
   418→    </button>
   419→  )
   420→}
