   382→function NavItem({ icon, label, active, onClick, badge, collapsed }: {
   383→  icon: React.ReactNode
   384→  label: string
   385→  active: boolean
   386→  onClick: () => void
   387→  badge?: string
   388→  collapsed?: boolean
   389→}) {
   390→  return (
   391→    <button
   392→      onClick={onClick}
   393→      className={cn(
   394→        'w-full flex items-center gap-3 px-2 py-2 rounded-lg text-sm transition-colors',
   395→        'hover:bg-muted/80 focus-visible:outline-none focus-visible:bg-muted/80',
   396→        active ? 'bg-primary/10 text-primary font-medium' : 'text-muted-foreground',
   397→      )}
   398→      title={collapsed ? label : undefined}
   399→    >
   400→      <span className="shrink-0 mx-auto lg:mx-0">{icon}</span>
   401→      {!collapsed && <span className="hidden lg:block flex-1 text-left truncate">{label}</span>}
   402→      {badge && !collapsed && (
   403→        <Badge className="size-5 p-0 flex items-center justify-center text-[10px] bg-emerald-500 text-white border-0">
   404→          {badge === 'online' ? '\u25CF' : badge}
   405→        </Badge>
   406→      )}
   407→      {badge && collapsed && (
   408→        <Badge className="hidden lg:flex absolute top-0 right-0 size-2 p-0 bg-emerald-500 border-0" />
   409→      )}
   410→    </button>
   411→  )
   412→}

   414→function StatusIndicator({ label, connected, collapsed }: { label: string; connected: boolean; collapsed?: boolean }) {
   415→  return (
   416→    <div className="flex items-center gap-2 px-2 py-1.5 text-xs" title={collapsed ? `${label}: ${connected ? 'ON' : 'OFF'}` : undefined}>
   417→      <span className={cn('size-2 rounded-full shrink-0 mx-auto lg:mx-0', connected ? 'bg-emerald-500' : 'bg-muted-foreground/30')} />

   419→      {!collapsed && <span className="hidden lg:block ml-auto font-medium text-[10px]">{connected ? 'ON' : 'OFF'}</span>}
