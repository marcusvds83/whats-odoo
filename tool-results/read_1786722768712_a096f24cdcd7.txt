    86→function formatMessageTime(dateStr: string): string {
    87→  return new Date(dateStr).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    88→}
    89→
    90→function formatMessageDate(dateStr: string): string {
    91→  const date = new Date(dateStr)
    92→  const now = new Date()
    93→  const diffDays = Math.floor((now.getTime() - date.getTime()) / (1000 * 60 * 60 * 24))
    94→  if (diffDays === 0) return 'Hoje'
    95→  if (diffDays === 1) return 'Ontem'
    96→  return date.toLocaleDateString([], { month: 'long', day: 'numeric', year: 'numeric' })
    97→}

    99→function getMediaIcon(mediaType: string | null) {
   100→  switch (mediaType) {
   101→    case 'image': return <ImageIcon className="size-4" />
   102→    case 'video': return <Video className="size-4" />
   103→    case 'audio': case 'ptt': return <Music className="size-4" />
   104→    case 'document': case 'pdf': return <FileText className="size-4" />
   105→    default: return <Paperclip className="size-4" />
   106→  }
   107→}

   110→  switch (mediaType) {
   111→    case 'image': return 'Foto'
   112→    case 'video': return 'Video'
   113→    case 'audio': return 'Audio'
   114→    case 'ptt': return 'Mensagem de voz'
   115→    case 'document': return 'Documento'
   116→    case 'pdf': return 'PDF'
   117→    case 'sticker': return 'Sticker'
   118→    default: return 'Midia'
   119→  }

   121→
   122→function getMessageStatusIcon(status: string, fromMe: boolean) {
   123→  if (!fromMe) return null
   124→  switch (status) {
   125→    case 'pending': return <Clock className="size-3.5 text-muted-foreground" />
   126→    case 'sent': return <Check className="size-3.5 text-muted-foreground" />
   127→    case 'delivered': return <CheckCheck className="size-3.5 text-muted-foreground" />
   128→    case 'read': return <CheckCheck className="size-3.5 text-blue-500" />
   129→    default: return <Clock className="size-3.5 text-muted-foreground" />
   130→  }
   131→}
   132→
   133→function shouldShowDateDivider(currentMsg: { timestamp: string }, prevMsg: { timestamp: string } | undefined): boolean {
   134→  if (!prevMsg) return true
   135→  return new Date(currentMsg.timestamp).toDateString() !== new Date(prevMsg.timestamp).toDateString()
   136→}

   138→export function ChatView({
   139→  conversation,
   140→  messages,
   141→  onSendMessage,
   142→  onSendMedia,
   143→  onMarkRead,
   144→  odooConnected,
   145→  odooLinkedRecords,
   146→  onFetchOdooHistory,
   147→  onInjectHistory,
   148→  onDeleteConversation,
   149→  onRefreshMessages,
   150→}: ChatViewProps) {
   151→  const [inputText, setInputText] = useState('')
   152→  const [isSending, setIsSending] = useState(false)
   153→  const [isSendingMedia, setIsSendingMedia] = useState(false)
   154→  const [showScrollButton, setShowScrollButton] = useState(false)
   155→  const [isPullingHistory, setIsPullingHistory] = useState(false)
   156→  const [pullResult, setPullResult] = useState<{ added: number; skipped: number } | null>(null)
   157→  const [isRefreshing, setIsRefreshing] = useState(false)
   158→  const [refreshResult, setRefreshResult] = useState<{ count: number; serverFetchAttempted: boolean; methods?: string[] } | null>(null)
   159→  const messagesEndRef = useRef<HTMLDivElement>(null)
   160→  const scrollViewportRef = useRef<HTMLDivElement>(null)
   161→  const inputRef = useRef<HTMLInputElement>(null)
   162→  const imageInputRef = useRef<HTMLInputElement>(null)
   163→  const audioInputRef = useRef<HTMLInputElement>(null)
   164→  const fileInputRef = useRef<HTMLInputElement>(null)
   165→
   166→  const scrollToBottom = useCallback((behavior: ScrollBehavior = 'smooth') => {
   167→    messagesEndRef.current?.scrollIntoView({ behavior })
   168→  }, [])
   169→
   170→  useEffect(() => { scrollToBottom() }, [messages.length, scrollToBottom])
   171→
   172→  useEffect(() => {
   173→    if (conversation) onMarkRead(conversation.jid)
   174→  }, [conversation?.jid, onMarkRead])
   175→
   176→  // Reset pull result when conversation changes
   177→  useEffect(() => {
   178→    setPullResult(null)
   179→    setRefreshResult(null)
   180→  }, [conversation?.jid])
   181→
   182→  // Pull history from Odoo — used when the user comes back the next day
   183→  // and the local conversation history was lost from memory.
   184→  const handlePullFromOdoo = useCallback(async () => {
   185→    if (!conversation || !odooLinkedRecords || odooLinkedRecords.length === 0) return
   186→    if (!onFetchOdooHistory || !onInjectHistory) return
   187→
   188→    setIsPullingHistory(true)
   189→    setPullResult(null)
   190→    try {
   191→      // Pull from all linked records (typically just one lead or contact)
   192→      const allMessages: any[] = []
   193→      for (const rec of odooLinkedRecords) {
   194→        const result = await onFetchOdooHistory(rec.model, rec.recordId)
   195→        if (result.success && result.data) {
   196→          allMessages.push(...result.data)
   197→        }
   198→      }
   199→
   200→      if (allMessages.length === 0) {
   201→        setPullResult({ added: 0, skipped: 0 })
   202→        setTimeout(() => setPullResult(null), 3000)
   203→        return
   204→      }
   205→
   206→      // Inject into local conversation
   207→      const injectResult = await onInjectHistory(conversation.jid, allMessages)
   208→      if (injectResult.success) {
   209→        setPullResult({
   210→          added: injectResult.added || 0,
   211→          skipped: injectResult.skipped || 0,
   212→        })
   213→        setTimeout(() => setPullResult(null), 5000)
   214→      } else {
   215→        setPullResult({ added: 0, skipped: 0 })
   216→        setTimeout(() => setPullResult(null), 3000)
   217→      }
   218→    } catch (err) {
   219→      setPullResult({ added: 0, skipped: 0 })
   220→      setTimeout(() => setPullResult(null), 3000)
   221→    } finally {
   222→      setIsPullingHistory(false)
   223→    }
   224→  }, [conversation, odooLinkedRecords, onFetchOdooHistory, onInjectHistory])
   225→
   226→  // v7.14/v7.15: Manual refresh messages — fetches latest messages FROM THE PHONE.
   227→  // Calls server which uses BOTH `fetchMessageHistory` (older messages) AND
   228→  // `resyncAppState` (force full app-state re-sync) to bring in messages that
   229→  // may have been missed by `messages.upsert`. This is the manual fallback the
   230→  // user requested — "o botão na conversa pra trazer mensagens... do aparelho".
   231→  const handleRefreshMessages = useCallback(async () => {
   232→    if (!conversation || !onRefreshMessages) return
   233→    setIsRefreshing(true)
   234→    setRefreshResult(null)
   235→    try {
   236→      const result = await onRefreshMessages(conversation.jid)
   237→      if (result.success) {
   238→        setRefreshResult({
   239→          count: result.count || 0,
   240→          serverFetchAttempted: !!result.serverFetchAttempted,
   241→          // v7.15: show what server methods were attempted
   242→          methods: (result as any).serverFetchMethods || [],
   243→        })
   244→        setTimeout(() => setRefreshResult(null), 5000)
   245→      } else {
   246→        setRefreshResult({ count: 0, serverFetchAttempted: false, methods: [] })
   247→        setTimeout(() => setRefreshResult(null), 3000)
   248→      }
   249→    } catch (err) {
   250→      setRefreshResult({ count: 0, serverFetchAttempted: false, methods: [] })
   251→      setTimeout(() => setRefreshResult(null), 3000)
   252→    } finally {
   253→      setIsRefreshing(false)
   254→    }
   255→  }, [conversation, onRefreshMessages])
   256→
   257→  // Track scroll position for scroll-to-bottom button
   258→  useEffect(() => {
   259→    // The ScrollArea viewport is the element with data-slot="scroll-area-viewport"
   260→    const viewport = scrollViewportRef.current?.querySelector('[data-slot="scroll-area-viewport"]') as HTMLElement | null
   261→    if (!viewport) return
   262→
   263→    const handleScroll = () => {
   264→      const { scrollTop, scrollHeight, clientHeight } = viewport
   265→      setShowScrollButton(scrollHeight - scrollTop - clientHeight > 150)
   266→    }
   267→    viewport.addEventListener('scroll', handleScroll)
   268→    return () => viewport.removeEventListener('scroll', handleScroll)
   269→  }, [])
   270→
   271→  const handleSend = async () => {
   272→    const text = inputText.trim()
   273→    if (!text || !conversation || isSending) return
   274→    setIsSending(true)
   275→    try {
   276→      const success = await onSendMessage(conversation.jid, text)
   277→      if (success) setInputText('')
   278→    } finally {
   279→      setIsSending(false)
   280→      inputRef.current?.focus()
   281→    }
   282→  }
   283→
   284→  const handleKeyDown = (e: React.KeyboardEvent) => {
   285→    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); handleSend() }
   286→  }
   287→
   288→  // v7.22: Handle media file selection (image/audio/document)
   289→  const handleFileSelect = async (file: File, caption?: string) => {
   290→    if (!conversation || !onSendMedia || isSendingMedia) return
   291→    setIsSendingMedia(true)
   292→    try {
   293→      const result = await onSendMedia(conversation.jid, file, caption)
   294→      if (!result.success) {
   295→        console.error('[ChatView] Failed to send media:', result.error)
   296→        // Could add a toast here, but the existing code uses console only
   297→      }
   298→    } finally {
   299→      setIsSendingMedia(false)
   300→    }
   301→  }
   302→
   303→  const handleImageSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   304→    const file = e.target.files?.[0]
   305→    if (file) handleFileSelect(file)
   306→    // Reset input so same file can be selected again
   307→    e.target.value = ''
   308→  }
   309→
   310→  const handleAudioSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   311→    const file = e.target.files?.[0]
   312→    if (file) handleFileSelect(file)
   313→    e.target.value = ''
   314→  }
   315→
   316→  const handleDocumentSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   317→    const file = e.target.files?.[0]
   318→    if (file) handleFileSelect(file)
   319→    e.target.value = ''
   320→  }
   321→
   322→  // Insert emoji at cursor position in the input
   323→  const handleEmojiSelect = (emoji: string) => {
   324→    const input = inputRef.current
   325→    if (!input) {
   326→      setInputText(prev => prev + emoji)
   327→      return
   328→    }
   329→    const start = input.selectionStart ?? inputText.length
   330→    const end = input.selectionEnd ?? inputText.length
   331→    const next = inputText.slice(0, start) + emoji + inputText.slice(end)
   332→    setInputText(next)
   333→    // Restore cursor position after emoji
   334→    requestAnimationFrame(() => {
   335→      const pos = start + emoji.length
   336→      input.focus()
   337→      input.setSelectionRange(pos, pos)
   338→    })
   339→  }
   340→
   341→  const displayName = conversation?.name || conversation?.pushName || conversation?.phone || conversation?.jid?.split('@')[0] || ''
   342→
   343→  if (!conversation) {
   344→    return (
   345→      <div className="flex flex-col items-center justify-center h-full bg-muted/20">
   346→        <div className="flex flex-col items-center gap-3 text-center px-4">
   347→          <div className="size-20 rounded-full bg-muted/60 flex items-center justify-center">
   348→            <MessageSquare className="size-8 text-muted-foreground/50" />
   349→          </div>
   350→          <h3 className="text-lg font-semibold text-muted-foreground">Nenhuma conversa selecionada</h3>
   351→          <p className="text-sm text-muted-foreground/70 max-w-[280px]">
   352→            Selecione uma conversa da lista para ver as mensagens
   353→          </p>
   354→        </div>
   355→      </div>
   356→    )
   357→  }
   358→
   359→  return (
   360→    <div className="flex flex-col h-full min-h-0">
   361→      {/* Header — fixed */}
   362→      <div className="shrink-0 border-b bg-background/95 backdrop-blur">
   363→        <div className="flex items-center gap-3 px-4 py-3">
   364→          <Avatar className="size-10 shrink-0">
   365→            {conversation.avatarUrl && <AvatarImage src={conversation.avatarUrl} alt={displayName} />}
   366→            <AvatarFallback className="bg-primary/10 text-primary text-sm">
   367→              {displayName.split(' ').map(w => w[0]).filter(Boolean).slice(0, 2).join('').toUpperCase() || <User className="size-4" />}
   368→            </AvatarFallback>
   369→          </Avatar>
   370→          <div className="flex-1 min-w-0">
   371→            <h3 className="font-semibold text-sm truncate">{displayName}</h3>
   372→            <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   373→              {conversation.phone ? (
   374→                <><Phone className="size-3" /><span>{conversation.phone}</span></>
   375→              ) : (
   376→                <span>{conversation.jid.split('@')[0]}</span>
   377→              )}
   378→              {odooLinkedRecords && odooLinkedRecords.length > 0 && (
   379→                <Badge variant="outline" className="ml-1 text-[10px] gap-0.5 py-0 h-4 bg-amber-50 text-amber-700 border-amber-200">
   380→                  <Link2 className="size-2.5" />
   381→                  Odoo
   382→                </Badge>
   383→              )}
   384→            </div>
   385→          </div>
   386→
   387→          {/* Pull from Odoo button — only shown when conversation is linked to Odoo */}
   388→          {odooConnected && odooLinkedRecords && odooLinkedRecords.length > 0 && onFetchOdooHistory && onInjectHistory && (
   389→            <TooltipProvider delayDuration={300}>
   390→              <Tooltip>
   391→                <TooltipTrigger asChild>
   392→                  <Button
   393→                    variant="outline"
   394→                    size="sm"
   395→                    className="h-8 gap-1.5 shrink-0 text-xs"
   396→                    disabled={isPullingHistory}
   397→                    onClick={handlePullFromOdoo}
   398→                  >
   399→                    {isPullingHistory ? (
   400→                      <Loader2 className="size-3.5 animate-spin" />
   401→                    ) : (
   402→                      <DownloadCloud className="size-3.5" />
   403→                    )}
   404→                    <span className="hidden sm:inline">
   405→                      {isPullingHistory ? 'Buscando...' : 'Trazer do Odoo'}
   406→                    </span>
   407→                  </Button>
   408→                </TooltipTrigger>
   409→                <TooltipContent side="bottom" className="max-w-xs">
   410→                  <p>Busca o histórico da conversa armazenado no Odoo (chatter) e mescla com a conversa local atual.</p>
   411→                  <p className="text-xs text-muted-foreground mt-1">
   412→                    Use quando voltar no dia seguinte e o histórico do WhatsApp não estiver mais carregado.
   413→                  </p>
   414→                </TooltipContent>
   415→              </Tooltip>
   416→            </TooltipProvider>
   417→          )}
   418→
   419→          {/* v7.15: Refresh messages button — fetches latest messages FROM THE PHONE */}
   420→          {onRefreshMessages && (
   421→            <TooltipProvider delayDuration={300}>
   422→              <Tooltip>
   423→                <TooltipTrigger asChild>
   424→                  <Button
   425→                    variant="outline"
   426→                    size="sm"
   427→                    className="h-8 gap-1.5 shrink-0 text-xs border-emerald-200 bg-emerald-50 hover:bg-emerald-100 text-emerald-700"
   428→                    disabled={isRefreshing}
   429→                    onClick={handleRefreshMessages}
   430→                  >
   431→                    {isRefreshing ? (
   432→                      <Loader2 className="size-3.5 animate-spin" />
   433→                    ) : (
   434→                      <RefreshCw className="size-3.5" />
   435→                    )}
   436→                    <span className="hidden sm:inline">
   437→                      {isRefreshing ? 'Buscando...' : 'Buscar no aparelho'}
   438→                    </span>
   439→                  </Button>
   440→                </TooltipTrigger>
   441→                <TooltipContent side="bottom" className="max-w-xs">
   442→                  <p className="font-medium">Busca as mensagens mais recentes do aparelho conectado.</p>
   443→                  <p className="text-xs text-muted-foreground mt-1">
   444→                    Força uma ressincronização com o WhatsApp (resyncAppState + fetchMessageHistory).
   445→                    Use se uma mensagem não chegou automaticamente.
   446→                  </p>
   447→                </TooltipContent>
   448→              </Tooltip>
   449→            </TooltipProvider>
   450→          )}
   451→
   452→          {/* v7.15: Refresh result feedback */}
   453→          {refreshResult && (
   454→            <div className={cn(
   455→              'flex items-center gap-1.5 px-2 py-1 rounded-md text-[11px] shrink-0',
   456→              refreshResult.count > 0
   457→                ? 'bg-emerald-50 text-emerald-700 border border-emerald-200'
   458→                : 'bg-muted text-muted-foreground border'
   459→            )}>
   460→              <CheckCheck className="size-3" />
   461→              <span>{refreshResult.count} msgs</span>
   462→              {refreshResult.serverFetchAttempted && (
   463→                <span className="text-[9px] text-muted-foreground ml-1">
   464→                  ({refreshResult.methods?.join('+') || 'servidor'})
   465→                </span>
   466→              )}
   467→            </div>
   468→          )}
   469→
   470→          {/* Pull result feedback */}
   471→          {pullResult && (
   472→            <div className={cn(
   473→              'flex items-center gap-1.5 px-2 py-1 rounded-md text-[11px] shrink-0',
   474→              pullResult.added > 0
   475→                ? 'bg-emerald-50 text-emerald-700 border border-emerald-200'
   476→                : 'bg-muted text-muted-foreground border'
   477→            )}>
   478→              {pullResult.added > 0 ? (
   479→                <>
   480→                  <CheckCheck className="size-3" />
   481→                  <span>+{pullResult.added} mensagens</span>
   482→                </>
   483→              ) : (
   484→                <span>Nada novo</span>
   485→              )}
   486→            </div>
   487→          )}
   488→
   489→          {/* Delete conversation button */}
   490→          {onDeleteConversation && (
   491→            <TooltipProvider delayDuration={300}>
   492→              <Tooltip>
   493→                <TooltipTrigger asChild>
   494→                  <Button
   495→                    variant="ghost"
   496→                    size="icon"
   497→                    className="size-8 shrink-0 hover:bg-destructive/10 hover:text-destructive"
   498→                    onClick={() => onDeleteConversation(conversation.jid)}
   499→                  >
   500→                    <Trash2 className="size-4" />
   501→                  </Button>
   502→                </TooltipTrigger>
   503→                <TooltipContent side="bottom" className="text-xs">
   504→                  Excluir conversa
   505→                </TooltipContent>
   506→              </Tooltip>
   507→            </TooltipProvider>
   508→          )}
   509→        </div>
   510→      </div>
   511→
   512→      {/* Messages — scrollable */}
   513→      <div className="flex-1 min-h-0 relative">
   514→        <ScrollArea ref={scrollViewportRef} className="h-full">
   515→          <div className="px-4 py-3 space-y-1">
   516→            {messages.length === 0 ? (
   517→              <div className="flex flex-col items-center justify-center py-16">
   518→                <div className="size-14 rounded-full bg-muted/50 flex items-center justify-center mb-3">
   519→                  <MessageSquare className="size-6 text-muted-foreground/40" />
   520→                </div>
   521→                <p className="text-sm text-muted-foreground">Sem mensagens</p>
   522→                <p className="text-xs text-muted-foreground/60 mt-1">Envie uma mensagem para iniciar</p>
   523→              </div>
   524→            ) : (
   525→              messages.map((message, index) => {
   526→                const prevMessage = index > 0 ? messages[index - 1] : undefined
   527→                const showDateDivider = shouldShowDateDivider(message, prevMessage)
   528→                const isConsecutive = prevMessage && prevMessage.fromMe === message.fromMe &&
   529→                  new Date(message.timestamp).getTime() - new Date(prevMessage.timestamp).getTime() < 60000
   530→
   531→                return (
   532→                  <div key={message.id}>
   533→                    {showDateDivider && (
   534→                      <div className="flex items-center justify-center py-3">
