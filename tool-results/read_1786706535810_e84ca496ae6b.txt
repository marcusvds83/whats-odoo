    61→// ========== Odoo XML-RPC Client ==========
    62→function makeXmlRpcClient(path: string) {
    63→  const url = new URL(odooConfig.url)
    64→  const isHttps = url.protocol === 'https:'
    65→  const options = {
    66→    host: url.hostname,
    67→    port: parseInt(url.port) || (isHttps ? 443 : 80),
    68→    path,
    69→  }
    70→  // CRITICAL: Must use createSecureClient for HTTPS (Odoo SaaS)
    71→  return isHttps ? createSecureClient(options) : createClient(options)
    72→}
    73→
    74→function getOdooClient(path: string = '/xmlrpc/2/object') {
    75→  return makeXmlRpcClient(path)
    76→}

    78→function odooAuthenticate(): Promise<number> {
    79→  return new Promise((resolve, reject) => {
    80→    const client = makeXmlRpcClient('/xmlrpc/2/common')
    81→
    82→    client.methodCall('authenticate', [
    83→      odooConfig.db,
    84→      odooConfig.username,
    85→      odooConfig.password,
    86→      {},
    87→    ], (error: any, value: any) => {
    88→      if (error) {
    89→        reject(error)
    90→      } else if (!value) {
    91→        reject(new Error('Authentication failed - invalid credentials'))
    92→      } else {
    93→        resolve(value)
    94→      }
    95→    })
    96→  })
    97→}

    99→function odooExecuteKw(
   100→  model: string,
   101→  method: string,
   102→  args: any[],
   103→  kwargs: any = {}
   104→): Promise<any> {
   105→  return new Promise((resolve, reject) => {
   106→    if (!odooConfig.uid) {
   107→      reject(new Error('Not authenticated with Odoo'))
   108→      return
   109→    }
   110→
   111→    const client = getOdooClient()
   112→    client.methodCall(
   113→      'execute_kw',
   114→      [
   115→        odooConfig.db,
   116→        odooConfig.uid,
   117→        odooConfig.password,
   118→        model,
   119→        method,
   120→        args,
   121→        kwargs,
   122→      ],
   123→      (error: any, value: any) => {
   124→        if (error) {
   125→          reject(error)
   126→        } else {
   127→          resolve(value)
   128→        }
   129→      }
   130→    )
   131→  })
   132→}

   136→async function getAvailableFields(model: string): Promise<Set<string>> {
   137→  if (modelFieldsCache.has(model)) {
   138→    return modelFieldsCache.get(model)!
   139→  }
   140→
   141→  try {
   142→    const fields = await odooExecuteKw(model, 'fields_get', [], {
   143→      attributes: ['string', 'type'],
   144→    })
   145→    const fieldNames = new Set(Object.keys(fields))
   146→    modelFieldsCache.set(model, fieldNames)
   147→    console.log(`[Odoo] Model ${model} has ${fieldNames.size} fields. Custom checks: whatsapp=${fieldNames.has('whatsapp')}, whatsapp_number=${fieldNames.has('whatsapp_number')}`)
   148→    return fieldNames
   149→  } catch (error: any) {
   150→    console.error(`[Odoo] Failed to get fields for ${model}:`, error.message)
   151→    return new Set()
   152→  }
   153→}

   155→async function filterExistingFields(model: string, requestedFields: string[]): Promise<string[]> {
   156→  const available = await getAvailableFields(model)
   157→  const existing = requestedFields.filter(f => available.has(f))
   158→  if (!existing.includes('id') && available.has('id')) existing.unshift('id')
   159→  if (!existing.includes('name') && available.has('name')) existing.push('name')
   160→  return existing
   161→}

   163→async function buildSafeValues(model: string, values: Record<string, any>): Promise<Record<string, any>> {
   164→  const available = await getAvailableFields(model)
   165→  const safe: Record<string, any> = {}
   166→  for (const [key, value] of Object.entries(values)) {
   167→    if (available.has(key)) {
   168→      safe[key] = value
   169→    } else {
   170→      console.log(`[Odoo] Field "${key}" does not exist on ${model}, skipping`)
   171→    }
   172→  }
   173→  return safe
   174→}

   176→async function smartWriteWhatsAppNumber(model: string, ids: number[], phone: string): Promise<boolean> {
   177→  const available = await getAvailableFields(model)
   178→  const values: Record<string, any> = {}
   179→
   180→  if (available.has('whatsapp')) {
   181→    values.whatsapp = phone
   182→    console.log(`[Odoo] Using custom field "whatsapp" on ${model}`)
   183→  }
   184→  if (available.has('whatsapp_number')) {
   185→    values.whatsapp_number = phone
   186→    console.log(`[Odoo] Using custom field "whatsapp_number" on ${model}`)
   187→  }
   188→  if (available.has('phone')) {
   189→    values.phone = phone
   190→  }
   191→  if (available.has('mobile') && !values.whatsapp) {
   192→    values.mobile = phone
   193→  }
   194→
   195→  if (Object.keys(values).length === 0) {
   196→    console.log(`[Odoo] No phone/whatsapp fields found on ${model}, skipping write`)
   197→    return false
   198→  }
   199→
   200→  return odooWrite(model, ids, values)
   201→}

   205→async function odooSearch(
   206→  model: string,
   207→  domain: any[],
   208→  fields: string[] = [],
   209→  limit: number = 80,
   210→  offset: number = 0
   211→): Promise<any[]> {
   212→  const safeFields = fields.length > 0
   213→    ? await filterExistingFields(model, fields)
   214→    : []
   215→  return odooExecuteKw(model, 'search_read', [domain], {
   216→    fields: safeFields.length > 0 ? safeFields : undefined,
   217→    limit,
   218→    offset,
   219→  })
   220→}

   222→async function odooRead(
   223→  model: string,
   224→  ids: number[],
   225→  fields: string[] = []
   226→): Promise<any[]> {
   227→  const safeFields = fields.length > 0
   228→    ? await filterExistingFields(model, fields)
   229→    : []
   230→  return odooExecuteKw(model, 'read', [ids], {
   231→    fields: safeFields.length > 0 ? safeFields : undefined,
   232→  })
   233→}

   235→async function odooCreate(model: string, values: Record<string, any>): Promise<number> {
   236→  const safeValues = await buildSafeValues(model, values)
   237→  return odooExecuteKw(model, 'create', [safeValues])
   238→}

   240→async function odooWrite(model: string, ids: number[], values: Record<string, any>): Promise<boolean> {
   241→  const safeValues = await buildSafeValues(model, values)
   242→  return odooExecuteKw(model, 'write', [ids, safeValues])
   243→}

   245→async function odooSearchOrCreate(
   246→  model: string,
   247→  domain: any[],
   248→  values: Record<string, any>
   249→): Promise<{ id: number; created: boolean }> {
   250→  const existing = await odooExecuteKw(model, 'search', [domain], { limit: 1 })
   251→  if (existing && existing.length > 0) {
   252→    const safeValues = await buildSafeValues(model, values)
   253→    await odooWrite(model, existing, safeValues)
   254→    return { id: existing[0], created: false }
   255→  }
   256→  const safeValues = await buildSafeValues(model, values)
   257→  const newId = await odooCreate(model, safeValues)
   258→  return { id: newId, created: true }
   259→}

   325→async function autoSyncWhatsAppMessage(data: {
   326→  jid: string
   327→  phone: string
   328→  pushName?: string | null
   329→  textContent?: string | null
   330→  mediaType?: string | null
   331→  fromMe: boolean
   332→  timestamp: string
   333→}): Promise<{
   334→  partnerId: number | null
   335→  leadId: number | null
   336→  mailMessageId: number | null
   337→  activityId: number | null
   338→  created: { partner: boolean; lead: boolean }
   339→  errors: string[]
   340→}> {
   341→  const result = {
   342→    partnerId: null as number | null,
   343→    leadId: null as number | null,
   344→    mailMessageId: null as number | null,
   345→    activityId: null as number | null,
   346→    created: { partner: false, lead: false },
   347→    errors: [] as string[],
   348→  }
   349→
   350→  if (!autoSyncSettings.enabled || !odooConfig.uid) {
   351→    return result
   352→  }
   353→
   354→  console.log(`[AutoSync] Processing message from ${data.phone} (${data.pushName || 'unknown'})`)
   355→
   356→  try {
   357→    // Step 1: Create or update contact in res.partner
   358→    if (autoSyncSettings.autoCreateContact) {
   359→      const contactName = data.pushName || `WhatsApp ${data.phone}`
   360→      const domain = ['|', ['phone', 'ilike', data.phone], ['mobile', 'ilike', data.phone]]
   361→      const contactValues: Record<string, any> = {
   362→        name: contactName,
   363→        phone: data.phone,
   364→        mobile: data.phone,
   365→      }
   366→
   367→      // Try to set whatsapp field if it exists
   368→      const partnerFields = await getAvailableFields('res.partner')
   369→      if (partnerFields.has('whatsapp')) {
   370→        contactValues.whatsapp = data.phone
   371→      }
   372→      if (partnerFields.has('whatsapp_number')) {
   373→        contactValues.whatsapp_number = data.phone
   374→      }
   375→
   376→      const partnerResult = await odooSearchOrCreate('res.partner', domain, contactValues)
   377→      result.partnerId = partnerResult.id
   378→      result.created.partner = partnerResult.created
   379→
   380→      // Update cache
   381→      const cached = phoneToPartnerCache.get(data.phone)
   382→      if (cached) {
   383→        cached.partnerId = partnerResult.id
   384→      } else {
   385→        phoneToPartnerCache.set(data.phone, { partnerId: partnerResult.id, leadId: null, leadCreated: false })
   386→      }
   387→
   388→      console.log(`[AutoSync] Contact ${partnerResult.created ? 'created' : 'updated'}: res.partner#${partnerResult.id}`)
   389→    }
   390→
   391→    // Step 2: Create lead in crm.lead for new conversations (only for incoming messages)
   392→    if (autoSyncSettings.autoCreateLead && !data.fromMe && result.partnerId) {
   393→      // Check cache first
   394→      const cached = phoneToPartnerCache.get(data.phone)
   395→      if (cached && cached.leadId && cached.leadCreated) {
   396→        result.leadId = cached.leadId
   397→      } else {
   398→        // Check if there's already a lead for this partner with WhatsApp prefix
   399→        const existingLeads = await odooSearch('crm.lead', [
   400→          ['partner_id', '=', result.partnerId],
   401→          ['name', 'like', autoSyncSettings.leadPrefix],
   402→          ['type', '=', 'lead'],
   403→        ], ['id', 'name'], 1)
   404→
   405→        if (existingLeads && existingLeads.length > 0) {
   406→          result.leadId = existingLeads[0].id
   407→          // Update cache
   408→          if (cached) {
   409→            cached.leadId = existingLeads[0].id
   410→            cached.leadCreated = true
   411→          }
   412→          console.log(`[AutoSync] Found existing lead: crm.lead#${existingLeads[0].id}`)
   413→        } else {
   414→          // Create new lead
   415→          const leadName = `${autoSyncSettings.leadPrefix}${data.pushName || data.phone}`
   416→          const leadValues: Record<string, any> = {
   417→            name: leadName,
   418→            type: 'lead',
   419→            partner_id: result.partnerId,
   420→            phone: data.phone,
   421→            description: `Conversa iniciada via WhatsApp em ${new Date().toLocaleString('pt-BR')}`,
   422→          }
   423→
   424→          // Try to set custom whatsapp field
   425→          const leadFields = await getAvailableFields('crm.lead')
   426→          if (leadFields.has('whatsapp_number')) {
   427→            leadValues.whatsapp_number = data.phone
   428→          }
   429→
   430→          if (autoSyncSettings.leadTeamId) {
   431→            leadValues.team_id = autoSyncSettings.leadTeamId
   432→          }
   433→          if (autoSyncSettings.leadUserId) {
   434→            leadValues.user_id = autoSyncSettings.leadUserId
   435→          }
   436→
   437→          result.leadId = await odooCreate('crm.lead', leadValues)
   438→          result.created.lead = true
   439→
   440→          // Update cache
   441→          if (cached) {
   442→            cached.leadId = result.leadId
   443→            cached.leadCreated = true
   444→          } else {
   445→            phoneToPartnerCache.set(data.phone, { partnerId: result.partnerId, leadId: result.leadId, leadCreated: true })
   446→          }
   447→
   448→          console.log(`[AutoSync] Lead created: crm.lead#${result.leadId}`)
   449→        }
   450→      }
   451→    }
   452→
   453→    // Step 3: Post message as mail.message in Odoo chatter
   454→    if (autoSyncSettings.autoPostMessages && !data.fromMe) {
   455→      const targetModel = result.leadId ? 'crm.lead' : (result.partnerId ? 'res.partner' : null)
   456→      const targetId = result.leadId || result.partnerId
   457→
   458→      if (targetModel && targetId) {
   459→        const direction = data.fromMe ? 'Enviada' : 'Recebida'
   460→        const mediaLabel = data.mediaType ? ` [${data.mediaType}]` : ''
   461→        const msgBody = data.textContent
   462→          ? `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel}</p><p>${escapeHtml(data.textContent)}</p>`
   463→          : `<p><strong>📱 WhatsApp ${direction}:</strong>${mediaLabel} [Mídia]</p>`
   464→
   465→        try {
   466→          result.mailMessageId = await odooPostMessage(targetModel, targetId, msgBody)
   467→          console.log(`[AutoSync] Message posted on ${targetModel}#${targetId}: mail.message#${result.mailMessageId}`)
   468→        } catch (error: any) {
   469→          result.errors.push(`Failed to post message: ${error.message}`)
   470→          console.error(`[AutoSync] Failed to post message:`, error.message)
   471→        }
   472→      }
   473→    }
   474→
   475→    // Step 4: Create activity notification for the first message of a new lead
   476→    if (autoSyncSettings.autoCreateActivity && result.created.lead && result.leadId) {
   477→      const summary = 'Nova mensagem WhatsApp'
   478→      const note = `Contato ${data.pushName || data.phone} iniciou uma conversa via WhatsApp.\n\nMensagem: ${data.textContent || '[Mídia]'}`
   479→
   480→      try {
   481→        result.activityId = await odooCreateActivity('crm.lead', result.leadId, summary, note)
   482→        console.log(`[AutoSync] Activity created: mail.activity#${result.activityId}`)
   483→      } catch (error: any) {
   484→        result.errors.push(`Failed to create activity: ${error.message}`)
   485→      }
   486→    }
   487→
   488→  } catch (error: any) {
   489→    result.errors.push(`Auto-sync error: ${error.message}`)
   490→    console.error(`[AutoSync] Error:`, error.message)
   491→  }
   492→
   493→  // Emit sync result to all connected clients
   494→  io.emit('odoo:autosync:result', {
   495→    phone: data.phone,
   496→    ...result,
   497→  })
   498→
   499→  return result
   500→}

   502→function escapeHtml(text: string): string {
   503→  return text
   504→    .replace(/&/g, '&amp;')
   505→    .replace(/</g, '&lt;')
   506→    .replace(/>/g, '&gt;')
   507→    .replace(/\n/g, '<br/>')
   508→}
   509→
   510→function stripHtml(html: string): string {
   511→  // Replace <br/> and <br> and <p> with newlines first
   512→  let text = html
   513→    .replace(/<br\s*\/?>/gi, '\n')
   514→    .replace(/<\/p>/gi, '\n')
   515→    .replace(/<p[^>]*>/gi, '')
   516→    .replace(/<[^>]+>/g, '') // strip all remaining tags
   517→    .replace(/&nbsp;/g, ' ')
   518→    .replace(/&amp;/g, '&')
   519→    .replace(/&lt;/g, '<')
   520→    .replace(/&gt;/g, '>')
   521→    .replace(/&quot;/g, '"')
   522→    .replace(/&#39;/g, "'")
   523→    .trim()
   524→  // Collapse multiple blank lines
   525→  text = text.replace(/\n{3,}/g, '\n\n')
   526→  return text
   527→}

   589→        io.emit('odoo:autosync:settings', autoSyncSettings)
   590→        callback({ success: true, settings: autoSyncSettings })
   591→      } catch (error: any) {
   592→        callback({ success: false, error: error.message })
   593→      }
   594→    }

   763→  // ===== Fetch conversation history from Odoo chatter =====
   764→  // Reads mail.message records attached to a given model/record and converts
   765→  // them back to WhatsApp-style message objects so the frontend can merge
   766→  // them into the local conversation.
   767→  socket.on(
   768→    'odoo:fetch-history',
   769→    async (data: {
   770→      model: string
   771→      recordId: number
   772→      limit?: number
   773→    }, callback) => {
   774→      try {
   775→        if (!odooConfig.uid) {
   776→          callback({ success: false, error: 'Not authenticated with Odoo' })
   777→          return
   778→        }
   779→
   780→        // Read mail.message records linked to this record
   781→        const domain = [
   782→          ['model', '=', data.model],
   783→          ['res_id', '=', data.recordId],
   784→        ]
   785→        const messages = await odooExecuteKw('mail.message', 'search_read', [domain], {
   786→          fields: ['id', 'body', 'author_id', 'email_from', 'date', 'message_type', 'subtype_id', 'create_date'],
   787→          order: 'date asc',
   788→          limit: data.limit || 200,
   789→        })
   790→
   791→        // Convert each mail.message back to a WhatsApp-style message
   792→        // We try to detect direction (sent/received) and strip HTML
   793→        const converted = (messages || []).map((msg: any) => {
   794→          const body: string = msg.body || ''
   795→          // Strip HTML tags
   796→          const plainText = stripHtml(body)
   797→          // Detect direction from "Enviada" / "Recebida" markers we wrote earlier
   798→          const isSent = /WhatsApp\s+Enviada/i.test(body)
   799→          const isReceived = /WhatsApp\s+Recebida/i.test(body)
   800→
   801→          // If neither marker matches, this is some other chatter message (note, etc.)
   802→          // Treat as a received message by default
   803→          const fromMe = isSent
   804→
   805→          // Try to extract a timestamp — prefer the explicit date in body, fall back to msg.date
   806→          let timestamp: string
   807→          const dateMatch = body.match(/\(([^)]+)\)/)
   808→          if (dateMatch) {
   809→            const parsed = new Date(dateMatch[1])
   810→            if (!isNaN(parsed.getTime())) {
   811→              timestamp = parsed.toISOString()
   812→            } else {
   813→              timestamp = msg.date || msg.create_date || new Date().toISOString()
   814→            }
   815→          } else {
   816→            timestamp = msg.date || msg.create_date || new Date().toISOString()
   817→          }
   818→
   819→          return {
   820→            externalId: `odoo-${msg.id}`,
   821→            fromMe,
   822→            textContent: plainText || null,
   823→            mediaType: null,
   824→            timestamp,
   825→            source: 'odoo',
   826→          }
   827→        })
   828→
   829→        callback({ success: true, data: converted })
   830→      } catch (error: any) {
   831→        callback({ success: false, error: error.message })
   832→      }
   833→    }
   834→  )

  1078→// ========== Auto-Authenticate from Environment Variables ==========
  1079→async function autoAuthenticateFromEnv() {
  1080→  const envUrl = process.env.ODOO_URL
  1081→  const envDb = process.env.ODOO_DB
  1082→  const envUsername = process.env.ODOO_USERNAME
  1083→  const envPassword = process.env.ODOO_PASSWORD
  1084→
  1085→  if (envUrl && envDb && envUsername && envPassword) {
  1086→    console.log(`[Odoo] Auto-authenticating with env vars: ${envUrl} / ${envDb} / ${envUsername}`)
  1087→    try {
  1088→      odooConfig = {
  1089→        url: envUrl,
  1090→        db: envDb,
  1091→        username: envUsername,
  1092→        password: envPassword,
  1093→        uid: null,
  1094→      }
  1095→      modelFieldsCache.clear()
  1096→      phoneToPartnerCache.clear()
  1097→      const uid = await odooAuthenticate()
  1098→      odooConfig.uid = uid
  1099→      console.log(`[Odoo] Auto-authenticated as ${envUsername} (uid: ${uid})`)
  1100→
  1101→      // Pre-cache fields for main models
  1102→      await getAvailableFields('res.partner')
  1103→      await getAvailableFields('crm.lead')
  1104→
  1105→      io.emit('odoo:status', {
  1106→        connected: true,
  1107→        url: odooConfig.url,
  1108→        db: odooConfig.db,
  1109→        username: odooConfig.username,
  1110→      })
  1111→    } catch (error: any) {
  1112→      console.error(`[Odoo] Auto-authentication failed: ${error.message}`)
  1113→      console.error('[Odoo] You can still connect manually via the UI')
  1114→    }
  1115→  } else {
  1116→    console.log('[Odoo] No ODOO_URL/ODOO_DB/ODOO_USERNAME/ODOO_PASSWORD env vars set. Waiting for manual connection via UI.')
  1117→  }
  1118→}
