   389→function NavItem({ icon, label, active, onClick, badge, collapsed }: {
   390→  icon: React.ReactNode
   391→  label: string
   392→  active: boolean
   393→  onClick: () => void
   394→  badge?: string
   395→  collapsed?: boolean
   396→}) {
   397→  return (
   398→    <button
   399→      onClick={onClick}
   400→      className={cn(
   401→        'w-full flex items-center gap-3 px-2 py-2 rounded-lg text-sm transition-colors',
   402→        'hover:bg-muted/80 focus-visible:outline-none focus-visible:bg-muted/80',
   403→        active ? 'bg-primary/10 text-primary font-medium' : 'text-muted-foreground',
   404→      )}
   405→      title={collapsed ? label : undefined}
   406→    >
   407→      <span className="shrink-0 mx-auto lg:mx-0">{icon}</span>
   408→      {!collapsed && <span className="hidden lg:block flex-1 text-left truncate">{label}</span>}
   409→      {badge && !collapsed && (
   410→        <Badge className="size-5 p-0 flex items-center justify-center text-[10px] bg-emerald-500 text-white border-0">
   411→          {badge === 'online' ? '\u25CF' : badge}
   412→        </Badge>
   413→      )}
   414→      {badge && collapsed && (
   415→        <Badge className="hidden lg:flex absolute top-0 right-0 size-2 p-0 bg-emerald-500 border-0" />
   416→      )}
   417→    </button>
   418→  )
   419→}

   442→  // v7.22: Auth guard — if not authenticated, redirect to /login

   444→    if (!authLoading && !isAuthenticated) {
   445→      router.replace('/login')

   447→  }, [authLoading, isAuthenticated, router])

   449→  // v7.32: Auto-request QR when user opens the WhatsApp tab and there's no
   450→  // connection and no QR yet. Previously the user had to manually click
   451→  // "Solicitar QR Code" every time — now the QR is requested automatically
   452→  // the moment they navigate to the WhatsApp tab (if needed).
   453→  // Runs only when: (a) user just switched to the whatsapp tab, (b) socket
   454→  // is connected to the server, (c) WA is not yet connected, (d) no QR
   455→  // is currently displayed. The check is debounced 1.5s to let the
   456→  // onWAConnection initial status arrive first.
   457→  useEffect(() => {
   458→    if (activeTab !== 'whatsapp') return
   459→    if (!isAuthenticated) return
   460→    if (wa.status.connected) return
