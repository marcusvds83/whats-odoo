    90→const EMPTY_EDIT: EditState = {

   103→export function UsersPanel() {
   104→  const [users, setUsers] = useState<UserRow[]>([])
   105→  const [isLoading, setIsLoading] = useState(true)
   106→  const [error, setError] = useState<string | null>(null)
   107→  const [dialogOpen, setDialogOpen] = useState(false)
   108→  const [editState, setEditState] = useState<EditState>(EMPTY_EDIT)
   109→  const [isSaving, setIsSaving] = useState(false)
   110→  // v7.24 (R6): pre-deploy backup state
   111→  const odoo = useOdoo()
   112→  const [isBackingUp, setIsBackingUp] = useState(false)
   113→  const [backupResult, setBackupResult] = useState<{
   114→    success: boolean
   115→    backed: number
   116→    total: number
   117→    failed: Array<{ userId: string; email: string; error: string }>
   118→    at: string
   119→  } | null>(null)
   120→  // v7.26: Test-login state
   121→  const [testLoginUser, setTestLoginUser] = useState<UserRow | null>(null)
   122→  const [testLoginPassword, setTestLoginPassword] = useState('')
   123→  const [testLoginResult, setTestLoginResult] = useState<null | {
   124→    found: boolean
   125→    isActive: boolean
   126→    role: string
   127→    passwordOk: boolean
   128→    hashPrefix: string
   129→    hashLength: number
   130→    providedHashPreview: string
   131→  }>(null)
   132→  const [isTestingLogin, setIsTestingLogin] = useState(false)
   133→  const [showPassword, setShowPassword] = useState(false)
   134→  // v7.29: "Alterar senha" dialog state
   135→  const [pwdUser, setPwdUser] = useState<UserRow | null>(null)
   136→  const [newPassword, setNewPassword] = useState('')
   137→  const [showNewPassword, setShowNewPassword] = useState(false)
   138→  const [isChangingPwd, setIsChangingPwd] = useState(false)
   139→  const [pwdCopied, setPwdCopied] = useState(false)
   140→
   141→  // v7.31: Firebase/Firestore status — surfaced in the admin UI so the
   142→  // admin can see at-a-glance whether user records are being persisted
   143→  // to Firestore (survives Render sleeps/deploys) or are stuck in SQLite
   144→  // (wiped on every Render deploy). Defined here, populated by
   145→  // loadFirebaseStatus() below (after loadUsers is declared).
   146→  const [firebaseStatus, setFirebaseStatus] = useState<null | {
   147→    loading: boolean
   148→    configured: boolean
   149→    initialized: boolean
   150→    configSource: string | null
   151→    initError: string | null
   152→    firestoreUsers: number
   153→    sqliteUsers: number
   154→    sample: string[]
   155→    recommendations: string[]
   156→  }>(null)
   157→  const [isMigrating, setIsMigrating] = useState(false)
   158→  const [migrationResult, setMigrationResult] = useState<null | {
   159→    success: boolean
   160→    migrated: number
   161→    skipped: number
   162→    errors: string[]
   163→  }>(null)
   164→
   165→  const loadUsers = useCallback(async () => {
   166→    setIsLoading(true)
   167→    setError(null)
   168→    try {
   169→      const res = await fetch('/api/users', { cache: 'no-store' })
   170→      const data = await res.json()
   171→      if (data.success) {
   172→        setUsers(data.users)
   173→      } else {
   174→        setError(data.error || 'Falha ao carregar usuários')
   175→      }
   176→    } catch (err: any) {
   177→      setError(err.message)
   178→    } finally {
   179→      setIsLoading(false)
   180→    }
   181→  }, [])
   182→
   183→  useEffect(() => {
   184→    loadUsers()
   185→  }, [loadUsers])
   186→
   187→  // v7.31: Load Firebase/Firestore status from /api/auth/debug.
   188→  // Called on mount and via the "Atualizar status" button in the UI.
   189→  const loadFirebaseStatus = useCallback(async () => {
   190→    setFirebaseStatus(prev => prev
   191→      ? { ...prev, loading: true }
   192→      : { loading: true, configured: false, initialized: false, configSource: null, initError: null, firestoreUsers: 0, sqliteUsers: 0, sample: [], recommendations: [] })
   193→    try {
   194→      const res = await fetch('/api/auth/debug', { cache: 'no-store' })
   195→      const data = await res.json()
   196→      if (data.success) {
   197→        setFirebaseStatus({
   198→          loading: false,
   199→          configured: data.firebase?.configured ?? false,
   200→          initialized: data.firebase?.initialized ?? false,
   201→          configSource: data.firebase?.configSource ?? null,
   202→          initError: data.firebase?.initError ?? null,
   203→          firestoreUsers: data.firestore?.userCount ?? 0,
   204→          sqliteUsers: data.sqlite?.userCount ?? 0,
   205→          sample: data.firestore?.sampleUsers ?? [],
   206→          recommendations: data.recommendations ?? [],
   207→        })
   208→      } else {
   209→        setFirebaseStatus({
   210→          loading: false, configured: false, initialized: false, configSource: null,
   211→          initError: data.error || 'Failed to load', firestoreUsers: 0, sqliteUsers: 0,
   212→          sample: [], recommendations: [],
   213→        })
   214→      }
   215→    } catch (err: any) {
   216→      setFirebaseStatus({
   217→        loading: false, configured: false, initialized: false, configSource: null,
   218→        initError: err.message, firestoreUsers: 0, sqliteUsers: 0,
   219→        sample: [], recommendations: [],
   220→      })
   221→    }
   222→  }, [])
   223→
   224→  // v7.31: Manually trigger SQLite → Firestore migration.
   225→  const runMigration = useCallback(async () => {
   226→    if (isMigrating) return
   227→    setIsMigrating(true)
   228→    setMigrationResult(null)
   229→    try {
   230→      const res = await fetch('/api/auth/migrate-users', { method: 'POST' })
   231→      const data = await res.json()
   232→      setMigrationResult({
   233→        success: !!data.success,
   234→        migrated: data.migrated ?? 0,
   235→        skipped: data.skipped ?? 0,
   236→        errors: data.errors ?? [],
   237→      })
   238→      await loadFirebaseStatus()
   239→      await loadUsers()
   240→    } catch (err: any) {
   241→      setMigrationResult({
   242→        success: false, migrated: 0, skipped: 0, errors: [err.message],
   243→      })
   244→    } finally {
   245→      setIsMigrating(false)
   246→    }
   247→  }, [isMigrating, loadFirebaseStatus, loadUsers])
   248→
   249→  useEffect(() => {
   250→    loadFirebaseStatus()
   251→  }, [loadFirebaseStatus])
   252→
   253→  // v7.24 (R6): Backup all active user sessions to Odoo chatter.
   254→  // Should be triggered manually BEFORE any deploy.
   255→  const handleBackupToOdoo = useCallback(async () => {
   256→    if (isBackingUp) return
   257→    setIsBackingUp(true)
   258→    setBackupResult(null)
   259→    try {
   260→      const r = await odoo.backupAllToOdoo()
   261→      if (r.success) {
   262→        setBackupResult({
   263→          success: true,
   264→          backed: r.backed || 0,
   265→          total: r.total || 0,
   266→          failed: r.failed || [],
   267→          at: new Date().toISOString(),
   268→        })
   269→      } else {
   270→        setBackupResult({
   271→          success: false,
   272→          backed: r.backed || 0,
   273→          total: r.total || 0,
   274→          failed: r.failed || [{ userId: '-', email: '-', error: r.error || 'Falha desconhecida' }],
   275→          at: new Date().toISOString(),
   276→        })
   277→      }
   278→    } catch (err: any) {
   279→      setBackupResult({
   280→        success: false,
   281→        backed: 0,
   282→        total: 0,
   283→        failed: [{ userId: '-', email: '-', error: err.message }],
   284→        at: new Date().toISOString(),
   285→      })
   286→    } finally {
   287→      setIsBackingUp(false)
   288→    }
   289→  }, [isBackingUp, odoo])
   290→
   291→  const handleCreate = () => {
   292→    setEditState(EMPTY_EDIT)
   293→    setDialogOpen(true)
   294→  }
   295→
   296→  const handleEdit = (user: UserRow) => {
   297→    setEditState({
   298→      id: user.id,
   299→      email: user.email,
   300→      name: user.name || '',
   301→      password: '',
   302→      role: user.role as 'user' | 'admin',
   303→      isActive: user.isActive,
   304→      whatsappPhone: user.whatsappPhone || '',
   305→      odooUrl: user.odooUrl || '',
   306→      odooDb: user.odooDb || '',
   307→      odooUsername: user.odooUsername || '',
   308→      odooPassword: '',
   309→    })
   310→    setDialogOpen(true)
   311→  }
   312→
   313→  const handleSave = async () => {
   314→    setIsSaving(true)
   315→    setError(null)
   316→    try {
   317→      const isEditing = !!editState.id
   318→      const url = isEditing ? `/api/users/${editState.id}` : '/api/users'
   319→      const method = isEditing ? 'PATCH' : 'POST'
   320→
   321→      const body: any = {
   322→        email: editState.email,
   323→        name: editState.name,
   324→        role: editState.role,
   325→        isActive: editState.isActive,
   326→        whatsappPhone: editState.whatsappPhone,
   327→        odooUrl: editState.odooUrl,
   328→        odooDb: editState.odooDb,
   329→        odooUsername: editState.odooUsername,
   330→      }
   331→      if (editState.password) body.password = editState.password
   332→      if (editState.odooPassword) body.odooPassword = editState.odooPassword
   333→
   334→      const res = await fetch(url, {
   335→        method,
   336→        headers: { 'Content-Type': 'application/json' },
   337→        body: JSON.stringify(body),
   338→      })
   339→      const data = await res.json()
   340→      if (data.success) {
   341→        setDialogOpen(false)
   342→        await loadUsers()
   343→      } else {
   344→        setError(data.error || 'Falha ao salvar')
   345→      }
   346→    } catch (err: any) {
   347→      setError(err.message)
   348→    } finally {
   349→      setIsSaving(false)
   350→    }
   351→  }
   352→
   353→  // v7.26: Generate a random strong password and put it into the edit form
   354→  const generatePassword = () => {
   355→    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789!@#$%&*'
   356→    let pwd = ''
   357→    const arr = new Uint32Array(12)
   358→    crypto.getRandomValues(arr)
   359→    for (let i = 0; i < 12; i++) pwd += chars[arr[i] % chars.length]
   360→    setEditState(s => ({ ...s, password: pwd }))
   361→    setShowPassword(true)
   362→  }
   363→
   364→  // v7.26: Test a user's password without impersonating them
   365→  const handleTestLogin = async () => {
   366→    if (!testLoginUser || !testLoginPassword) return
   367→    setIsTestingLogin(true)
   368→    setTestLoginResult(null)
   369→    try {
   370→      const res = await fetch(`/api/users/${testLoginUser.id}/test-login`, {
   371→        method: 'POST',
   372→        headers: { 'Content-Type': 'application/json' },
   373→        body: JSON.stringify({ password: testLoginPassword }),
   374→      })
   375→      const data = await res.json()
   376→      if (data.success) {
   377→        setTestLoginResult({
   378→          found: data.found,
   379→          isActive: data.isActive,
   380→          role: data.role,
   381→          passwordOk: data.passwordOk,
   382→          hashPrefix: data.hashPrefix,
   383→          hashLength: data.hashLength,
   384→          providedHashPreview: data.providedHashPreview,
   385→        })
   386→      } else {
   387→        setTestLoginResult({
   388→          found: false,
   389→          isActive: false,
   390→          role: '?',
   391→          passwordOk: false,
   392→          hashPrefix: 'error: ' + (data.error || 'unknown'),
   393→          hashLength: 0,
   394→          providedHashPreview: '',
   395→        })
   396→      }
   397→    } catch (err: any) {
   398→      setTestLoginResult({
   399→        found: false,
   400→        isActive: false,
   401→        role: '?',
   402→        passwordOk: false,
   403→        hashPrefix: 'fetch error: ' + err.message,
   404→        hashLength: 0,
   405→        providedHashPreview: '',
   406→      })
   407→    } finally {
   408→      setIsTestingLogin(false)
   409→    }
   410→  }
   411→
   412→  const handleDelete = async (user: UserRow) => {
   413→    if (!confirm(`Excluir o usuário "${user.email}"? Esta ação não pode ser desfeita.`)) return
   414→    try {
   415→      const res = await fetch(`/api/users/${user.id}`, { method: 'DELETE' })
   416→      const data = await res.json()
   417→      if (data.success) {
   418→        await loadUsers()
   419→      } else {
   420→        setError(data.error || 'Falha ao excluir')
   421→      }
   422→    } catch (err: any) {
   423→      setError(err.message)
   424→    }
   425→  }
   426→
   427→  const activeCount = users.filter(u => u.isActive).length
   428→  const adminCount = users.filter(u => u.role === 'admin').length
   429→
   430→  // v7.29: Open the "Alterar senha" dialog for a user
   431→  const openChangePassword = (user: UserRow) => {
   432→    setPwdUser(user)
   433→    setNewPassword('')
   434→    setShowNewPassword(false)
   435→    setPwdCopied(false)
   436→  }
   437→
   438→  // v7.29: Save a new password for the user (hashed on the server)
   439→  const handleChangePassword = async () => {
   440→    if (!pwdUser || !newPassword.trim()) return
   441→    setIsChangingPwd(true)
   442→    setError(null)
   443→    try {
   444→      const res = await fetch(`/api/users/${pwdUser.id}`, {
   445→        method: 'PATCH',
   446→        headers: { 'Content-Type': 'application/json' },
   447→        body: JSON.stringify({ password: newPassword.trim() }),
   448→      })
   449→      const data = await res.json()
   450→      if (data.success) {
   451→        setPwdUser(null)
   452→        await loadUsers()
   453→      } else {
   454→        setError(data.error || 'Falha ao alterar a senha')
   455→      }
   456→    } catch (err: any) {
   457→      setError(err.message)
   458→    } finally {
   459→      setIsChangingPwd(false)
   460→    }
   461→  }
   462→
   463→  const copyNewPassword = async () => {
   464→    try {
   465→      await navigator.clipboard?.writeText(newPassword)
   466→      setPwdCopied(true)
   467→      setTimeout(() => setPwdCopied(false), 2000)
   468→    } catch {}
   469→  }
   470→
   471→  return (
   472→    <div className="space-y-4">
   473→      <div className="flex items-center justify-between gap-3">
   474→        <div>
   475→          <h2 className="text-xl font-bold flex items-center gap-2">
   476→            <UserIcon className="size-5" />
   477→            Usuários do Middleware
   478→          </h2>
   479→          <p className="text-sm text-muted-foreground mt-1">
   480→            {users.length} total • {activeCount} ativos • {adminCount} admin
   481→          </p>
   482→        </div>
   483→        <div className="flex gap-2">
   484→          <Button variant="outline" size="sm" onClick={loadUsers} disabled={isLoading}>
   485→            <RefreshCw className={`size-4 mr-1.5 ${isLoading ? 'animate-spin' : ''}`} />
   486→            Atualizar
   487→          </Button>
   488→          {/* v7.24 (R6): Pre-deploy backup button */}
   489→          <Button
   490→            variant="outline"
   491→            size="sm"
   492→            onClick={handleBackupToOdoo}
   493→            disabled={isBackingUp || !odoo.isConnected}
   494→            title={odoo.isConnected ? 'Salva credenciais e conversas de todos os usuários no chatter do Odoo (use antes de deploy)' : 'Conecte ao Odoo primeiro'}
   495→          >
   496→            {isBackingUp ? <Loader2 className="size-4 mr-1.5 animate-spin" /> : <Database className="size-4 mr-1.5" />}
   497→            {isBackingUp ? 'Backup em andamento…' : 'Backup no Odoo'}
   498→          </Button>
   499→          <Button size="sm" onClick={handleCreate} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   500→            <UserPlus className="size-4 mr-1.5" />
   501→            Novo Usuário
   502→          </Button>
   503→        </div>
   504→      </div>
   505→
   506→      {/* v7.24 (R6): Backup result banner */}
   507→      {backupResult && (
   508→        <Alert variant={backupResult.success ? 'default' : 'destructive'}>
   509→          {backupResult.success
   510→            ? <CheckCircle2 className="size-4" />
   511→            : <AlertCircle className="size-4" />}
   512→          <AlertDescription>
   513→            {backupResult.success
   514→              ? `Backup concluído: ${backupResult.backed}/${backupResult.total} usuário(s) salvos no chatter do Odoo.`
   515→              : `Backup parcial: ${backupResult.backed}/${backupResult.total} ok, ${backupResult.failed.length} falha(s).`}
   516→            {backupResult.failed.length > 0 && (
   517→              <ul className="mt-1 ml-4 text-xs list-disc">
   518→                {backupResult.failed.map((f, i) => (
   519→                  <li key={i}>{f.email}: {f.error}</li>
   520→                ))}
   521→              </ul>
   522→            )}
   523→            <span className="block mt-1 text-xs text-muted-foreground">
   524→              Realizado em {new Date(backupResult.at).toLocaleString('pt-BR')}
   525→            </span>
   526→          </AlertDescription>
   527→        </Alert>
   528→      )}
   529→
   530→      {error && (
   531→        <Alert variant="destructive">
   532→          <AlertDescription>{error}</AlertDescription>
   533→        </Alert>
   534→      )}
   535→
   536→      {/* v7.31: Firebase / Firestore status banner.
   537→          -------------------------------------------------------------
   538→          Surfaces the actual Firebase init state to the admin so they can
   539→          see WHY users aren't appearing in the Firebase console. The most
   540→          common cause is: Firebase env vars aren't set on Render, so the
   541→          userStore silently falls back to SQLite — users "work" but they
   542→          don't land in Firestore, so they vanish on every Render restart.
   543→
   544→          This banner shows:
   545→          - Whether Firebase env vars are present (configured)
   546→          - Whether the Admin SDK actually initialized (initialized)
   547→          - The init error message (if any) — this is the key debug info
   548→          - User counts in Firestore vs SQLite
   549→          - A manual "Migrar agora" button to trigger the SQLite → Firestore
   550→            backfill via /api/auth/migrate-users
   551→          ------------------------------------------------------------- */}
   552→      <Card className="border-amber-200 dark:border-amber-900/50 bg-amber-50/40 dark:bg-amber-950/10">
   553→        <CardHeader className="pb-3">
   554→          <div className="flex items-center justify-between gap-2 flex-wrap">
   555→            <CardTitle className="text-base flex items-center gap-2">
   556→              {firebaseStatus?.loading ? (
   557→                <Loader2 className="size-4 animate-spin text-muted-foreground" />
   558→              ) : firebaseStatus?.configured && firebaseStatus?.initialized ? (
   559→                <Cloud className="size-4 text-emerald-600" />
   560→              ) : firebaseStatus?.configured && !firebaseStatus?.initialized ? (
   561→                <AlertCircle className="size-4 text-amber-600" />
   562→              ) : (
   563→                <CloudOff className="size-4 text-amber-600" />
   564→              )}
   565→              Status do Firebase / Firestore
   566→            </CardTitle>
   567→            <Button
   568→              variant="outline"
   569→              size="sm"
   570→              onClick={loadFirebaseStatus}
   571→              disabled={firebaseStatus?.loading}
   572→            >
   573→              <RefreshCw className={`size-3.5 mr-1.5 ${firebaseStatus?.loading ? 'animate-spin' : ''}`} />
   574→              Atualizar status
   575→            </Button>
   576→          </div>
   577→        </CardHeader>
   578→        <CardContent className="space-y-3 text-sm">
   579→          {!firebaseStatus ? (
   580→            <p className="text-muted-foreground">Carregando status...</p>
   581→          ) : (
   582→            <>
